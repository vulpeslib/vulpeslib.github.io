66template <typename Container>
67std::ostream& print(
68 std::ostream& os,
69 const Container& container,
70 const std::string& separator = ",",
71 const std::string& postpend = "\n",
72 const std::string& prepend = "") {
73 auto it = container.begin();
74 if (!prepend.empty()) {
75 os << prepend;
76 }
77 if (it != container.end()) {
78 os << *it;
79 ++it;
80 for (; it != container.end(); ++it) {
81 os << separator << *it;
82 }
83 }
84 if (!postpend.empty()) {
85 os << postpend;
86 }
87 return os;
88}

116template <typename T, typename IndexType>
117std::vector<T> slice(
118 const std::vector<T>& source, const std::vector<IndexType>& indices) {
119 std::vector<T> result;
120 result.reserve(indices.size());
121 for (size_t index : indices) {
122 result.push_back(source[index]);
123 }
124 return result;
125}

127// Implementation helper for for_each(): applies func to every tuple
128// element.
129template <typename Tuple, typename Func, std::size_t... Is>
130void for_each_impl(Tuple&& tuple, Func&& func, std::index_sequence<Is...>) {
131 (func(std::get<Is>(tuple)), ...); // Fold expression (C++17)
132}

141template <typename Tuple, typename Func>
142void for_each(Tuple&& tuple, Func&& func) {

145 std::make_index_sequence<std::tuple_size<std::decay_t<Tuple>>::value>{});
146}
