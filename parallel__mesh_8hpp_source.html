497template <typename Mesh>
498std::vector<SCOTCH_Num> ParallelMesh<Mesh>::partitionWithParmetis() {
499 using real_t = float;
500 const auto rank = mpi::rank();
501 const auto size = mpi::size();
502 // how many vertices on each proc
503 std::vector<SCOTCH_Num> vtxdist(size + 1);
504 // crs map in Metis format
505 std::vector<SCOTCH_Num> xadj;
506 std::vector<SCOTCH_Num> adjncy;
507
508 if (rank == 0) {
509 const auto& dual = m_rank0_mesh->getDualMap();
510
511 // how many vertices on each proc
512 vtxdist[0] = 0;
513 for (int i = 0; i < size; ++i) {
514 vtxdist[i + 1] =
515 vtxdist[i] + dual.size() / size + (i < (dual.size() % size));
516 }
517 assert(vtxdist.back() == dual.size());
518
519 // build local crs map for others
520 for (int p = 1; p < size; ++p) {
521 xadj.resize(vtxdist[p + 1] - vtxdist[p] + 1);
522 const std::size_t adjncy_size =
523 dual.indices()[vtxdist[p + 1]] - dual.indices()[vtxdist[p]];
524 const auto offset = dual.indices()[vtxdist[p]];
525 for (std::size_t j = 0; j < xadj.size(); ++j) {
526 xadj[j] = dual.indices()[j + vtxdist[p]] - offset;
527 }
528 mpi::send(xadj, p);
529 mpi::send(adjncy_size, p);
530 mpi::send(&dual.data()[offset], adjncy_size, p);
531 }
532
533 // build local crs map for self
534 xadj.assign(
535 dual.indices().begin(), dual.indices().begin() + vtxdist[1] + 1);
536 adjncy.assign(
537 dual.data().begin(), dual.data().begin() + dual.indices()[vtxdist[1]]);
538 } else {
539 mpi::recv(xadj, 0);
540 adjncy.resize(mpi::recv<std::size_t>(0));
541 mpi::recv(adjncy.data(), static_cast<int32_t>(adjncy.size()), 0);
542 }
543 mpi::bcast(vtxdist, 0);
544
545 // ParMETIS parameters
546 SCOTCH_Num ncon = 1;
547 SCOTCH_Num wgtflag = 0;
548 SCOTCH_Num numflag = 0;
549 SCOTCH_Num nparts = size;
550 real_t ubvec = 1.05;
551 const SCOTCH_Num random_seed = 0;
552 SCOTCH_Num options[3] = {1, 0, random_seed};
553 std::vector<real_t> tpwgts(nparts);
554 for (int i = 0; i < size; ++i) {
555 tpwgts[i] = 1. / static_cast<double>(nparts);
556 }
557
558 // result buffer
559 std::vector<SCOTCH_Num> part(vtxdist[rank + 1] - vtxdist[rank]);
560
561 std::cout << "Partition the graph;" << std::endl;
562 // Partition the graph
563 auto comm = MPI_COMM_WORLD;
564 ParMETIS_V3_PartKway(
565 vtxdist.data(), xadj.data(), adjncy.data(), nullptr, nullptr, &wgtflag,
566 &numflag, &ncon, &nparts, tpwgts.data(), &ubvec, options, &m_edgecut,
567 part.data(), &comm);
568 return part;
569}

571template <typename Mesh>
572void ParallelMesh<Mesh>::reconstructPartitioningAndScatterDual(
573 std::vector<index_t>& part) {
574 const auto size = mpi::size();
575 const auto& dual = m_rank0_mesh->getDualMap();
576 std::vector<index_t> all_parts;
577 if (mpi::root()) {
578 std::cout << "Partitioning completed. Edge cuts: " << m_edgecut
579 << std::endl;
580 // std::cout << "rank 0 | part: " << part << std::endl;
581 all_parts = part;
582 for (int p = 1; p < size; ++p) {
583 mpi::recv(part, p);
584 // std::cout << "rank " << p << " | part: " << part << std::endl;
585 all_parts.insert(all_parts.end(), part.begin(), part.end());
586 }
587
588 // we re-order the partition to minimize the bandwidth
589 // of the processors block
590 // In practice if two partitions are connected they should
591 // belong also to procs that are close together
592 Eigen::MatrixXi dense_proc_adj = Eigen::MatrixXi::Zero(size, size);
593 for (index_t cell_id = 0; cell_id < dual.size(); ++cell_id) {
594 const auto src_cell_proc = all_parts[cell_id];
595 for (index_t neigh_cell_id : dual[cell_id]) {
596 const auto dst_cell_proc = all_parts[neigh_cell_id];
597 dense_proc_adj(src_cell_proc, dst_cell_proc) = 1;
598 }
599 }
600 datastructure::CrsMapUncompressed<index_t> uncomp_proc_adj;
601 uncomp_proc_adj.resize(size);
602 for (Eigen::Index i = 0; i < dense_proc_adj.rows(); ++i) {
603 for (Eigen::Index j = 0; j < dense_proc_adj.cols(); ++j) {
604 if (dense_proc_adj(i, j)) {
605 uncomp_proc_adj[i].push_back(j);
606 }
607 }
608 }
609 const auto proc_adj = uncomp_proc_adj.template compress<index_t>();
610 const auto [perm, perm_inv] =
611 dofs::computeReverseCuthillMckeeOrdering(proc_adj, false);
612 for (std::size_t i = 0; i < all_parts.size(); ++i) {
613 all_parts[i] = perm[all_parts[i]];
614 }
615
616 for (std::size_t cell_id = 0; cell_id < all_parts.size(); ++cell_id) {
617 m_all_own_and_ghost_cell_id[all_parts[cell_id]].push_back(cell_id);
618 }
619
620 for (int p = 0; p < size; ++p) {
621 m_all_own_size[p] = m_all_own_and_ghost_cell_id[p].size();
622 }
623
624 // send dual map of owned cells
625 for (int p = 1; p < size; ++p) {
626 mpi::send(m_all_own_and_ghost_cell_id[p], p);
627 mpi::send(dual.slice(m_all_own_and_ghost_cell_id[p]), p);
628 }
629 m_local_numbering[self_dim] = m_all_own_and_ghost_cell_id[0];
630 this->getDualMap() = dual.slice(m_all_own_and_ghost_cell_id[0]);
631 } else {
632 mpi::send(part, 0);
633 mpi::recv(m_local_numbering[self_dim], 0);
634 mpi::recv(this->getDualMap(), 0);
635 }
636}

638template <typename Mesh>
639void ParallelMesh<Mesh>::buildGhostCellsFromWeakDual() {
640 const auto size = mpi::size();
641 SetType ghost_cell_set(size);
642 for (index_t loc_cid = 0; loc_cid < m_weak_dual.size(); ++loc_cid) {
643 for (auto glob_neigh_id : m_weak_dual[loc_cid]) {
644 auto it = std::lower_bound(
645 m_local_numbering[self_dim].begin(),
646 m_local_numbering[self_dim].end(), glob_neigh_id);
647 if ((it == m_local_numbering[self_dim].end()) || (*it != glob_neigh_id)) {
648 ghost_cell_set.insert(glob_neigh_id);
649 }
650 }
651 }
652 m_own_enty_size[self_dim] = m_local_numbering[self_dim].size();
653 m_ghost_enty_size[self_dim] = ghost_cell_set.size();
654 m_local_numbering[self_dim].insert(
655 m_local_numbering[self_dim].end(), ghost_cell_set.begin(),
656 ghost_cell_set.end());
657}

659template <typename Mesh>
660void ParallelMesh<Mesh>::collectGhostCellsInRoot() {
661 const auto size = mpi::size();
662 if (mpi::root()) {
663 // recive ghost from procs
664 m_all_own_and_ghost_cell_id[0] = m_local_numbering[self_dim];
665 m_all_ghost_size[0] =
666 m_all_own_and_ghost_cell_id[0].size() - m_all_own_size[0];
667 for (int p = 1; p < size; ++p) {
668 m_all_ghost_size[p] = mpi::recv<index_t>(p);
669 m_all_own_and_ghost_cell_id[p].resize(
670 m_all_own_and_ghost_cell_id[p].size() + m_all_ghost_size[p]);
671 mpi::recv(
672 &m_all_own_and_ghost_cell_id[p][m_all_own_size[p]],
673 m_all_ghost_size[p], p);
674 }
675 } else {
676 mpi::send(m_ghost_enty_size[self_dim], 0);
677 mpi::send(
678 &m_local_numbering[self_dim][m_own_enty_size[self_dim]],
679 m_ghost_enty_size[self_dim], 0, MPI_COMM_WORLD);
680 }
681}

683template <typename Mesh>
684void ParallelMesh<Mesh>::buildGhostCells(std::vector<index_t>& part) {
685 const auto size = mpi::size();
686 m_all_own_and_ghost_cell_id.resize(size);
687 m_all_own_size.resize(size);
688 m_all_ghost_size.resize(size);
689 // send partitions to root and compute ghost
690 // reconstruct partitioning and send dual map owning slice
691 PROFILE_INTERVAL(reconstructPartitioningAndScatterDual(part));
692
693 // In the ghost cells we want also the weakly connected cells
694 // that is the cells that share also only a vertex
695 // Indeed, we need this information to compute the sparsity pattern
696 PROFILE_INTERVAL(buildWeakDual());
697 PROFILE_INTERVAL(buildGhostCellsFromWeakDual());
698 PROFILE_INTERVAL(collectGhostCellsInRoot());
699}

701template <typename Mesh>
702void ParallelMesh<Mesh>::buildLocalCellToEntityMaps() {
703 const auto size = mpi::size();
704 // build and send local maps
705 if (mpi::root()) {
706 for (int p = 1; p < size; ++p) {
707 mpi::send(
708 m_rank0_mesh->getDualMap().slice(m_all_own_and_ghost_cell_id[p]), p);
709 const auto cell_to_vtx_loc =
710 m_rank0_mesh->getCellToVtxMap().slice(m_all_own_and_ghost_cell_id[p]);
711 mpi::send(cell_to_vtx_loc, p);
712 const auto cell_to_facet_loc = m_rank0_mesh->getCellToFacetMap().slice(
713 m_all_own_and_ghost_cell_id[p]);
714 mpi::send(cell_to_facet_loc, p);
715 if constexpr (self_dim == 3) {
716 const auto cell_to_edge_loc = m_rank0_mesh->getCellToEdgeMap().slice(
717 m_all_own_and_ghost_cell_id[p]);
718 mpi::send(cell_to_edge_loc, p);
719 }
720 }
721 // build for self
722 this->getDualMap() =
723 m_rank0_mesh->getDualMap().slice(m_all_own_and_ghost_cell_id[0]);
724 this->getCellToVtxMap() =
725 m_rank0_mesh->getCellToVtxMap().slice(m_all_own_and_ghost_cell_id[0]);
726 this->getCellToFacetMap() =
727 m_rank0_mesh->getCellToFacetMap().slice(m_all_own_and_ghost_cell_id[0]);
728 if constexpr (self_dim == 3) {
729 this->getCellToEdgeMap() = m_rank0_mesh->getCellToEdgeMap().slice(
730 m_all_own_and_ghost_cell_id[0]);
731 }
732 } else {
733 mpi::recv(this->getDualMap(), 0);
734 mpi::recv(this->getCellToVtxMap(), 0);
735 mpi::recv(this->getCellToFacetMap(), 0);
736 if constexpr (self_dim == 3) {
737 mpi::recv(this->getCellToEdgeMap(), 0);
738 }
739 }

742template <typename Mesh>
743void ParallelMesh<Mesh>::buildWeakDual() {
744 const auto size = mpi::size();
745 // scatter the cell to vtx map
746 if (mpi::root()) {
747 for (int p = 0; p < size; ++p) {
748 const auto cell_to_vtx_loc =
749 m_rank0_mesh->getCellToVtxMap().slice(m_all_own_and_ghost_cell_id[p]);
750 if (p > 0) {
751 mpi::send(cell_to_vtx_loc, p);
752 } else {
753 this->getCellToVtxMap() = cell_to_vtx_loc;
754 }
755 }
756 } else {
757 mpi::recv(this->getCellToVtxMap(), 0);
758 }
759 // find unique local vertices
760 const auto& vtx_loc = this->getCellToVtxMap().data();
761 SetType need_vtxs_set(vtx_loc.begin(), vtx_loc.end());
762 std::vector<index_t> need_vtxs(need_vtxs_set.begin(), need_vtxs_set.end());
763 // send to root the unique vertices and scatter the vtx to cell map
764 if (mpi::root()) {
765 std::vector<index_t> need_vtxs_tmp;
766 for (int p = 1; p < size; ++p) {
767 mpi::recv(need_vtxs_tmp, p);
768 mpi::send(m_rank0_mesh->getVtxToCellMap().slice(need_vtxs_tmp), p);
769 }
770 this->getVtxToCellMap() = m_rank0_mesh->getVtxToCellMap().slice(need_vtxs);
771 } else {
772 mpi::send(need_vtxs, 0);
773 mpi::recv(this->getVtxToCellMap(), 0);
774 }
775 // invert vtx mapping
776 boost::unordered_flat_map<index_t, index_t> vtx_glob_to_loc;
777 for (std::size_t i = 0; i < need_vtxs.size(); ++i) {
778 vtx_glob_to_loc[need_vtxs[i]] = i;
779 }
780 // compute weak dual
781 const auto n_loc_cell = this->getCellToVtxMap().size();
782 datastructure::CrsMapUncompressed<index_t, 32> weak_dual_uncompressed;
783 weak_dual_uncompressed.resize(n_loc_cell);
784 for (index_t loc_cell_id = 0; loc_cell_id < n_loc_cell; ++loc_cell_id) {
785 SetType weak_neigh;
786 for (auto glob_vtx_id : this->getCellToVtxMap()[loc_cell_id]) {
787 const auto loc_vtx_id = vtx_glob_to_loc[glob_vtx_id];
788 for (auto glob_neigh_cell : this->getVtxToCellMap()[loc_vtx_id]) {
789 weak_neigh.insert(glob_neigh_cell);
790 }
791 }
792 weak_dual_uncompressed[loc_cell_id].assign(
793 weak_neigh.begin(), weak_neigh.end());
794 }
795 m_weak_dual = weak_dual_uncompressed.template compress<index_t>();
796}

798template <typename Mesh>
799void ParallelMesh<Mesh>::buildLocalRenumbering() {
800 SetType vtx_idxs, facet_idxs, edge_idxs;
801
802 // compute the owning part
803 const auto n_owned_cells = m_own_enty_size[self_dim];
804 for (index_t cell_id = 0; cell_id < n_owned_cells; ++cell_id) {
805 for (const auto vtx_id : this->getCellToVtxMap()[cell_id]) {
806 vtx_idxs.insert(vtx_id);
807 }
808 for (const auto facet_id : this->getCellToFacetMap()[cell_id]) {
809 facet_idxs.insert(facet_id);
810 }
811 if constexpr (self_dim == 3) {
812 for (const auto edge_id : this->getCellToEdgeMap()[cell_id]) {
813 edge_idxs.insert(edge_id);
814 }
815 }
816 }
817 // build local to global as vector
818 m_local_numbering[0].assign(vtx_idxs.begin(), vtx_idxs.end());
819 m_local_numbering[self_dim - 1].assign(facet_idxs.begin(), facet_idxs.end());
820 if constexpr (self_dim == 3) {
821 m_local_numbering[self_dim - 2].assign(edge_idxs.begin(), edge_idxs.end());
822 }
823 for (std::size_t d = 0; d < self_dim; ++d) {
824 m_own_enty_size[d] = m_local_numbering[d].size();
825 }
826 // compute the non-owning part
827 const auto n_ghost_cells = m_ghost_enty_size[self_dim];
828 const auto n_total_cells = n_ghost_cells + n_owned_cells;
829 for (index_t cell_id = n_owned_cells; cell_id < n_total_cells; ++cell_id) {
830 for (const auto vtx_id : this->getCellToVtxMap()[cell_id]) {
831 const auto [it, inserted] = vtx_idxs.insert(vtx_id);
832 if (inserted) {
833 m_local_numbering[0].push_back(vtx_id);
834 }
835 }
836 for (const auto facet_id : this->getCellToFacetMap()[cell_id]) {
837 const auto [it, inserted] = facet_idxs.insert(facet_id);
838 if (inserted) {
839 m_local_numbering[self_dim - 1].push_back(facet_id);
840 }
841 }
842 if constexpr (self_dim == 3) {
843 for (const auto edge_id : this->getCellToEdgeMap()[cell_id]) {
844 const auto [it, inserted] = edge_idxs.insert(edge_id);
845 if (inserted) {
846 m_local_numbering[self_dim - 2].push_back(edge_id);
847 }
848 }
849 }
850 }
851 for (std::size_t d = 0; d < self_dim; ++d) {
852 m_ghost_enty_size[d] = m_local_numbering[d].size() - m_own_enty_size[d];
853 }
854
855 // build global to local by inversion
856 for (std::size_t d = 0; d < self_dim + 1; ++d) {
857 const auto& local_map = m_local_numbering[d];
858 for (std::size_t i = 0; i < local_map.size(); ++i) {
859 m_global_numbering[d][local_map[i]] = i;
860 }
861 }

864template <typename Mesh>
865void ParallelMesh<Mesh>::buildLocalEntityToCellMaps() {
866 const auto size = mpi::size();
867 // build local maps in root and distribute them
868 if (mpi::root()) {
869 {
870 std::array<std::vector<index_t>, self_dim> local_numbering;
871 for (int p = 1; p < size; ++p) {
872 // recv local to global numbering
873 for (std::size_t d = 0; d < self_dim; ++d) {
874 mpi::recv(local_numbering[d], p);
875 }
876
877 // build and send vtxs and local maps (F2C F2V V2C [E2C])
878 const auto vtx_loc =
879 vutils::slice(m_rank0_mesh->getVertexes(), local_numbering[0]);
880 mpi::send(vtx_loc, p);
881 const auto facet_to_cell_loc = vutils::slice(
882 m_rank0_mesh->getFacetToCellMap(), local_numbering[self_dim - 1]);
883 mpi::send(facet_to_cell_loc, p);
884 const auto facet_to_vtx_loc = m_rank0_mesh->getFacetToVtxMap().slice(
885 local_numbering[self_dim - 1]);
886 mpi::send(facet_to_vtx_loc, p);
887 mpi::send(m_rank0_mesh->getVtxToCellMap().slice(local_numbering[0]), p);
888 if constexpr (self_dim == 3) {
889 const auto edge_to_cell_loc = m_rank0_mesh->getEdgeToCellMap().slice(
890 local_numbering[self_dim - 2]);
891 mpi::send(edge_to_cell_loc, p);
892
893 const auto edge_to_vtx_loc = vutils::slice(
894 m_rank0_mesh->getEdgeToVtxMap(), local_numbering[self_dim - 2]);
895 mpi::send(edge_to_vtx_loc, p);
896 }
897 }
898 }
899 // build for self
900 this->getVertexes() =
901 vutils::slice(m_rank0_mesh->getVertexes(), m_local_numbering[0]);
902 this->getFacetToCellMap() = vutils::slice(
903 m_rank0_mesh->getFacetToCellMap(), m_local_numbering[self_dim - 1]);
904 this->getFacetToVtxMap() =
905 m_rank0_mesh->getFacetToVtxMap().slice(m_local_numbering[self_dim - 1]);
906 this->getVtxToCellMap() =
907 m_rank0_mesh->getVtxToCellMap().slice(m_local_numbering[0]);
908 if constexpr (self_dim == 3) {
909 this->getEdgeToCellMap() = m_rank0_mesh->getEdgeToCellMap().slice(
910 m_local_numbering[self_dim - 2]);
911 this->getEdgeToVtxMap() = vutils::slice(
912 m_rank0_mesh->getEdgeToVtxMap(), m_local_numbering[self_dim - 2]);
913 }
914 } else {
915 // send local numbering to root
916 for (std::size_t d = 0; d < self_dim; ++d) {
917 mpi::send(m_local_numbering[d], 0);
918 }
919 // recv vertexes and local maps (F2C F2V V2C [E2C])
920 mpi::recv(this->getVertexes(), 0);
921 mpi::recv(this->getFacetToCellMap(), 0);
922 mpi::recv(this->getFacetToVtxMap(), 0);
923 mpi::recv(this->getVtxToCellMap(), 0);
924 if constexpr (self_dim == 3) {
925 mpi::recv(this->getEdgeToCellMap(), 0);
926 mpi::recv(this->getEdgeToVtxMap(), 0);
927 }
928 }
929 removeNotOwnedFromLocalEntityToCellMaps();
930}

932template <typename Mesh>
933void ParallelMesh<Mesh>::removeNotOwnedFromLocalEntityToCellMaps() {
934 // in the maps we may have references to cells that do not belong to
935 // the proc that owns that map.
936 // `owned_table` has size of # of global cells, which scales badly
937 // (linear with the number of proc). However this should not be a
938 // large issue since it is a temporary object that is allocated
939 // in a staggered way by the processors, since they enter in this
940 // function just after rank 0 has finished computing the local map
941 std::vector<bool> owned_table(m_global_entity_size.back(), false);
942 for (const auto global_cell_id : m_local_numbering[self_dim]) {
943 owned_table[global_cell_id] = true;
944 }
945 // Dual
946 assert(
947 this->getDualMap().size() ==
948 m_ghost_enty_size[self_dim] + m_own_enty_size[self_dim]);
949 for (auto& glob_cell_id : this->getDualMap().data()) {
950 if (!owned_table[glob_cell_id]) {
951 glob_cell_id = -1;
952 }
953 }
954 this->getDualMap().remove(-1);
955 // F2C
956 const auto n_loc_faces = this->getFacetToCellMap().size();
957 for (std::size_t loc_face_id = 0; loc_face_id < n_loc_faces; ++loc_face_id) {
958 for (auto& glob_cell_id : this->getFacetToCellMap()[loc_face_id]) {
959 if (glob_cell_id == -1) {
960 continue;
961 }
962 if (!owned_table[glob_cell_id]) {
963 glob_cell_id = -1;
964 }
965 }
966 }
967 // V2C
968 for (auto& glob_cell_id : this->getVtxToCellMap().data()) {
969 if (!owned_table[glob_cell_id]) {
970 glob_cell_id = -1;
971 }
972 }
973 this->getVtxToCellMap().remove(-1);
974 // E2C
975 if constexpr (self_dim == 3) {
976 for (auto& glob_cell_id : this->getEdgeToCellMap().data()) {
977 if (!owned_table[glob_cell_id]) {
978 glob_cell_id = -1;
979 }
980 }
981 this->getEdgeToCellMap().remove(-1);
982 }
983}

1046template <typename Mesh>
1047void ParallelMesh<Mesh>::distributePeriodicMetadata() {
1048 // Check if the serial mesh (rank 0) is periodic.
1049 // Returns immediately for non-periodic meshes, preserving existing behaviour.
1050 // Use int (not bool) because getMpiType<bool> is not defined.
1051 int is_periodic = 0;
1052 if (mpi::root() && m_rank0_mesh) {
1053 const auto& md = m_rank0_mesh->getMetadata();
1054 is_periodic =
1055 ((md.doubles.count("XPeriod") > 0) && (md.doubles.count("YPeriod") > 0))
1056 ? 1
1057 : 0;
1058 }
1059 mpi::bcast(is_periodic, 0);
1060 if (!is_periodic) {
1061 return;
1062 }
1063
1064 // Collect all required data on rank 0, then broadcast.
1065 std::vector<double> x_period, y_period, ghost_x, ghost_y;
1066 std::vector<index_t> global_n_vtx_vec;
1067 std::vector<index_t> serial_cdata, serial_cindices;
1068
1069 if (mpi::root()) {
1070 const auto& md = m_rank0_mesh->getMetadata();
1071 x_period = md.doubles.at("XPeriod");
1072 y_period = md.doubles.at("YPeriod");
1073 ghost_x = md.doubles.at("GhostVertices_x");
1074 ghost_y = md.doubles.at("GhostVertices_y");
1075 global_n_vtx_vec = {
1076 static_cast<index_t>(md.ints.at("NumVertices").front())};
1077
1078 // closePeriodicTopology stores connectivity in cell tags;
1079 // some VTU round-trips may put it in md.ints - check both.
1080 const auto& ct = m_rank0_mesh->getMeshTags().tags[Mesh::CellEnty];
1081 if (ct.count("CellToVtxMap_with_ghosts_data")) {
1082 serial_cdata.assign(
1083 ct.at("CellToVtxMap_with_ghosts_data").begin(),
1084 ct.at("CellToVtxMap_with_ghosts_data").end());
1085 serial_cindices.assign(
1086 ct.at("CellToVtxMap_with_ghosts_indices").begin(),
1087 ct.at("CellToVtxMap_with_ghosts_indices").end());
1088 } else {
1089 serial_cdata = md.ints.at("CellToVtxMap_with_ghosts_data");
1090 serial_cindices = md.ints.at("CellToVtxMap_with_ghosts_indices");
1091 }
1092 }
1093
1094 mpi::bcast(x_period, 0);
1095 mpi::bcast(y_period, 0);
1096 mpi::bcast(ghost_x, 0);
1097 mpi::bcast(ghost_y, 0);
1098 mpi::bcast(global_n_vtx_vec, 0);
1099 mpi::bcast(serial_cdata, 0);
1100 mpi::bcast(serial_cindices, 0);
1101
1102 const index_t global_n_vtx = global_n_vtx_vec[0];
1103 // All vertices in the local mesh (owned + MPI-ghost) are REAL vertices
1104 // (global IDs 0..global_n_vtx-1). Periodic ghost vertices (virtual images
1105 // used for cell geometry) are appended after local_n_vtx.
1106 const index_t local_n_vtx = static_cast<index_t>(m_local_numbering[0].size());
1107
1108 // Rebuild CellToVtxMap_with_ghosts with local vertex IDs for this rank.
1109 const index_t n_local_cells =
1110 static_cast<index_t>(m_local_numbering[self_dim].size());
1111 std::vector<index_t> local_cdata;
1112 std::vector<index_t> local_cindices;
1113 local_cindices.reserve(n_local_cells + 1);
1114 local_cindices.push_back(0);
1115
1116 for (index_t loc_cid = 0; loc_cid < n_local_cells; ++loc_cid) {
1117 const index_t glob_cid = m_local_numbering[self_dim][loc_cid];
1118 const index_t start = serial_cindices[glob_cid];
1119 const index_t end = serial_cindices[glob_cid + 1];
1120 for (index_t k = start; k < end; ++k) {
1121 const index_t vtx = serial_cdata[k];
1122 if (vtx < global_n_vtx) {
1123 // Real vertex: translate global to local ID.
1124 local_cdata.push_back(m_global_numbering[0].at(vtx));
1125 } else {
1126 // Periodic ghost vertex (virtual copy, not an actual mesh vertex):
1127 // offset past all local real vertices so getCellVertexesPeriodic
1128 // looks it up in GhostVertices_x/y.
1129 local_cdata.push_back(local_n_vtx + (vtx - global_n_vtx));
1130 }
1131 }
1132 local_cindices.push_back(static_cast<index_t>(local_cdata.size()));
1133 }
1134
1135 // Store in the local mesh metadata. getCellVertexesPeriodic prefers
1136 // md.ints over cell tags for connectivity, so we write there.
1137 auto& md = this->getMetadata();
1138 md.doubles["XPeriod"] = std::move(x_period);
1139 md.doubles["YPeriod"] = std::move(y_period);
1140 md.doubles["GhostVertices_x"] = std::move(ghost_x);
1141 md.doubles["GhostVertices_y"] = std::move(ghost_y);
1142 md.ints["NumVertices"] = {local_n_vtx};
1143 md.ints["CellToVtxMap_with_ghosts_data"] = std::move(local_cdata);
1144 md.ints["CellToVtxMap_with_ghosts_indices"] = std::move(local_cindices);
1145}
