100 template <typename DofHandler>
101 void init(
102 const DofHandler& dof_handler,
103 Type type = Plain,
104 PetscInt block_size = 1,
105 PetscInt aij_block_size = 1) {
106 if ((block_size != 1) && (type != Block)) {
107 std::cout
108 << "ERROR: block size != 1 available only with block type matrices"
109 << std::endl;
110 std::exit(-1);
111 }
112 createMatrix(type);
113
114 const PetscInt n_glob = dof_handler.ndofsGlobal();
115 PetscInt dof_begin = 0;
116 PetscInt dof_end = n_glob;
117 if constexpr (DofHandler::parallel) {
118 dof_begin = dof_handler.dofOffset()[mpi::rank()];
119 dof_end = dof_handler.dofOffset()[mpi::rank() + 1];
120 }
121 const PetscInt n_loc = dof_end - dof_begin;
122 // TODO: check not too imbalanced
123 // Print ownership details only in Debug builds.
124#ifndef NDEBUG
125 std::cout << "[SparseMatrix::init] own: " << n_loc << "/" << n_glob
126 << "@rank:" << mpi::rank() << std::endl;
127#endif
128 if (type == Block) {
129 PetscCallAbortWorld(MatSetBlockSize(m_petsc_matrix, block_size));
130 setMatrixSizes(PETSC_DECIDE, n_glob);
131 }
132 if (aij_block_size != 1) {
133 // n_loc differs between ranks: reduce the check so that all the ranks
134 // abort together instead of hanging in MatSetSizes
135 int bad =
136 aij_block_size < 1 || type == Block || n_loc % aij_block_size != 0;
137 MPI_Allreduce(MPI_IN_PLACE, &bad, 1, MPI_INT, MPI_LOR, PETSC_COMM_WORLD);
138 if (bad) {
139 pcout << "ERROR: aij_block_size must be positive, divide the local "
140 << "number of rows on every rank and is not available with "
141 << "Block type matrices" << std::endl;
142 MPI_Abort(PETSC_COMM_WORLD, -1);
143 }
144 // must be set before the sizes: the block size of a layout cannot be
145 // changed once it is set up
146 PetscCallAbortWorld(
147 MatSetBlockSizes(m_petsc_matrix, aij_block_size, aij_block_size));
148 }
149 setMatrixSizes(n_loc * block_size, n_glob * block_size);
150
151 if (DofHandler::parallel && (!dof_handler.hasRenumbering())) {
152 // if we do not have MPI renumbering, we do not know the sparsity pattern
153 return;
154 }
155 std::vector<PetscInt> dnn(n_loc, 0), onn(n_loc, 0);
156 computePreallocationData(dof_handler, dof_begin, dof_end, dnn, onn);
157 allocateMatrix(type, dnn, onn, block_size);
158 }

382 template <typename DofHandler>
383 void computePreallocationData(
384 const DofHandler& dof_handler,
385 const PetscInt dof_begin,
386 const PetscInt dof_end,
387 std::vector<PetscInt>& dnn,
388 std::vector<PetscInt>& onn) {
389 for (auto dof_id = dof_begin; dof_id < dof_end; ++dof_id) {
390 // assert that the key is in the map
391 auto loc_dof_id = dof_id;
392 if (DofHandler::parallel) {
393 loc_dof_id = dof_handler.getCurrToRefNumbering().at(dof_id);
394 }
395 const auto sp_row = dof_handler.getSparsityPattern()[loc_dof_id];
396 for (const auto glob_dof_nz : sp_row) {
397 if (dof_begin <= glob_dof_nz && glob_dof_nz < dof_end) {
398 dnn[dof_id - dof_begin]++;
399 } else {
400 onn[dof_id - dof_begin]++;
401 }
402 }
403 }
404 }
