162 template <typename FloatType, typename Polynomial>
163 void evalOnReferenceFrame(
164 const std::vector<Polynomial>& basis,
165 const qrule::QuadratureRule<FloatType, Dim>& qrule) {
166 const int64_t basis_size = basis.size();
167 const int64_t nodes_size = qrule.nodes.size();
168 resizeTables(basis_size, nodes_size);
169
170 if constexpr (internal::IsUnscaledSparsePolynomial<Polynomial>::value) {
171 evalSparseBasis(basis, qrule);
172 } else {
173 for (int64_t i = 0; i < basis_size; ++i) {
174 const auto& basis_func = basis[i];
175 const auto grad = basis_func.gradient();
176 for (int64_t j = 0; j < nodes_size; ++j) {
177 m_scalar_shapes_ref(i, j) = vmath::to<real_t>(
178 basis_func.evaluate(qrule.nodes[j].point.data()));
179 for (std::size_t d = 0; d < Dim; ++d) {
180 m_grads_shapes_ref(i, j)(d) = vmath::to<real_t>(
181 grad[d].evaluate(qrule.nodes[j].point.data()));
182 }
183 }
184 }
185 }
186
187 for (int64_t j = 0; j < nodes_size; ++j) {
188 m_quadrature_weights_ref(j) = vmath::to<real_t>(qrule.nodes[j].weight);
189 for (std::size_t d = 0; d < Dim; ++d) {
190 m_physical_qpts_ref(d, j) = vmath::to<real_t>(qrule.nodes[j].point[d]);
191 }
192 }
193 m_tabulated_basis = &basis;
194 m_stamp = nextStamp();
195 }

232 template <typename MappingType>
233 void evalAndApplyMapping(const MappingType& mapping, UpdateFlagsType flags) {
234 if (flags & UpdateWeights) {
235 for (int64_t i = 0; i < m_quadrature_weights.size(); ++i) {
236 m_quadrature_weights[i] =
237 m_quadrature_weights_ref[i] * mapping.jacobianDet({});
238 }
239 }
240
241 if (flags & UpdateGradients) {
242 for (int64_t i = 0; i < m_grads_shapes.rows(); ++i) {
243 for (int64_t j = 0; j < m_grads_shapes.cols(); ++j) {
244 m_grads_shapes(i, j) = mapping.jacobianInv({}).array() *
245 m_grads_shapes_ref(i, j).array();
246 }
247 }
248 }
249
250 if (flags & UpdateValues) {
251 m_scalar_shapes = m_scalar_shapes_ref;
252 }
253
254 if (flags & UpdateQuadraturePts) {
255 mapping.transform(m_physical_qpts_ref, m_physical_qpts);
256 }
257 m_stamp = nextStamp();
258 }

287 // Values and gradients of an unscaled sparse basis with the powers of each
288 // coordinate tabulated once per node; the operations and their order are
289 // those of SparsePolynomial::evaluate on the basis and on its derivative().
290 template <typename FloatType, typename Polynomial>
291 void evalSparseBasis(
292 const std::vector<Polynomial>& basis,
293 const qrule::QuadratureRule<FloatType, Dim>& qrule) {
294 static_assert(
295 Polynomial::dim == Dim, "basis and FeValues dimensions differ");
296 static_assert(
297 std::is_same_v<FloatType, double>, "quadrature nodes must be double");
298 using ExpType = std::decay_t<decltype(basis[0].data()[0].exp()[0])>;
299 std::size_t n_pow = 1;
300 for (const auto& poly : basis) {
301 for (const auto& mono : poly.data()) {
302 for (const auto e : mono.exp()) {
303 n_pow = std::max<std::size_t>(n_pow, static_cast<std::size_t>(e) + 1);
304 }
305 }
306 }
307 m_pow_table.resize(Dim * n_pow);
308 const int64_t nodes_size = qrule.nodes.size();
309 for (int64_t j = 0; j < nodes_size; ++j) {
310 const double* pt = qrule.nodes[j].point.data();
311 for (std::size_t d = 0; d < Dim; ++d) {
312 for (std::size_t k = 0; k < n_pow; ++k) {
313 m_pow_table[d * n_pow + k] =
314 vmath::pow(pt[d], static_cast<ExpType>(k));
315 }
316 }
317 for (std::size_t i = 0; i < basis.size(); ++i) {
318 double value = 0.0;
319 Eigen::Vector<double, Dim> grad = Eigen::Vector<double, Dim>::Zero();
320 for (const auto& mono : basis[i].data()) {
321 const auto& e = mono.exp();
322 double term = mono.coef();
323 for (std::size_t d = 0; d < Dim; ++d) {
324 term *= m_pow_table[d * n_pow + e[d]];
325 }
326 value += term;
327 for (std::size_t c = 0; c < Dim; ++c) {
328 if (e[c] == 0) {
329 continue;
330 }
331 double dterm = mono.coef() * e[c];
332 if (dterm == 0.0) {
333 continue;
334 }
335 for (std::size_t d = 0; d < Dim; ++d) {
336 dterm *= m_pow_table[d * n_pow + e[d] - (d == c ? 1 : 0)];
337 }
338 grad[c] += dterm;
339 }
340 }
341 m_scalar_shapes_ref(i, j) = value;
342 m_grads_shapes_ref(i, j) = grad;
343 }
344 }
345 }
