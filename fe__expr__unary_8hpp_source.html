34template <std::size_t Dim, typename E>
35class FeExpressionUnaryMinus
36 : public FeExpressionBase<Dim, FeExpressionUnaryMinus<Dim, E>> {
37 public:
38 static constexpr bool is_leaf = false;
39 static constexpr uint8_t is_bilinear = E::is_bilinear;
40
41 explicit constexpr FeExpressionUnaryMinus(const E& expr) : m_expr(expr) {};
42
43 template <std::size_t J = 0, std::size_t I = 0>
44 decltype(auto) evaluate(
45 const FeValues<Dim>& fe_values,
46 std::size_t j,
47 std::size_t i,
48 std::size_t q) const {
49 using TypeEval =
50 decltype(m_expr.template evaluate<J, I>(fe_values, j, i, q));
51 if constexpr (std::is_same_v<TypeEval, void>) {
52 return void();
53 } else {
54 return -m_expr.template evaluate<J, I>(fe_values, j, i, q);
55 }
56 }
57
58 constexpr UpdateFlagsType computeUpdateFlags() const {
59 return m_expr.computeUpdateFlags();
60 }
61 constexpr std::size_t computeOrder() const { return m_expr.computeOrder(); }
62
63 decltype(auto) evaluatePoly(
64 const FeValues<Dim>& fe_values,
65 std::size_t j,
66 std::size_t i,
67 std::size_t bi,
68 std::size_t bj) const {
69 return m_expr.evaluatePoly(fe_values, j, i, bi, bj);
70 }
71 decltype(auto) evaluatePolyCoef(
72 const auto& mapping,
73 const auto& fe_values,
74 std::size_t bi,
75 std::size_t bj) const {
76 using MatType = std::decay_t<decltype(m_expr.evaluatePolyCoef(
77 mapping, fe_values, bi, bj))>;
78 return MatType{-m_expr.evaluatePolyCoef(mapping, fe_values, bi, bj)};
79 }
80 constexpr bool precompute(
81 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
82 return m_expr.precompute(fe_values, cells);
83 }
84
85 protected:
86 typename std::conditional_t<E::is_leaf, const E&, const E> m_expr;
87};

127template <std::size_t Dim, typename E>
128class FeExpressionFieldComponent
129 : public FeExpressionBase<Dim, FeExpressionFieldComponent<Dim, E>> {
130 private:
131 template <typename T, typename = void>
132 struct FsTypeOf {
133 using type = void;
134 };
135 template <typename T>
136 struct FsTypeOf<T, std::void_t<typename T::FeSpaceType>> {
137 using type = typename T::FeSpaceType;
138 };
139
140 public:
141 using OutputType = double;
142 using FeSpaceType = typename FsTypeOf<E>::type;
143 static constexpr bool is_leaf = false;
144 static constexpr uint8_t is_bilinear = E::is_bilinear;
145 // Forward the block range of E unchanged: selecting a field block does not
146 // change which test/trial fields the expression spans, so nblocks (the
147 // number of rhs integrator iterations) stays equal to E's.
148 static constexpr std::size_t k_index_begin = E::k_index_begin;
149 static constexpr std::size_t k_index_end = E::k_index_end;
150
151 explicit constexpr FeExpressionFieldComponent(const E& expr)
152 : m_expr(expr) {};
153
154 // Forward feSpace() when E is a test/trial expression (has feSpace()).
155 // SFINAE excludes this for IsNothing expressions (no feSpace() on E).
156 template <
157 typename U = E,
158 typename = std::void_t<decltype(std::declval<const U&>().feSpace())>>
159 constexpr decltype(auto) feSpace() const {
160 return m_expr.feSpace();
161 }
162
163 template <std::size_t J = 0, std::size_t I = 0>
164 decltype(auto) evaluate(
165 const FeValues<Dim>& fe_values,
166 std::size_t j,
167 std::size_t i,
168 std::size_t q) const {
169 // Two-index Eigen access (row, col): the test/trial value is an
170 // EigenVulpes::OneComponentVector with a compile-time active index and no
171 // LinearAccessBit, so single-index operator[] is ill-formed. (row, 0) also
172 // works for plain Eigen column vectors, so this stays general over E.
173 if constexpr (is_bilinear == IsBilinear) {
174 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(J, I);
175 } else if constexpr (is_bilinear == IsLinearTest) {
176 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(I, 0);
177 } else if constexpr (is_bilinear == IsLinearTrial) {
178 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(J, 0);
179 } else {
180 static_assert(
181 Dim * 0 &&
182 "fe::field_component requires a test/trial expression (IsLinearTest, "
183 "IsLinearTrial, or IsBilinear). For spatial component extraction "
184 "from "
185 "a coefficient use fe::spatial_component<K>.");
186 }
187 }
188
189 constexpr UpdateFlagsType computeUpdateFlags() const {
190 return m_expr.computeUpdateFlags();
191 }
192 constexpr std::size_t computeOrder() const { return m_expr.computeOrder(); }
193
194 constexpr bool precompute(
195 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
196 return m_expr.precompute(fe_values, cells);
197 }
198
199 protected:
200 typename std::conditional_t<E::is_leaf, const E&, const E> m_expr;
201};

397template <std::size_t Dim, typename E>
398class FeExpressionAbs : public FeExpressionBase<Dim, FeExpressionAbs<Dim, E>> {
399 public:
400 static constexpr bool is_leaf = false;
401 static constexpr uint8_t is_bilinear = IsNothing;
402
403 explicit FeExpressionAbs(E const& e) : m_e(e) {}
404
405 template <std::size_t J = 0, std::size_t I = 0>
406 decltype(auto) evaluate(
407 const FeValues<Dim>& fe_values,
408 std::size_t j,
409 std::size_t i,
410 std::size_t q) const {
411 using RetType = decltype(m_e.template evaluate<J, I>(fe_values, j, i, q));
412 if constexpr (std::is_same_v<RetType, void>) {
413 return void();
414 } else {
415 return std::abs(m_e.template evaluate<J, I>(fe_values, j, i, q));
416 }
417 }
418
419 decltype(auto) evaluatePoly(
420 const FeValues<Dim>& fe_values,
421 std::size_t j,
422 std::size_t i,
423 std::size_t bi,
424 std::size_t bj) const {
425 return m_e.evaluatePoly(fe_values, j, i, bi, bj);
426 }
427
428 decltype(auto) evaluatePolyCoef(
429 const auto& mapping,
430 const auto& fe_values,
431 std::size_t bi,
432 std::size_t bj) const {
433 // |p| is not a polynomial: abs is applied only at the quadrature points
434 static const bool warned = [] {
435 pcout << "Warning: abs is ignored by the qfree integrators, it is "
436 "applied only at the quadrature points"
437 << std::endl;
438 return true;
439 }();
440 (void)warned;
441 return m_e.evaluatePolyCoef(mapping, fe_values, bi, bj);
442 }
443
444 constexpr std::size_t computeOrder() const { return m_e.computeOrder(); }
445
446 constexpr UpdateFlagsType computeUpdateFlags() const {
447 return m_e.computeUpdateFlags();
448 }
449
450 constexpr bool precompute(
451 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
452 return m_e.precompute(fe_values, cells);
453 }
454
455 private:
456 std::conditional_t<E::is_leaf, const E&, const E> m_e;
457};
