38template <std::size_t Dim, typename E>
39class FeExpressionUnaryMinus
40 : public FeExpressionBase<Dim, FeExpressionUnaryMinus<Dim, E>> {
41 public:
42 static constexpr bool is_leaf = false;
43 static constexpr uint8_t is_bilinear = E::is_bilinear;
44
45 explicit constexpr FeExpressionUnaryMinus(const E& expr) : m_expr(expr) {};
46
47 template <std::size_t J = 0, std::size_t I = 0>
48 decltype(auto) evaluate(
49 const FeValues<Dim>& fe_values,
50 std::size_t j,
51 std::size_t i,
52 std::size_t q) const {
53 using TypeEval =
54 decltype(m_expr.template evaluate<J, I>(fe_values, j, i, q));
55 if constexpr (std::is_same_v<TypeEval, void>) {
56 return void();
57 } else {
58 return -m_expr.template evaluate<J, I>(fe_values, j, i, q);
59 }
60 }
61
62 constexpr UpdateFlagsType computeUpdateFlags() const {
63 return m_expr.computeUpdateFlags();
64 }
65 constexpr std::size_t computeOrder() const { return m_expr.computeOrder(); }
66
67 decltype(auto) evaluatePoly(
68 const FeValues<Dim>& fe_values,
69 std::size_t j,
70 std::size_t i,
71 std::size_t bi,
72 std::size_t bj) const {
73 return m_expr.evaluatePoly(fe_values, j, i, bi, bj);
74 }
75 decltype(auto) evaluatePolyCoef(
76 const auto& mapping,
77 const auto& fe_values,
78 std::size_t bi,
79 std::size_t bj) const {
80 using MatType = std::decay_t<decltype(m_expr.evaluatePolyCoef(
81 mapping, fe_values, bi, bj))>;
82 return MatType{-m_expr.evaluatePolyCoef(mapping, fe_values, bi, bj)};
83 }
84 constexpr bool precompute(
85 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
86 return m_expr.precompute(fe_values, cells);
87 }
88
89 protected:
90 typename std::conditional_t<E::is_leaf, const E&, const E> m_expr;
91};

131template <std::size_t Dim, typename E>
132class FeExpressionFieldComponent
133 : public FeExpressionBase<Dim, FeExpressionFieldComponent<Dim, E>> {
134 private:
135 template <typename T, typename = void>
136 struct FsTypeOf {
137 using type = void;
138 };
139 template <typename T>
140 struct FsTypeOf<T, std::void_t<typename T::FeSpaceType>> {
141 using type = typename T::FeSpaceType;
142 };
143
144 public:
145 using OutputType = double;
146 using FeSpaceType = typename FsTypeOf<E>::type;
147 static constexpr bool is_leaf = false;
148 static constexpr uint8_t is_bilinear = E::is_bilinear;
149 // Forward the block range of E unchanged: selecting a field block does not
150 // change which test/trial fields the expression spans, so nblocks (the
151 // number of rhs integrator iterations) stays equal to E's.
152 static constexpr std::size_t k_index_begin = E::k_index_begin;
153 static constexpr std::size_t k_index_end = E::k_index_end;
154
155 explicit constexpr FeExpressionFieldComponent(const E& expr)
156 : m_expr(expr) {};
157
158 // Forward feSpace() when E is a test/trial expression (has feSpace()).
159 // SFINAE excludes this for IsNothing expressions (no feSpace() on E).
160 template <
161 typename U = E,
162 typename = std::void_t<decltype(std::declval<const U&>().feSpace())>>
163 constexpr decltype(auto) feSpace() const {
164 return m_expr.feSpace();
165 }
166
167 template <std::size_t J = 0, std::size_t I = 0>
168 decltype(auto) evaluate(
169 const FeValues<Dim>& fe_values,
170 std::size_t j,
171 std::size_t i,
172 std::size_t q) const {
173 // Two-index Eigen access (row, col): the test/trial value is an
174 // EigenVulpes::OneComponentVector with a compile-time active index and no
175 // LinearAccessBit, so single-index operator[] is ill-formed. (row, 0) also
176 // works for plain Eigen column vectors, so this stays general over E.
177 if constexpr (is_bilinear == IsBilinear) {
178 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(J, I);
179 } else if constexpr (is_bilinear == IsLinearTest) {
180 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(I, 0);
181 } else if constexpr (is_bilinear == IsLinearTrial) {
182 return m_expr.template evaluate<J, I>(fe_values, j, i, q)(J, 0);
183 } else {
184 static_assert(
185 Dim * 0 &&
186 "fe::field_component requires a test/trial expression (IsLinearTest, "
187 "IsLinearTrial, or IsBilinear). For spatial component extraction "
188 "from "
189 "a coefficient use fe::spatial_component<K>.");
190 }
191 }
192
193 constexpr UpdateFlagsType computeUpdateFlags() const {
194 return m_expr.computeUpdateFlags();
195 }
196 constexpr std::size_t computeOrder() const { return m_expr.computeOrder(); }
197
198 constexpr bool precompute(
199 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
200 return m_expr.precompute(fe_values, cells);
201 }
202
203 protected:
204 typename std::conditional_t<E::is_leaf, const E&, const E> m_expr;
205};

401template <std::size_t Dim, typename E>
402class FeExpressionAbs : public FeExpressionBase<Dim, FeExpressionAbs<Dim, E>> {
403 public:
404 static constexpr bool is_leaf = false;
405 static constexpr uint8_t is_bilinear = IsNothing;
406
407 explicit FeExpressionAbs(E const& e) : m_e(e) {}
408
409 template <std::size_t J = 0, std::size_t I = 0>
410 decltype(auto) evaluate(
411 const FeValues<Dim>& fe_values,
412 std::size_t j,
413 std::size_t i,
414 std::size_t q) const {
415 using RetType = decltype(m_e.template evaluate<J, I>(fe_values, j, i, q));
416 if constexpr (std::is_same_v<RetType, void>) {
417 return void();
418 } else {
419 return std::abs(m_e.template evaluate<J, I>(fe_values, j, i, q));
420 }
421 }

424 const FeValues<Dim>& fe_values,

429 return m_e.evaluatePoly(fe_values, j, i, bi, bj);

432 decltype(auto) evaluatePolyCoef(
433 const auto& mapping,
434 const auto& fe_values,
435 std::size_t bi,
436 std::size_t bj) const {
437 // |p| is not a polynomial: abs is applied only at the quadrature points
438 static const bool warned = [] {
439 pcout << "Warning: abs is ignored by the qfree integrators, it is "
440 "applied only at the quadrature points"
441 << std::endl;
442 return true;
443 }();
444 (void)warned;
445 return m_e.evaluatePolyCoef(mapping, fe_values, bi, bj);
446 }

448 constexpr std::size_t computeOrder() const { return m_e.computeOrder(); }

452 }

457 }
458
459 private:
460 std::conditional_t<E::is_leaf, const E&, const E> m_e;

481template <std::size_t Dim, typename E>
482class FeExpressionQptsCache
483 : public FeExpressionBase<Dim, FeExpressionQptsCache<Dim, E>> {
484 static_assert(
485 E::is_bilinear == IsNothing,
486 "fe::cached requires a coefficient expression (no test or trial "
487 "function)");
488
489 public:
490 static constexpr bool is_leaf = false;
491 static constexpr uint8_t is_bilinear = IsNothing;
492 using EvalType = std::decay_t<decltype(std::declval<const E&>().evaluate(
493 std::declval<const FeValues<Dim>&>(), 0, 0, 0))>;
494 // plain storage type of the value (evaluated Eigen expression, or scalar)
495 using OutputType = typename decltype([] {
496 if constexpr (std::is_arithmetic_v<EvalType>) {
497 return std::type_identity<EvalType>{};
498 } else {
499 return std::type_identity<
500 std::decay_t<decltype(std::declval<EvalType>().eval())>>{};
501 }
502 }())::type;
503
504 explicit FeExpressionQptsCache(const E& expr) : m_expr(expr) {}
505
506 template <std::size_t J = 0, std::size_t I = 0>
507 const OutputType& evaluate(
508 const FeValues<Dim>&, std::size_t, std::size_t, std::size_t q) const {
509 return m_values[q];
510 }
511
512 constexpr UpdateFlagsType computeUpdateFlags() const {
513 return m_expr.computeUpdateFlags();
514 }
515 constexpr std::size_t computeOrder() const { return m_expr.computeOrder(); }
516
517 decltype(auto) evaluatePoly(
518 const FeValues<Dim>& fe_values,
519 std::size_t j,
520 std::size_t i,
521 std::size_t bi,
522 std::size_t bj) const {
523 return m_expr.evaluatePoly(fe_values, j, i, bi, bj);
524 }
525 decltype(auto) evaluatePolyCoef(
526 const auto& mapping,
527 const auto& fe_values,
528 std::size_t bi,
529 std::size_t bj) const {
530 return m_expr.evaluatePolyCoef(mapping, fe_values, bi, bj);
531 }
532 bool precompute(
533 const FeValues<Dim>& fe_values, const CellsType<Dim>& cells) const {
534 if (!m_expr.precompute(fe_values, cells)) {
535 return false;
536 }
537 const std::size_t nq = fe_values.weights().size();
538 m_values.resize(nq);
539 for (std::size_t q = 0; q < nq; ++q) {
540 m_values[q] = m_expr.evaluate(fe_values, 0, 0, q);
541 }
542 return true;
543 }
544
545 protected:
546 typename std::conditional_t<E::is_leaf, const E&, const E> m_expr;
547 mutable std::vector<OutputType> m_values;
548};
