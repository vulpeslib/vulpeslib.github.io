45template <typename T>
46MPI_Datatype getMpiType() {
47 if constexpr (std::is_same_v<T, unsigned char>) {
48 return MPI_UNSIGNED_CHAR;
49 } else if constexpr (std::is_same_v<T, char>) {
50 return MPI_CHAR;
51 } else if constexpr (std::is_same_v<T, short>) { // NOLINT(runtime/int)
52 return MPI_SHORT;
53 } else if constexpr (std::is_same_v<T, int>) {
54 return MPI_INT;
55 } else if constexpr (std::is_same_v<T, long>) { // NOLINT(runtime/int)
56 return MPI_LONG;
57 } else if constexpr (std::is_same_v<T, long long>) { // NOLINT(runtime/int)
58 return MPI_LONG_LONG;
59 } else if constexpr (std::is_same_v<
60 T, unsigned short>) { // NOLINT(runtime/int)
61 return MPI_UNSIGNED_SHORT;
62 } else if constexpr (std::is_same_v<T, unsigned int>) {
63 return MPI_UNSIGNED;
64 } else if constexpr (std::is_same_v<
65 T, unsigned long>) { // NOLINT(runtime/int)
66 return MPI_UNSIGNED_LONG;
67 } else if constexpr (std::is_same_v<
68 T, unsigned long long>) { // NOLINT(runtime/int)
69 return MPI_UNSIGNED_LONG_LONG;
70 } else if constexpr (std::is_same_v<T, float>) {
71 return MPI_FLOAT;
72 } else if constexpr (std::is_same_v<T, double>) {
73 return MPI_DOUBLE;
74 } else if constexpr (std::is_same_v<T, long double>) {
75 return MPI_LONG_DOUBLE;
76 } else {
77 static_assert(sizeof(T) * 0 && "Unknown conversion to MPI type.");
78 }
79}

348template <typename T, int N, int M, int Options>
349void bcast(
350 Eigen::Matrix<T, N, M, Options>& data,
351 int root,
352 MPI_Comm communicator = MPI_COMM_WORLD) {
353 std::size_t nrows = 0, ncols = 0;
354 if (mpi::rank() == root) {
355 nrows = data.rows();
356 ncols = data.cols();
357 }
358 Profiler::startInterval("MPI-bcast");
359 MPI_Bcast(&nrows, 1, getMpiType<std::size_t>(), root, communicator);
360 MPI_Bcast(&ncols, 1, getMpiType<std::size_t>(), root, communicator);
361 if (!(mpi::rank() == root)) {
362 data.resize(nrows, ncols);
363 }
364 MPI_Bcast(data.data(), nrows * ncols, getMpiType<T>(), root, communicator);
365 Profiler::stopInterval("MPI-bcast");
366}

404// assume everyone knows count
405inline auto computeChuncks(int32_t count, int32_t stride = 1) {
406 const int32_t comm_size = mpi::size();
407 std::vector<int32_t> displacements(comm_size), counts(comm_size);
408
409 const int32_t chunck_sz = count / comm_size;
410
411 for (int32_t i = 0; i < comm_size; i++) {
412 counts[i] = stride * (chunck_sz + (i < (count % comm_size)));
413 }
414
415 displacements[0] = 0;
416 for (int32_t i = 1; i < comm_size; i++) {
417 displacements[i] = displacements[i - 1] + counts[i - 1];
418 }
419
420 return std::pair{displacements, counts};
421}

533class MpiHandle {
534 public:
536 static int init(
537 int& argc,
538 char**& argv,
539 int required = s_default_thread_required,
540 int* provided = nullptr) {
541 return init(&argc, &argv, required, provided);
542 }
544 static int init(
545 int* argc = nullptr,
546 char*** argv = nullptr,
547 int required = s_default_thread_required,
548 int* provided = nullptr) {
549 assert(!isInitialized() && "MPI already initialized!");
550 int mpi_res;
551 if (required == MPI_THREAD_SINGLE) {
552 mpi_res = MPI_Init(argc, argv);
553 } else {
554 mpi_res = MPI_Init_thread(argc, argv, required, provided);
555 }
556 if (mpi_res != MPI_SUCCESS) {
557 std::cout << "Error during MPI_Init()!" << std::endl;
558 std::exit(-1);
559 }
560 // configure threads before Petsc init
561 vutils::configureThreading();
562 PetscCall(PetscInitialize(argc, argv, nullptr, nullptr));
563 singleton();
564 return 0;
565 }
566
567 private:
569 inline constexpr static int s_default_thread_required = MPI_THREAD_SINGLE;
571 static MpiHandle& singleton() {
572 static MpiHandle mpi;
573 return mpi;
574 }
576 static int finalize() {
577 if (isPetscInitialized() && !isPetscFinalized()) {
578 PetscCall(PetscFinalize());
579 }
580 if (isInitialized() && !isFinalized()) {
581 MPI_Finalize();
582 }
583 return 0;
584 }
586 ~MpiHandle() { finalize(); }
588 MpiHandle() = default;
589};
