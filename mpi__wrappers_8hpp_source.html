55template <typename T>
56MPI_Datatype getMpiType() {
57 if constexpr (std::is_same_v<T, unsigned char>) {
58 return MPI_UNSIGNED_CHAR;
59 } else if constexpr (std::is_same_v<T, char>) {
60 return MPI_CHAR;
61 } else if constexpr (std::is_same_v<T, short>) { // NOLINT(runtime/int)
62 return MPI_SHORT;
63 } else if constexpr (std::is_same_v<T, int>) {
64 return MPI_INT;
65 } else if constexpr (std::is_same_v<T, long>) { // NOLINT(runtime/int)
66 return MPI_LONG;
67 } else if constexpr (std::is_same_v<T, long long>) { // NOLINT(runtime/int)
68 return MPI_LONG_LONG;
69 } else if constexpr (std::is_same_v<
70 T, unsigned short>) { // NOLINT(runtime/int)
71 return MPI_UNSIGNED_SHORT;
72 } else if constexpr (std::is_same_v<T, unsigned int>) {
73 return MPI_UNSIGNED;
74 } else if constexpr (std::is_same_v<
75 T, unsigned long>) { // NOLINT(runtime/int)
76 return MPI_UNSIGNED_LONG;
77 } else if constexpr (std::is_same_v<
78 T, unsigned long long>) { // NOLINT(runtime/int)
79 return MPI_UNSIGNED_LONG_LONG;
80 } else if constexpr (std::is_same_v<T, float>) {
81 return MPI_FLOAT;
82 } else if constexpr (std::is_same_v<T, double>) {
83 return MPI_DOUBLE;
84 } else if constexpr (std::is_same_v<T, long double>) {
85 return MPI_LONG_DOUBLE;
86 } else {
87 static_assert(sizeof(T) * 0 && "Unknown conversion to MPI type.");
88 }
89}

569template <typename T, int N, int M, int Options>
570void bcast(
571 Eigen::Matrix<T, N, M, Options>& data,
572 int root,
573 MPI_Comm communicator = MPI_COMM_WORLD) {
574 std::size_t nrows = 0, ncols = 0;
575 if (mpi::rank() == root) {
576 nrows = data.rows();
577 ncols = data.cols();
578 }
579 Profiler::startInterval("MPI-bcast");
580 MPI_Bcast(&nrows, 1, getMpiType<std::size_t>(), root, communicator);
581 MPI_Bcast(&ncols, 1, getMpiType<std::size_t>(), root, communicator);
582 if (!(mpi::rank() == root)) {
583 data.resize(nrows, ncols);
584 }
585 MPI_Bcast(data.data(), nrows * ncols, getMpiType<T>(), root, communicator);
586 Profiler::stopInterval("MPI-bcast");
587}

656// assume everyone knows count
657inline auto computeChuncks(int32_t count, int32_t stride = 1) {
658 const int32_t comm_size = mpi::size();
659 std::vector<int32_t> displacements(comm_size), counts(comm_size);
660
661 const int32_t chunck_sz = count / comm_size;
662
663 for (int32_t i = 0; i < comm_size; i++) {
664 counts[i] = stride * (chunck_sz + (i < (count % comm_size)));
665 }
666
667 displacements[0] = 0;
668 for (int32_t i = 1; i < comm_size; i++) {
669 displacements[i] = displacements[i - 1] + counts[i - 1];
670 }
671
672 return std::pair{displacements, counts};
673}

860class MpiHandle {
861 public:
863 static int init(
864 int& argc,
865 char**& argv,
866 int required = s_default_thread_required,
867 int* provided = nullptr) {
868 return init(&argc, &argv, required, provided);
869 }
871 static int init(
872 int* argc = nullptr,
873 char*** argv = nullptr,
874 int required = s_default_thread_required,
875 int* provided = nullptr) {
876 assert(!isInitialized() && "MPI already initialized!");
877 int mpi_res;
878 if (required == MPI_THREAD_SINGLE) {
879 mpi_res = MPI_Init(argc, argv);
880 } else {
881 mpi_res = MPI_Init_thread(argc, argv, required, provided);
882 }
883 if (mpi_res != MPI_SUCCESS) {
884 throw std::runtime_error("MPI_Init failed");
885 }
886 // Configure VTK before any VTK parallel operation.
887 vutils::configureVtkThreading();
888 // Initialize PETSc
889 PetscCall(PetscInitialize(argc, argv, nullptr, nullptr));
890 // PETSc has now loaded and initialized its BLAS backend.
891 vutils::checkOpenBlasThreading();
892 // initialize the Mpi singleton.
893 singleton();
894 return 0;
895 }
896
897 private:
899 inline constexpr static int s_default_thread_required = MPI_THREAD_SINGLE;
901 static MpiHandle& singleton() {
902 static MpiHandle mpi;
903 return mpi;
904 }
906 static int finalize() {
907 if (isPetscInitialized() && !isPetscFinalized()) {
908 PetscCall(PetscFinalize());
909 }
910 if (isInitialized() && !isFinalized()) {
911 MPI_Finalize();
912 }
913 return 0;
914 }
916 ~MpiHandle() { finalize(); }
918 MpiHandle() = default;
919};
