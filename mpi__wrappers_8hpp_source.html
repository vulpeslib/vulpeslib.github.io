53template <typename T>
54MPI_Datatype getMpiType() {
55 if constexpr (std::is_same_v<T, unsigned char>) {
56 return MPI_UNSIGNED_CHAR;
57 } else if constexpr (std::is_same_v<T, char>) {
58 return MPI_CHAR;
59 } else if constexpr (std::is_same_v<T, short>) { // NOLINT(runtime/int)
60 return MPI_SHORT;
61 } else if constexpr (std::is_same_v<T, int>) {
62 return MPI_INT;
63 } else if constexpr (std::is_same_v<T, long>) { // NOLINT(runtime/int)
64 return MPI_LONG;
65 } else if constexpr (std::is_same_v<T, long long>) { // NOLINT(runtime/int)
66 return MPI_LONG_LONG;
67 } else if constexpr (std::is_same_v<
68 T, unsigned short>) { // NOLINT(runtime/int)
69 return MPI_UNSIGNED_SHORT;
70 } else if constexpr (std::is_same_v<T, unsigned int>) {
71 return MPI_UNSIGNED;
72 } else if constexpr (std::is_same_v<
73 T, unsigned long>) { // NOLINT(runtime/int)
74 return MPI_UNSIGNED_LONG;
75 } else if constexpr (std::is_same_v<
76 T, unsigned long long>) { // NOLINT(runtime/int)
77 return MPI_UNSIGNED_LONG_LONG;
78 } else if constexpr (std::is_same_v<T, float>) {
79 return MPI_FLOAT;
80 } else if constexpr (std::is_same_v<T, double>) {
81 return MPI_DOUBLE;
82 } else if constexpr (std::is_same_v<T, long double>) {
83 return MPI_LONG_DOUBLE;
84 } else {
85 static_assert(sizeof(T) * 0 && "Unknown conversion to MPI type.");
86 }
87}

567template <typename T, int N, int M, int Options>
568void bcast(
569 Eigen::Matrix<T, N, M, Options>& data,
570 int root,
571 MPI_Comm communicator = MPI_COMM_WORLD) {
572 std::size_t nrows = 0, ncols = 0;
573 if (mpi::rank() == root) {
574 nrows = data.rows();
575 ncols = data.cols();
576 }
577 Profiler::startInterval("MPI-bcast");
578 MPI_Bcast(&nrows, 1, getMpiType<std::size_t>(), root, communicator);
579 MPI_Bcast(&ncols, 1, getMpiType<std::size_t>(), root, communicator);
580 if (!(mpi::rank() == root)) {
581 data.resize(nrows, ncols);
582 }
583 MPI_Bcast(data.data(), nrows * ncols, getMpiType<T>(), root, communicator);
584 Profiler::stopInterval("MPI-bcast");
585}

654// assume everyone knows count
655inline auto computeChuncks(int32_t count, int32_t stride = 1) {
656 const int32_t comm_size = mpi::size();
657 std::vector<int32_t> displacements(comm_size), counts(comm_size);
658
659 const int32_t chunck_sz = count / comm_size;
660
661 for (int32_t i = 0; i < comm_size; i++) {
662 counts[i] = stride * (chunck_sz + (i < (count % comm_size)));
663 }
664
665 displacements[0] = 0;
666 for (int32_t i = 1; i < comm_size; i++) {
667 displacements[i] = displacements[i - 1] + counts[i - 1];
668 }
669
670 return std::pair{displacements, counts};
671}

858class MpiHandle {
859 public:
861 static int init(
862 int& argc,
863 char**& argv,
864 int required = s_default_thread_required,
865 int* provided = nullptr) {
866 return init(&argc, &argv, required, provided);
867 }
869 static int init(
870 int* argc = nullptr,
871 char*** argv = nullptr,
872 int required = s_default_thread_required,
873 int* provided = nullptr) {
874 assert(!isInitialized() && "MPI already initialized!");
875 int mpi_res;
876 if (required == MPI_THREAD_SINGLE) {
877 mpi_res = MPI_Init(argc, argv);
878 } else {
879 mpi_res = MPI_Init_thread(argc, argv, required, provided);
880 }
881 if (mpi_res != MPI_SUCCESS) {
882 std::cout << "Error during MPI_Init()!" << std::endl;
883 std::exit(-1);
884 }
885 // Configure VTK before any VTK parallel operation.
886 vutils::configureVtkThreading();
887 // Initialize PETSc
888 PetscCall(PetscInitialize(argc, argv, nullptr, nullptr));
889 // PETSc has now loaded and initialized its BLAS backend.
890 vutils::checkOpenBlasThreading();
891 // initialize the Mpi singleton.
892 singleton();
893 return 0;
894 }
895
896 private:
898 inline constexpr static int s_default_thread_required = MPI_THREAD_SINGLE;
900 static MpiHandle& singleton() {
901 static MpiHandle mpi;
902 return mpi;
903 }
905 static int finalize() {
906 if (isPetscInitialized() && !isPetscFinalized()) {
907 PetscCall(PetscFinalize());
908 }
909 if (isInitialized() && !isFinalized()) {
910 MPI_Finalize();
911 }
912 return 0;
913 }
915 ~MpiHandle() { finalize(); }
917 MpiHandle() = default;
918};
