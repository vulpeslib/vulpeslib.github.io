55template <typename MeshType, typename FeSpaceType>
56Eigen::MatrixXd projectSolution(
57 const dofs::DofHandler<MeshType>& dof_handler,
58 const assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
59 const la::Vector& solution) {
60 static_assert(
61 !MeshType::parallel(),
62 "projectSolution() on full mesh is not supported for parallel meshes. "
63 "Use projectSolutionSubmesh() instead.");
64
65 using index_t = MeshType::index_t;
66 using Point = MeshType::SelfPoint;
67
68 const auto& mesh = cell_manager.getMesh();
69 const auto n_cells = mesh.numCells();
70 const auto& vtxs = mesh.getVertexes();
71 const auto& fe_space = cell_manager.getFeSpace();
72 const auto& basis = fe_space.getBasisFast();
73
74 constexpr bool need_scaling = FeSpaceType::needBasisScaling();
75
76 const auto bs = solution.getBlockSize();
77 const auto sol_data = solution.toEigenMap();
78
79 Eigen::MatrixXd solution_on_vtxs = Eigen::MatrixXd::Zero(bs, vtxs.size());
80 for (index_t cell_id = 0; cell_id < n_cells; ++cell_id) {
81 const auto& cell = cell_manager.getCells()[cell_id];
82 const double scaling =
83 need_scaling
84 ? std::sqrt(cell_manager.getMapping(cell_id)->jacobianDet({}))
85 : 1.0;
86 const auto dof_ids = dof_handler.getCellToDof()[cell_id];
87 const auto cell_vtxs = mesh.getCellToVtxMap()[cell_id];
88 for (index_t loc_vtx_id = 0; loc_vtx_id < cell_vtxs.size(); ++loc_vtx_id) {
89 const Point ref_pt = cell->vtxs().col(loc_vtx_id);
90 for (std::size_t fn_id = 0; fn_id < basis.size(); ++fn_id) {
91 const double fn_eval = basis[fn_id].evaluate(ref_pt);
92 for (PetscInt b = 0; b < bs; ++b) {
93 const double coef = sol_data[dof_ids[fn_id] * bs + b] / scaling;
94 solution_on_vtxs(b, cell_vtxs[loc_vtx_id]) += coef * fn_eval;
95 }
96 }
97 }
98 }
99 // if the space is discontinuous we average on the vertices
100 if (fe_space.getNumDofsForSubdimEntity(0) == 0) {
101 for (std::size_t vtx_id = 0; vtx_id < vtxs.size(); ++vtx_id) {
102 for (PetscInt b = 0; b < bs; ++b) {
103 solution_on_vtxs(b, vtx_id) /= mesh.getVtxToCellMap()[vtx_id].size();
104 }
105 }
106 }
107 return solution_on_vtxs;
108}

134template <typename MeshType, typename FeSpaceType>
135datastructure::CrsMap<double, typename MeshType::index_t>
136projectSolutionSubmesh(
137 const geometry::MeshExporterSubmesh<MeshType>& mesh_exporter,
138 const dofs::DofHandler<MeshType>& dof_handler,
139 const assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
140 const la::Vector& solution) {
141 using index_t = MeshType::index_t;
142 using Point = MeshType::SelfPoint;
143
144 if constexpr (MeshType::parallel()) {
145 PetscCheckAbort(
146 dof_handler.hasRenumbering(), PETSC_COMM_WORLD,
147 PETSC_ERR_ARG_WRONGSTATE,
148 "projectSolutionSubmesh in parallel needs renumberMpiSubdomainWise() "
149 "(dof_distrib_type 2)");
150 }
151
152 const auto& mesh = cell_manager.getMesh();
153 const index_t n_cells = [&]() -> index_t {
154 if constexpr (MeshType::parallel()) {
155 return mesh.nOwnedCells();
156 } else {
157 return mesh.numCells();
158 }
159 }();
160 const auto& fe_space = cell_manager.getFeSpace();
161 const auto& basis = fe_space.getBasisFast();
162 const auto& vtk_grids = mesh_exporter.getVtkGrids();
163 constexpr bool need_scaling = FeSpaceType::needBasisScaling();
164
165 const auto bs = solution.getBlockSize();
166 const auto sol_data = solution.toLocalEigenMap();
167 // range of global DoF IDs that the rank owns, in serial execution all DoFs
168 const auto ownership_range = solution.getOwnershipRange();
169
170 // helper to read locally-owned coefficient for global
171 // DoF given by dof_id and component b
172 const auto get_coef = [&](index_t dof_id, PetscInt b) -> double {
173 const PetscInt global_scalar_id = static_cast<PetscInt>(dof_id) * bs + b;
174 assert(global_scalar_id >= ownership_range[0]);
175 assert(global_scalar_id < ownership_range[1]);
176 const PetscInt local_scalar_id = global_scalar_id - ownership_range[0];
177 assert(local_scalar_id >= 0);
178 assert(local_scalar_id < sol_data.size());
179 return static_cast<double>(sol_data[local_scalar_id]);
180 };
181
182 datastructure::CrsMap<double, index_t> solution_on_vtxs;
183 solution_on_vtxs.indices().push_back(0);
184 std::vector<double> cell_vtx_evals;
185 assert(static_cast<index_t>(vtk_grids.size()) == n_cells);
186 for (index_t cell_id = 0; cell_id < n_cells; ++cell_id) {
187 const auto dof_ids = dof_handler.getCellToDof()[cell_id];
188 const double scaling =
189 need_scaling
190 ? std::sqrt(cell_manager.getMapping(cell_id)->jacobianDet({}))
191 : 1.0;
192 auto vtk_grid = vtk_grids[cell_id];
193 vtkPoints* points = vtk_grid->GetPoints();
194 const auto n_loc_vtxs = points->GetNumberOfPoints();
195 cell_vtx_evals.resize(n_loc_vtxs * bs);
196 std::fill(cell_vtx_evals.begin(), cell_vtx_evals.end(), 0.0);
197 for (index_t loc_vtx_id = 0; loc_vtx_id < n_loc_vtxs; ++loc_vtx_id) {
198 static_assert(
199 Point::RowsAtCompileTime > 0 && Point::RowsAtCompileTime <= 3);
200 double point_buffer[3];
201 points->GetPoint(loc_vtx_id, point_buffer);
202 Eigen::Map<Point> ref_pt(point_buffer);
203 cell_manager.getMapping(cell_id)->transformInv_(ref_pt);
204 for (std::size_t fn_id = 0; fn_id < basis.size(); ++fn_id) {
205 const double fn_eval = basis[fn_id].evaluate(ref_pt);
206 for (PetscInt b = 0; b < bs; ++b) {
207 const double coef = get_coef(dof_ids[fn_id], b) / scaling;
208 cell_vtx_evals[loc_vtx_id * bs + b] += coef * fn_eval;
209 }
210 }
211 }
212 solution_on_vtxs.pushBack(cell_vtx_evals);
213 }
214 return solution_on_vtxs;
215}
