64template <typename MeshType, typename FeSpaceType>
65Eigen::MatrixXd projectSolution(
66 const dofs::DofHandler<MeshType>& dof_handler,
67 const assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
68 const la::Vector& solution) {
69 static_assert(
70 !MeshType::parallel(),
71 "projectSolution() on full mesh is not supported for parallel meshes. "
72 "Use projectSolutionSubmesh() instead.");
73
74 using index_t = MeshType::index_t;
75 using Point = MeshType::SelfPoint;
76
77 const auto& mesh = cell_manager.getMesh();
78 const auto n_cells = mesh.numCells();
79 const auto& vtxs = mesh.getVertexes();
80 const auto& fe_space = cell_manager.getFeSpace();
81 const auto& basis = fe_space.getBasisFast();
82
83 // TODO: manage MPI --> (Francesco) I'd avoid to parallelize
84 // this function cause final step (averaging) is non independent
85 // (on shared vertexes)
86 const auto bs = solution.getBlockSize();
87 const auto sol_data = solution.toEigenMap();
88
89 Eigen::MatrixXd solution_on_vtxs = Eigen::MatrixXd::Zero(bs, vtxs.size());
90 for (index_t cell_id = 0; cell_id < n_cells; ++cell_id) {
91 const auto& cell = cell_manager.getCells()[cell_id];
92 const auto dof_ids = dof_handler.getCellToDof()[cell_id];
93 const auto cell_vtxs = mesh.getCellToVtxMap()[cell_id];
94 for (index_t loc_vtx_id = 0; loc_vtx_id < cell_vtxs.size(); ++loc_vtx_id) {
95 const Point ref_pt = cell->vtxs().col(loc_vtx_id);
96 for (std::size_t fn_id = 0; fn_id < basis.size(); ++fn_id) {
97 const double fn_eval = basis[fn_id].evaluate(ref_pt);
98 for (PetscInt b = 0; b < bs; ++b) {
99 const double coef = sol_data[dof_ids[fn_id] * bs + b];
100 solution_on_vtxs(b, cell_vtxs[loc_vtx_id]) += coef * fn_eval;
101 }
102 }
103 }
104 }
105 // if the space is discontinuous we average on the vertices
106 if (fe_space.getNumDofsForSubdimEntity(0) == 0) {
107 for (std::size_t vtx_id = 0; vtx_id < vtxs.size(); ++vtx_id) {
108 for (PetscInt b = 0; b < bs; ++b) {
109 solution_on_vtxs(b, vtx_id) /= mesh.getVtxToCellMap()[vtx_id].size();
110 }
111 }
112 }
113 return solution_on_vtxs;
114}

140template <typename MeshType, typename FeSpaceType>
141datastructure::CrsMap<double, typename MeshType::index_t>
142projectSolutionSubmesh(
143 const geometry::MeshExporterSubmesh<MeshType>& mesh_exporter,
144 const dofs::DofHandler<MeshType>& dof_handler,
145 const assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
146 const la::Vector& solution) {
147 using index_t = MeshType::index_t;
148 using Point = MeshType::SelfPoint;
149
150 const auto& mesh = cell_manager.getMesh();
151 const index_t n_cells = [&]() -> index_t {
152 if constexpr (MeshType::parallel()) {
153 return mesh.nOwnedCells();
154 } else {
155 return mesh.numCells();
156 }
157 }();
158 const auto& fe_space = cell_manager.getFeSpace();
159 const auto& basis = fe_space.getBasisFast();
160 const auto& vtk_grids = mesh_exporter.getVtkGrids();
161
162 const auto bs = solution.getBlockSize();
163 const auto sol_data = solution.toLocalEigenMap();
164 // range of global DoF IDs that the rank owns, in serial execution all DoFs
165 const auto ownership_range = solution.getOwnershipRange();
166
167 // helper to read locally-owned coefficient for global
168 // DoF given by dof_id and component b
169 const auto get_coef = [&](index_t dof_id, PetscInt b) -> double {
170 const PetscInt global_scalar_id = static_cast<PetscInt>(dof_id) * bs + b;
171 assert(global_scalar_id >= ownership_range[0]);
172 assert(global_scalar_id < ownership_range[1]);
173 const PetscInt local_scalar_id = global_scalar_id - ownership_range[0];
174 assert(local_scalar_id >= 0);
175 assert(local_scalar_id < sol_data.size());
176 return static_cast<double>(sol_data[local_scalar_id]);
177 };
178
179 datastructure::CrsMap<double, index_t> solution_on_vtxs;
180 solution_on_vtxs.indices().push_back(0);
181 std::vector<double> cell_vtx_evals;
182 assert(static_cast<index_t>(vtk_grids.size()) == n_cells);
183 for (index_t cell_id = 0; cell_id < n_cells; ++cell_id) {
184 const auto dof_ids = dof_handler.getCellToDof()[cell_id];
185 auto vtk_grid = vtk_grids[cell_id];
186 vtkPoints* points = vtk_grid->GetPoints();
187 const auto n_loc_vtxs = points->GetNumberOfPoints();
188 cell_vtx_evals.resize(n_loc_vtxs * bs);
189 std::fill(cell_vtx_evals.begin(), cell_vtx_evals.end(), 0.0);
190 for (index_t loc_vtx_id = 0; loc_vtx_id < n_loc_vtxs; ++loc_vtx_id) {
191 static_assert(
192 Point::RowsAtCompileTime > 0 && Point::RowsAtCompileTime <= 3);
193 double point_buffer[3];
194 points->GetPoint(loc_vtx_id, point_buffer);
195 Eigen::Map<Point> ref_pt(point_buffer);
196 cell_manager.getMapping(cell_id)->transformInv_(ref_pt);
197 for (std::size_t fn_id = 0; fn_id < basis.size(); ++fn_id) {
198 const double fn_eval = basis[fn_id].evaluate(ref_pt);
199 for (PetscInt b = 0; b < bs; ++b) {
200 const double coef = get_coef(dof_ids[fn_id], b);
201 cell_vtx_evals[loc_vtx_id * bs + b] += coef * fn_eval;
202 }
203 }
204 }
205 solution_on_vtxs.pushBack(cell_vtx_evals);
206 }
207 return solution_on_vtxs;
208}
