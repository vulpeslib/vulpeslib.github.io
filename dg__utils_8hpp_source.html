89template <typename MeshType, typename FeSpaceType, typename ExprType>
90double integrateFeExpression(
91 const MeshType& mesh,
92 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
93 const ExprType& expression,
94 std::optional<uint32_t> quad_degree = std::nullopt) {
95 const uint32_t quad_order = static_cast<uint32_t>(expression.computeOrder());
96 assemble::ScalarExpressionIntegrator<MeshType, FeSpaceType, ExprType>
97 integrator(
98 mesh, cell_manager, expression,
99 quad_degree.value_or(2 * quad_order + 1));
100 double result = integrator.integrate();
101 if constexpr (MeshType::parallel()) {
102 MPI_Allreduce(
103 MPI_IN_PLACE, &result, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
104 }
105 return result;
106}

126template <
127 typename MeshType,
128 typename FeSpaceType,
129 typename DofHandlerType,
130 typename ExprType>
131double computeL2Error(
132 const MeshType& mesh,
133 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
134 const DofHandlerType& dof_handler,
135 const la::Vector& u_h,
136 const ExprType& u_exact_expr,
137 std::optional<uint32_t> quad_degree = std::nullopt) {
138 constexpr std::size_t Dim = MeshType::self_dim;
139 // Create the numerical solution as an FE expression
140 fe::FeExactExpressionFeProjection<Dim, FeSpaceType, DofHandlerType> u_h_expr(
141 u_h, cell_manager.getFeSpace(), dof_handler);
142
143 // Form the difference expression: (u_h - u_exact)
144 auto diff_expr = u_h_expr - u_exact_expr;
145
146 // Form the squared difference: (u_h - u_exact)^2
147 auto diff_squared = diff_expr * diff_expr;
148
149 // Integrate the squared difference over the domain
150 double l2_error_squared =
151 integrateFeExpression(mesh, cell_manager, diff_squared, quad_degree);
152
153 // Return the L2 error (square root of the integral)
154 return std::sqrt(std::max(0.0, l2_error_squared));
155}

173template <
174 typename MeshType,
175 typename FeSpaceType,
176 typename DofHandlerType,
177 typename ExprType>
178double computeH1SemiError(
179 const MeshType& mesh,
180 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
181 const DofHandlerType& dof_handler,
182 const la::Vector& u_h,
183 const ExprType& grad_u_exact_expr,
184 std::optional<uint32_t> quad_degree = std::nullopt) {
185 constexpr std::size_t Dim = MeshType::self_dim;
186
187 fe::FeGradProjectionExpr<Dim, FeSpaceType, DofHandlerType> grad_uh_expr(
188 u_h, cell_manager.getFeSpace(), dof_handler);
189
190 auto diff_expr = grad_uh_expr - grad_u_exact_expr;
191 auto diff_squared = fe::inner(diff_expr, diff_expr);
192
193 double h1_semi_error_squared =
194 integrateFeExpression(mesh, cell_manager, diff_squared, quad_degree);
195
196 return std::sqrt(std::max(0.0, h1_semi_error_squared));
197}

228template <std::size_t Dim>
229auto createPenaltyExpression(std::size_t order, double gamma) {
230 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
231 [order, gamma](
232 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
233 const auto& cell = *cells[fe_values.getCellId()];
234 const double penalty = gamma * order * order;
235 const double p_this = penalty / cell.diameter();
236 if (fe_values.hasFaceNeigh()) {
237 assert(&fe_values.neighValues());
238 const auto neigh_cell_id = fe_values.neighValues().getCellId();
239 const auto& neigh_cell = *cells[neigh_cell_id];
240 const double p_neigh = penalty / neigh_cell.diameter();
241 return std::max(p_this, p_neigh);
242 } else {
243 return p_this;
244 }
245 });
246}

258template <std::size_t Dim>
259auto createPenaltyExpressionKb(std::size_t order, double gamma) {
260 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
261 [order, gamma](
262 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
263 const auto& cell = *cells[fe_values.getCellId()];
264 const auto face_id = fe_values.getFaceId();
265 const double penalty = gamma * order * order;
266 const double Cinv = cell.measure() / cell.faceKb()[face_id];
267 const double facet_diam = cell.faceDiam()[face_id];
268 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
269 if (fe_values.hasFaceNeigh()) {
270 assert(&fe_values.neighValues());
271 const auto neigh_cell_id = fe_values.neighValues().getCellId();
272 const auto& neigh_cell = *cells[neigh_cell_id];
273 const auto neigh_face_id = fe_values.neighValues().getFaceId();
274 const double Cinv_neigh =
275 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
276 const double p_neigh =
277 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
278 return std::max(p_this, p_neigh);
279 } else {
280 return p_this;
281 }
282 });
283}

298template <std::size_t Dim>
299auto createPenaltyExpressionHarmonic(std::size_t order, double gamma) {
300 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
301 [order, gamma](
302 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
303 const auto& cell = *cells[fe_values.getCellId()];
304 const auto face_id = fe_values.getFaceId();
305 const double facet_diam = cell.faceDiam()(face_id);
306 const double penalty = gamma * order * order;
307 const double Cinv = cell.measure() / cell.faceKb()[face_id];
308 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
309
310 if (fe_values.hasFaceNeigh()) {
311 const auto neigh_cell_id = fe_values.neighValues().getCellId();
312 const auto& neigh_cell = *cells[neigh_cell_id];
313 const auto neigh_face_id = fe_values.neighValues().getFaceId();
314 const double Cinv_neigh =
315 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
316 const double p_neigh =
317 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
318 // Media armonica tra p_this e p_neigh
319 return 2.0 * p_this * p_neigh / (p_this + p_neigh);
320 } else {
321 return p_this;
322 }
323 });
324}

332template <std::size_t Dim, typename VelocityFunc>
333class FeExpressionUpwindFromFunctionWrapper
334 : public fe::FeExpressionBase<
335 Dim,
336 FeExpressionUpwindFromFunctionWrapper<Dim, VelocityFunc>> {
337 public:
338 static constexpr bool is_leaf = true;
339 static constexpr uint8_t is_bilinear = fe::IsNothing;
340
341 explicit constexpr FeExpressionUpwindFromFunctionWrapper(
342 const VelocityFunc& velocity_func)
343 : m_velocity_func(velocity_func) {}
344 FeExpressionUpwindFromFunctionWrapper(
345 const FeExpressionUpwindFromFunctionWrapper&) = delete;
346 FeExpressionUpwindFromFunctionWrapper& operator=(
347 const FeExpressionUpwindFromFunctionWrapper&) = delete;
348 FeExpressionUpwindFromFunctionWrapper(
349 FeExpressionUpwindFromFunctionWrapper&&) = delete;
350 FeExpressionUpwindFromFunctionWrapper& operator=(
351 FeExpressionUpwindFromFunctionWrapper&&) = delete;
352
353 template <std::size_t I = 0, std::size_t J = 0>
354 decltype(auto) evaluate(
355 const fe::FeValues<Dim>&, std::size_t, std::size_t, std::size_t q) const {
356 return m_evals[q];
357 }
358
359 constexpr fe::UpdateFlagsType computeUpdateFlags() const {
360 return fe::UpdateQuadraturePts | fe::UpdateNormals;
361 }
362 constexpr std::size_t computeOrder() const { return 0; }
363
364 constexpr bool precompute(
365 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>&) const {
366 const Eigen::Index nqpts = fe_values.weights().size();
367 assert(fe_values.qpts().cols() == nqpts);
368 m_evals.resize(nqpts);
369 const auto& normal = fe_values.getFaceNormal();
370 for (Eigen::Index q = 0; q < nqpts; ++q) {
371 const auto vel = m_velocity_func(fe_values.qpts().col(q));
372 const double vel_dot_n = vel.dot(normal);
373 m_evals[q] = 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
374 }
375 return true;
376 }
377
378 protected:
379 const VelocityFunc m_velocity_func;
380 mutable std::vector<double> m_evals;
381};

413template <std::size_t Dim>
414auto createUpwindFluxConstant(double vx, double vy = 0.0, double vz = 0.0) {
415 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
416 [vx, vy, vz](
417 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
418 const auto& cell = *cells[fe_values.getCellId()];
419 const auto face_id = fe_values.getFaceId();
420 const auto& normal = cell.normals().col(face_id);
421
422 // Compute velocity . n
423 double vel_dot_n = vx * normal(0);
424 if constexpr (Dim >= 2) {
425 vel_dot_n += vy * normal(1);
426 }
427 if constexpr (Dim >= 3) {
428 vel_dot_n += vz * normal(2);
429 }
430
431 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
432 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
433 });
434}

451template <std::size_t Dim, typename VelocityFunc>
452auto createUpwindFluxVariable(VelocityFunc velocity_func) {
453 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
454 [velocity_func](
455 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
456 const auto& cell = *cells[fe_values.getCellId()];
457 const auto face_id = fe_values.getFaceId();
458 const auto& normal = cell.normals().col(face_id);
459
460 // Face centroid: average of the two edge vertices for 2D polygons.
461 const Eigen::Index nvtxs = cell.vtxs().cols();
462 const Eigen::Vector<double, Dim> face_center =
463 0.5 *
464 (cell.vtxs().col(face_id) + cell.vtxs().col((face_id + 1) % nvtxs));
465
466 // Evaluate velocity at face center and compute vel . n
467 const auto vel = velocity_func(face_center);
468 const double vel_dot_n = vel.dot(normal);
469
470 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
471 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
472 });
473}

486template <typename MeshType>
487double computeError(
488 const dofs::DofHandler<MeshType>& dof_handler,
489 const la::Vector& u_h,
490 const la::Vector& u_exact,
491 const la::SparseMatrix& matrix) {
492 // Compute the residual: error_u = u_exact - u_h
493 la::Vector error_u(dof_handler, u_h.getBlockSize());
494 error_u.add(-1.0, u_h);
495 error_u.add(1.0, u_exact);
496 error_u.compress();
497
498 // Compute the error norm using the provided matrix
499 la::Vector temp(dof_handler, u_h.getBlockSize());
500 // std::cout << "Error Calc with matrix: " << std::endl;
501 matrix.vmult(error_u, temp); // temp = matrix * error_u
502
503 if (la::dot(error_u, temp) < 0) {
504 std::cout
505 << "Warning: Negative value in error computation. Setting error to 0."
506 << std::endl;
507 return 0.0;
508 } else {
509 return std::sqrt(la::dot(error_u, temp));
510 }
511}

543template <typename MeshType>
544la::Vector projectConstantOntoModalSpace(
545 const dofs::DofHandler<MeshType>& dof_handler,
546 double constant_value,
547 const la::SparseMatrix& mass_matrix,
548 const la::Vector& rhs_const) {
549 la::Vector scaled_rhs(rhs_const);
550 scaled_rhs.scale(constant_value);
551
552 la::Vector c_proj(dof_handler);
553 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
554 solver.setOperators(mass_matrix);
555 solver.solve(scaled_rhs, c_proj);
556
557 return c_proj;
558}

573template <typename MeshType>
574double forceEqualMean(
575 const dofs::DofHandler<MeshType>& dof_handler,
576 la::Vector& u,
577 const la::Vector& u_reference,
578 const la::SparseMatrix& mass_matrix,
579 const la::Vector& rhs_const,
580 double domain_measure) {
581 double avg_u = computeAverage(u, rhs_const, domain_measure);
582 double avg_reference = computeAverage(u_reference, rhs_const, domain_measure);
583
584 double shift_physical = avg_reference - avg_u;
585
586 la::Vector c_proj = projectConstantOntoModalSpace(
587 dof_handler, shift_physical, mass_matrix, rhs_const);
588
589 u.add(1.0, c_proj);
590 u.compress();
591
592 return shift_physical;
593}

603template <
604 typename MeshType,
605 typename CellManagerType,
606 typename DofHandlerType,
607 typename FeSpaceType,
608 typename Expression>
624la::Vector l2ProjectExpressionScalar(
625 const MeshType& mesh,
626 CellManagerType& cell_manager,
627 const DofHandlerType& dof_handler,
628 const FeSpaceType& fe_space,
629 const Expression& expr,
630 const la::SparseMatrix& mass_scalar) {
631 namespace fe = vulpes::fe;
632
633 fe::TestFunction v(fe_space);
634 auto wf_projection = expr * v;
635
636 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
637 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
638
639 la::Vector rhs(dof_handler);
640 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
641 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
642 int_smplx_projection.integrate(cell_id);
643 rhs.insertValues(
644 int_smplx_projection.getIntegral(&wf_projection),
645 dof_handler.getCellToDof()[cell_id]);
646 }
647 rhs.compress();
648
649 la::Vector projected(dof_handler);
650 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
651 solver.setOperators(mass_scalar);
652 solver.solve(rhs, projected);
653 return projected;
654}

693template <
694 std::size_t VecSize,
695 std::size_t NFields,
696 typename MeshType,
697 typename CellManagerType,
698 typename DofHandlerType,
699 typename FeSpaceType,
700 typename Expression>
722la::Vector l2ProjectExpressionBlockedSubset(
723 const MeshType& mesh,
724 CellManagerType& cell_manager,
725 const DofHandlerType& dof_handler,
726 const FeSpaceType& fe_space,
727 const Expression& expr,
728 std::size_t component_offset,
729 const la::SparseMatrix& mass_blocked) {
730 namespace fe = vulpes::fe;
731
732 assert(component_offset + VecSize <= NFields);
733
734 fe::TestFunction<FeSpaceType, 0, VecSize> v(fe_space);
735 auto wf_projection = fe::inner(expr, v);
736
737 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
738 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
739
740 la::Vector rhs(dof_handler, NFields);
741 const auto& cell_to_dof = dof_handler.getCellToDof();
742 const Eigen::Index ndof_comp = fe_space.getBasis().size();
743 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
744
745 std::array<PetscScalar, NFields> block_values{};
746 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
747 int_smplx_projection.integrate(cell_id);
748 const auto local = int_smplx_projection.getIntegral(&wf_projection);
749
750 for (Eigen::Index i = 0; i < ndof_comp; ++i) {
751 block_values.fill(0.0);
752 for (std::size_t c = 0; c < VecSize; ++c) {
753 block_values[component_offset + c] = local(
754 i * static_cast<Eigen::Index>(VecSize) +
755 static_cast<Eigen::Index>(c),
756 0);
757 }
758 const PetscInt block_index = cell_to_dof[cell_id][i];
759 rhs.insertValuesBlocked(block_values.data(), &block_index, 1, ADD_VALUES);
760 }
761 }
762 rhs.compress();
763
764 la::Vector projected(dof_handler, NFields);
765 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
766 solver.setOperators(mass_blocked);
767 solver.solve(rhs, projected);
768 return projected;
769}
