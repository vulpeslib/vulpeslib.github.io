70inline double sqrtNonNegative(double x, double tol = 1e-12) {
71 if (std::isnan(x)) {
72 return x;
73 }
74 if (x < -tol) {
75 pcout << "Warning: square root of negative value " << x << std::endl;
76 }
77 return std::sqrt(std::max(x, 0.0));
78}

108template <typename MeshType, typename FeSpaceType, typename ExprType>
109double integrateFeExpression(
110 const MeshType& mesh,
111 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
112 const ExprType& expression,
113 std::optional<uint32_t> quad_degree = std::nullopt) {
114 constexpr std::size_t Dim = MeshType::self_dim;
115 constexpr std::size_t max_tabled = Dim == 3
116 ? qrule::num_tetrahedron_tabled_rules
117 : qrule::num_triangle_tabled_rules;
118 const uint32_t degree = quad_degree.value_or(
119 static_cast<uint32_t>(1 + expression.computeOrder()));
120 if (Dim > 1 && degree > max_tabled) {
121 pcout << "Warning: quadrature degree " << degree
122 << " is above the tabled simplex rules" << std::endl;
123 }
124 assemble::ScalarExpressionIntegrator<MeshType, FeSpaceType, ExprType>
125 integrator(mesh, cell_manager, expression, degree);
126 double result = integrator.integrate();
127 if constexpr (MeshType::parallel()) {
128 MPI_Allreduce(
129 MPI_IN_PLACE, &result, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
130 }
131 return result;
132}

154template <
155 typename MeshType,
156 typename FeSpaceType,
157 typename DofHandlerType,
158 typename ExprType>
159double computeL2Error(
160 const MeshType& mesh,
161 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
162 const DofHandlerType& dof_handler,
163 const la::Vector& u_h,
164 const ExprType& u_exact_expr,
165 std::optional<uint32_t> quad_degree = std::nullopt) {
166 constexpr std::size_t Dim = MeshType::self_dim;
167 // Create the numerical solution as an FE expression
168 fe::FeExactExpressionFeProjection<Dim, FeSpaceType, DofHandlerType> u_h_expr(
169 u_h, cell_manager.getFeSpace(), dof_handler);
170
171 // Form the difference expression: (u_h - u_exact)
172 auto diff_expr = u_h_expr - u_exact_expr;
173
174 // Form the squared difference: (u_h - u_exact)^2
175 auto diff_squared = diff_expr * diff_expr;
176
177 // Integrate the squared difference over the domain
178 const std::size_t p = cell_manager.getFeSpace().order();
179 double l2_error_squared = integrateFeExpression(
180 mesh, cell_manager, diff_squared,
181 quad_degree.value_or(
182 std::max(1 + diff_squared.computeOrder(), 2 * p + 7)));
183
184 // Return the L2 error (square root of the integral)
185 return sqrtNonNegative(l2_error_squared);
186}

206template <
207 typename MeshType,
208 typename FeSpaceType,
209 typename DofHandlerType,
210 typename ExprType>
211double computeH1SemiError(
212 const MeshType& mesh,
213 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
214 const DofHandlerType& dof_handler,
215 const la::Vector& u_h,
216 const ExprType& grad_u_exact_expr,
217 std::optional<uint32_t> quad_degree = std::nullopt) {
218 constexpr std::size_t Dim = MeshType::self_dim;
219
220 fe::FeGradProjectionExpr<Dim, FeSpaceType, DofHandlerType> grad_uh_expr(
221 u_h, cell_manager.getFeSpace(), dof_handler);
222
223 auto diff_expr = grad_uh_expr - grad_u_exact_expr;
224 auto diff_squared = fe::inner(diff_expr, diff_expr);
225
226 const std::size_t p = cell_manager.getFeSpace().order();
227 double h1_semi_error_squared = integrateFeExpression(
228 mesh, cell_manager, diff_squared,
229 quad_degree.value_or(
230 std::max(1 + diff_squared.computeOrder(), 2 * p + 7)));
231
232 return sqrtNonNegative(h1_semi_error_squared);
233}

248 const la::Vector& u_h,
249 const la::SparseMatrix& matrix) {
250 la::Vector temp(dof_handler, u_h.getBlockSize());

264template <std::size_t Dim>
265auto createPenaltyExpression(std::size_t order, double gamma) {
266 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
267 [order, gamma](
268 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
269 const auto& cell = *cells[fe_values.getCellId()];
270 const double penalty = gamma * order * order;
271 const double p_this = penalty / cell.diameter();
272 if (fe_values.hasFaceNeigh()) {
273 assert(&fe_values.neighValues());
274 const auto neigh_cell_id = fe_values.neighValues().getCellId();
275 const auto& neigh_cell = *cells[neigh_cell_id];
276 const double p_neigh = penalty / neigh_cell.diameter();
277 return std::max(p_this, p_neigh);
278 } else {
279 return p_this;
280 }
281 });
282}

288template <std::size_t Dim, typename CellType>
289double faceMeasureForPenalty(const CellType& cell, std::size_t face_id) {
290 if constexpr (Dim == 3) {
291 assert(cell.faceMeasures().size() > 0);
292 return cell.faceMeasures()[face_id];
293 } else {
294 return cell.faceDiam()[face_id];

309template <std::size_t Dim>
310auto createPenaltyExpressionKb(std::size_t order, double gamma) {
311 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
312 [order, gamma](
313 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
314 const auto& cell = *cells[fe_values.getCellId()];
315 const auto face_id = fe_values.getFaceId();
316 const double penalty = gamma * order * order;
317 const double Cinv = cell.measure() / cell.faceKb()[face_id];
318 const double facet_measure = faceMeasureForPenalty<Dim>(cell, face_id);
319 const double p_this = penalty * Cinv * (facet_measure / cell.measure());
320 if (fe_values.hasFaceNeigh()) {
321 assert(&fe_values.neighValues());
322 const auto neigh_cell_id = fe_values.neighValues().getCellId();
323 const auto& neigh_cell = *cells[neigh_cell_id];
324 const auto neigh_face_id = fe_values.neighValues().getFaceId();
325 const double Cinv_neigh =
326 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
327 const double p_neigh =
328 penalty * Cinv_neigh * (facet_measure / neigh_cell.measure());
329 return std::max(p_this, p_neigh);
330 } else {
331 return p_this;
332 }
333 });
334}

350template <std::size_t Dim>
351auto createPenaltyExpressionHarmonic(std::size_t order, double gamma) {
352 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
353 [order, gamma](
354 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
355 const auto& cell = *cells[fe_values.getCellId()];
356 const auto face_id = fe_values.getFaceId();
357 const double facet_measure = faceMeasureForPenalty<Dim>(cell, face_id);
358 const double penalty = gamma * order * order;
359 const double Cinv = cell.measure() / cell.faceKb()[face_id];
360 const double p_this = penalty * Cinv * (facet_measure / cell.measure());
361
362 if (fe_values.hasFaceNeigh()) {
363 const auto neigh_cell_id = fe_values.neighValues().getCellId();
364 const auto& neigh_cell = *cells[neigh_cell_id];
365 const auto neigh_face_id = fe_values.neighValues().getFaceId();
366 const double Cinv_neigh =
367 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
368 const double p_neigh =
369 penalty * Cinv_neigh * (facet_measure / neigh_cell.measure());
370 // Media armonica tra p_this e p_neigh
371 return 2.0 * p_this * p_neigh / (p_this + p_neigh);
372 } else {
373 return p_this;
374 }
375 });
376}

384template <std::size_t Dim, typename VelocityFunc>
385class FeExpressionUpwindFromFunctionWrapper
386 : public fe::FeExpressionBase<
387 Dim,
388 FeExpressionUpwindFromFunctionWrapper<Dim, VelocityFunc>> {
389 public:
390 static constexpr bool is_leaf = true;
391 static constexpr uint8_t is_bilinear = fe::IsNothing;
392
393 explicit constexpr FeExpressionUpwindFromFunctionWrapper(
394 const VelocityFunc& velocity_func)
395 : m_velocity_func(velocity_func) {}
396 FeExpressionUpwindFromFunctionWrapper(
397 const FeExpressionUpwindFromFunctionWrapper&) = delete;
398 FeExpressionUpwindFromFunctionWrapper& operator=(
399 const FeExpressionUpwindFromFunctionWrapper&) = delete;
400 FeExpressionUpwindFromFunctionWrapper(
401 FeExpressionUpwindFromFunctionWrapper&&) = delete;
402 FeExpressionUpwindFromFunctionWrapper& operator=(
403 FeExpressionUpwindFromFunctionWrapper&&) = delete;
404
405 template <std::size_t I = 0, std::size_t J = 0>
406 decltype(auto) evaluate(
407 const fe::FeValues<Dim>&, std::size_t, std::size_t, std::size_t q) const {
408 return m_evals[q];
409 }
410
411 constexpr fe::UpdateFlagsType computeUpdateFlags() const {
412 return fe::UpdateQuadraturePts | fe::UpdateNormals;
413 }
414 constexpr std::size_t computeOrder() const { return 0; }
415
416 constexpr bool precompute(
417 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>&) const {
418 const Eigen::Index nqpts = fe_values.weights().size();
419 assert(fe_values.qpts().cols() == nqpts);
420 m_evals.resize(nqpts);
421 const auto& normal = fe_values.getFaceNormal();
422 for (Eigen::Index q = 0; q < nqpts; ++q) {
423 const auto vel = m_velocity_func(fe_values.qpts().col(q));
424 const double vel_dot_n = vel.dot(normal);
425 m_evals[q] = 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
426 }
427 return true;
428 }
429
430 protected:
431 const VelocityFunc m_velocity_func;
432 mutable std::vector<double> m_evals;
433};

465template <std::size_t Dim>
466auto createUpwindFluxConstant(double vx, double vy = 0.0, double vz = 0.0) {
467 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
468 [vx, vy, vz](
469 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
470 const auto& cell = *cells[fe_values.getCellId()];
471 const auto face_id = fe_values.getFaceId();
472 const auto& normal = cell.normals().col(face_id);
473
474 // Compute velocity . n
475 double vel_dot_n = vx * normal(0);
476 if constexpr (Dim >= 2) {
477 vel_dot_n += vy * normal(1);
478 }
479 if constexpr (Dim >= 3) {
480 vel_dot_n += vz * normal(2);
481 }
482
483 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
484 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
485 });
486}

503template <std::size_t Dim, typename VelocityFunc>
504auto createUpwindFluxVariable(VelocityFunc velocity_func) {
505 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
506 [velocity_func](
507 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
508 const auto& cell = *cells[fe_values.getCellId()];
509 const auto face_id = fe_values.getFaceId();
510 const auto& normal = cell.normals().col(face_id);
511
512 // Face centroid: average of the two edge vertices for 2D polygons.
513 const Eigen::Index nvtxs = cell.vtxs().cols();
514 const Eigen::Vector<double, Dim> face_center =
515 0.5 *
516 (cell.vtxs().col(face_id) + cell.vtxs().col((face_id + 1) % nvtxs));
517
518 // Evaluate velocity at face center and compute vel . n
519 const auto vel = velocity_func(face_center);
520 const double vel_dot_n = vel.dot(normal);
521
522 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
523 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
524 });
525}

538template <typename MeshType>
539double computeError(
540 const dofs::DofHandler<MeshType>& dof_handler,
541 const la::Vector& u_h,
542 const la::Vector& u_exact,
543 const la::SparseMatrix& matrix) {
544 // Compute the residual: error_u = u_exact - u_h
545 la::Vector error_u(dof_handler, u_h.getBlockSize());
546 error_u.add(-1.0, u_h);
547 error_u.add(1.0, u_exact);
548 error_u.compress();
549
550 // Compute the error norm using the provided matrix
551 la::Vector temp(dof_handler, u_h.getBlockSize());
552 // std::cout << "Error Calc with matrix: " << std::endl;
553 matrix.vmult(error_u, temp); // temp = matrix * error_u
554
555 if (la::dot(error_u, temp) < 0) {
556 std::cout
557 << "Warning: Negative value in error computation. Setting error to 0."
558 << std::endl;
559 return 0.0;
560 } else {
561 return std::sqrt(la::dot(error_u, temp));
562 }
563}

595template <typename MeshType>
596la::Vector projectConstantOntoModalSpace(
597 const dofs::DofHandler<MeshType>& dof_handler,
598 double constant_value,
599 const la::SparseMatrix& mass_matrix,
600 const la::Vector& rhs_const) {
601 la::Vector scaled_rhs(rhs_const);
602 scaled_rhs.scale(constant_value);
603
604 la::Vector c_proj(dof_handler);
605 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
606 solver.setOperators(mass_matrix);
607 solver.solve(scaled_rhs, c_proj);
608
609 return c_proj;
610}

625template <typename MeshType>
626double forceEqualMean(
627 const dofs::DofHandler<MeshType>& dof_handler,
628 la::Vector& u,
629 const la::Vector& u_reference,
630 const la::SparseMatrix& mass_matrix,
631 const la::Vector& rhs_const,
632 double domain_measure) {
633 double avg_u = computeAverage(u, rhs_const, domain_measure);
634 double avg_reference = computeAverage(u_reference, rhs_const, domain_measure);
635
636 double shift_physical = avg_reference - avg_u;
637
638 la::Vector c_proj = projectConstantOntoModalSpace(
639 dof_handler, shift_physical, mass_matrix, rhs_const);
640
641 u.add(1.0, c_proj);
642 u.compress();
643
644 return shift_physical;
645}

655template <
656 typename MeshType,
657 typename CellManagerType,
658 typename DofHandlerType,
659 typename FeSpaceType,
660 typename Expression>
676la::Vector l2ProjectExpressionScalar(
677 const MeshType& mesh,
678 CellManagerType& cell_manager,
679 const DofHandlerType& dof_handler,
680 const FeSpaceType& fe_space,
681 const Expression& expr,
682 const la::SparseMatrix& mass_scalar) {
683 namespace fe = vulpes::fe;
684
685 fe::TestFunction v(fe_space);
686 auto wf_projection = expr * v;
687
688 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
689 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
690
691 la::Vector rhs(dof_handler);
692 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
693 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
694 int_smplx_projection.integrate(cell_id);
695 rhs.insertValues(
696 int_smplx_projection.getIntegral(&wf_projection),
697 dof_handler.getCellToDof()[cell_id]);
698 }
699 rhs.compress();
700
701 la::Vector projected(dof_handler);
702 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
703 solver.setOperators(mass_scalar);
704 solver.solve(rhs, projected);
705 return projected;
706}

745template <
746 std::size_t VecSize,
747 std::size_t NFields,
748 typename MeshType,
749 typename CellManagerType,
750 typename DofHandlerType,
751 typename FeSpaceType,
752 typename Expression>
774la::Vector l2ProjectExpressionBlockedSubset(
775 const MeshType& mesh,
776 CellManagerType& cell_manager,
777 const DofHandlerType& dof_handler,
778 const FeSpaceType& fe_space,
779 const Expression& expr,
780 std::size_t component_offset,
781 const la::SparseMatrix& mass_blocked) {
782 namespace fe = vulpes::fe;
783
784 assert(component_offset + VecSize <= NFields);
785
786 fe::TestFunction<FeSpaceType, 0, VecSize> v(fe_space);
787 auto wf_projection = fe::inner(expr, v);
788
789 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
790 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
791
792 la::Vector rhs(dof_handler, NFields);
793 const auto& cell_to_dof = dof_handler.getCellToDof();
794 const Eigen::Index ndof_comp = fe_space.getBasis().size();
795 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
796
797 std::array<PetscScalar, NFields> block_values{};
798 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
799 int_smplx_projection.integrate(cell_id);
800 const auto local = int_smplx_projection.getIntegral(&wf_projection);
801
802 for (Eigen::Index i = 0; i < ndof_comp; ++i) {
803 block_values.fill(0.0);
804 for (std::size_t c = 0; c < VecSize; ++c) {
805 block_values[component_offset + c] = local(
806 i * static_cast<Eigen::Index>(VecSize) +
807 static_cast<Eigen::Index>(c),
808 0);
809 }
810 const PetscInt block_index = cell_to_dof[cell_id][i];
811 rhs.insertValuesBlocked(block_values.data(), &block_index, 1, ADD_VALUES);
812 }
813 }
814 rhs.compress();
815
816 la::Vector projected(dof_handler, NFields);
817 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
818 solver.setOperators(mass_blocked);
819 solver.solve(rhs, projected);
820 return projected;
821}
