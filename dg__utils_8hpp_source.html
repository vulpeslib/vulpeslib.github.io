87template <typename MeshType, typename FeSpaceType, typename ExprType>
88double integrateFeExpression(
89 const MeshType& mesh,
90 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
91 const ExprType& expression,
92 std::optional<uint32_t> quad_degree = std::nullopt) {
93 constexpr std::size_t Dim = MeshType::self_dim;
94 constexpr std::size_t max_tabled = Dim == 3
95 ? qrule::num_tetrahedron_tabled_rules
96 : qrule::num_triangle_tabled_rules;
97 const uint32_t degree = quad_degree.value_or(
98 static_cast<uint32_t>(1 + expression.computeOrder()));
99 if (Dim > 1 && degree > max_tabled) {
100 pcout << "Warning: quadrature degree " << degree
101 << " is above the tabled simplex rules" << std::endl;
102 }
103 assemble::ScalarExpressionIntegrator<MeshType, FeSpaceType, ExprType>
104 integrator(mesh, cell_manager, expression, degree);
105 double result = integrator.integrate();
106 if constexpr (MeshType::parallel()) {
107 MPI_Allreduce(
108 MPI_IN_PLACE, &result, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
109 }
110 return result;
111}

133template <
134 typename MeshType,
135 typename FeSpaceType,
136 typename DofHandlerType,
137 typename ExprType>
138double computeL2Error(
139 const MeshType& mesh,
140 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
141 const DofHandlerType& dof_handler,
142 const la::Vector& u_h,
143 const ExprType& u_exact_expr,
144 std::optional<uint32_t> quad_degree = std::nullopt) {
145 constexpr std::size_t Dim = MeshType::self_dim;
146 // Create the numerical solution as an FE expression
147 fe::FeExactExpressionFeProjection<Dim, FeSpaceType, DofHandlerType> u_h_expr(
148 u_h, cell_manager.getFeSpace(), dof_handler);
149
150 // Form the difference expression: (u_h - u_exact)
151 auto diff_expr = u_h_expr - u_exact_expr;
152
153 // Form the squared difference: (u_h - u_exact)^2
154 auto diff_squared = diff_expr * diff_expr;
155
156 // Integrate the squared difference over the domain
157 const std::size_t p = cell_manager.getFeSpace().order();
158 double l2_error_squared = integrateFeExpression(
159 mesh, cell_manager, diff_squared,
160 quad_degree.value_or(
161 std::max(1 + diff_squared.computeOrder(), 2 * p + 7)));
162
163 // Return the L2 error (square root of the integral)
164 return std::sqrt(std::max(0.0, l2_error_squared));
165}

185template <
186 typename MeshType,
187 typename FeSpaceType,
188 typename DofHandlerType,
189 typename ExprType>
190double computeH1SemiError(
191 const MeshType& mesh,
192 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
193 const DofHandlerType& dof_handler,
194 const la::Vector& u_h,
195 const ExprType& grad_u_exact_expr,
196 std::optional<uint32_t> quad_degree = std::nullopt) {
197 constexpr std::size_t Dim = MeshType::self_dim;
198
199 fe::FeGradProjectionExpr<Dim, FeSpaceType, DofHandlerType> grad_uh_expr(
200 u_h, cell_manager.getFeSpace(), dof_handler);
201
202 auto diff_expr = grad_uh_expr - grad_u_exact_expr;
203 auto diff_squared = fe::inner(diff_expr, diff_expr);
204
205 const std::size_t p = cell_manager.getFeSpace().order();
206 double h1_semi_error_squared = integrateFeExpression(
207 mesh, cell_manager, diff_squared,
208 quad_degree.value_or(
209 std::max(1 + diff_squared.computeOrder(), 2 * p + 7)));
210
211 return std::sqrt(std::max(0.0, h1_semi_error_squared));
212}

243template <std::size_t Dim>
244auto createPenaltyExpression(std::size_t order, double gamma) {
245 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
246 [order, gamma](
247 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
248 const auto& cell = *cells[fe_values.getCellId()];
249 const double penalty = gamma * order * order;
250 const double p_this = penalty / cell.diameter();
251 if (fe_values.hasFaceNeigh()) {
252 assert(&fe_values.neighValues());
253 const auto neigh_cell_id = fe_values.neighValues().getCellId();
254 const auto& neigh_cell = *cells[neigh_cell_id];
255 const double p_neigh = penalty / neigh_cell.diameter();
256 return std::max(p_this, p_neigh);
257 } else {
258 return p_this;
259 }
260 });
261}

273template <std::size_t Dim>
274auto createPenaltyExpressionKb(std::size_t order, double gamma) {
275 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
276 [order, gamma](
277 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
278 const auto& cell = *cells[fe_values.getCellId()];
279 const auto face_id = fe_values.getFaceId();
280 const double penalty = gamma * order * order;
281 const double Cinv = cell.measure() / cell.faceKb()[face_id];
282 const double facet_diam = cell.faceDiam()[face_id];
283 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
284 if (fe_values.hasFaceNeigh()) {
285 assert(&fe_values.neighValues());
286 const auto neigh_cell_id = fe_values.neighValues().getCellId();
287 const auto& neigh_cell = *cells[neigh_cell_id];
288 const auto neigh_face_id = fe_values.neighValues().getFaceId();
289 const double Cinv_neigh =
290 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
291 const double p_neigh =
292 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
293 return std::max(p_this, p_neigh);
294 } else {
295 return p_this;
296 }
297 });
298}

313template <std::size_t Dim>
314auto createPenaltyExpressionHarmonic(std::size_t order, double gamma) {
315 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
316 [order, gamma](
317 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
318 const auto& cell = *cells[fe_values.getCellId()];
319 const auto face_id = fe_values.getFaceId();
320 const double facet_diam = cell.faceDiam()(face_id);
321 const double penalty = gamma * order * order;
322 const double Cinv = cell.measure() / cell.faceKb()[face_id];
323 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
324
325 if (fe_values.hasFaceNeigh()) {
326 const auto neigh_cell_id = fe_values.neighValues().getCellId();
327 const auto& neigh_cell = *cells[neigh_cell_id];
328 const auto neigh_face_id = fe_values.neighValues().getFaceId();
329 const double Cinv_neigh =
330 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
331 const double p_neigh =
332 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
333 // Media armonica tra p_this e p_neigh
334 return 2.0 * p_this * p_neigh / (p_this + p_neigh);
335 } else {
336 return p_this;
337 }
338 });
339}

347template <std::size_t Dim, typename VelocityFunc>
348class FeExpressionUpwindFromFunctionWrapper
349 : public fe::FeExpressionBase<
350 Dim,
351 FeExpressionUpwindFromFunctionWrapper<Dim, VelocityFunc>> {
352 public:
353 static constexpr bool is_leaf = true;
354 static constexpr uint8_t is_bilinear = fe::IsNothing;
355
356 explicit constexpr FeExpressionUpwindFromFunctionWrapper(
357 const VelocityFunc& velocity_func)
358 : m_velocity_func(velocity_func) {}
359 FeExpressionUpwindFromFunctionWrapper(
360 const FeExpressionUpwindFromFunctionWrapper&) = delete;
361 FeExpressionUpwindFromFunctionWrapper& operator=(
362 const FeExpressionUpwindFromFunctionWrapper&) = delete;
363 FeExpressionUpwindFromFunctionWrapper(
364 FeExpressionUpwindFromFunctionWrapper&&) = delete;
365 FeExpressionUpwindFromFunctionWrapper& operator=(
366 FeExpressionUpwindFromFunctionWrapper&&) = delete;
367
368 template <std::size_t I = 0, std::size_t J = 0>
369 decltype(auto) evaluate(
370 const fe::FeValues<Dim>&, std::size_t, std::size_t, std::size_t q) const {
371 return m_evals[q];
372 }
373
374 constexpr fe::UpdateFlagsType computeUpdateFlags() const {
375 return fe::UpdateQuadraturePts | fe::UpdateNormals;
376 }
377 constexpr std::size_t computeOrder() const { return 0; }
378
379 constexpr bool precompute(
380 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>&) const {
381 const Eigen::Index nqpts = fe_values.weights().size();
382 assert(fe_values.qpts().cols() == nqpts);
383 m_evals.resize(nqpts);
384 const auto& normal = fe_values.getFaceNormal();
385 for (Eigen::Index q = 0; q < nqpts; ++q) {
386 const auto vel = m_velocity_func(fe_values.qpts().col(q));
387 const double vel_dot_n = vel.dot(normal);
388 m_evals[q] = 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
389 }
390 return true;
391 }
392
393 protected:
394 const VelocityFunc m_velocity_func;
395 mutable std::vector<double> m_evals;
396};

428template <std::size_t Dim>
429auto createUpwindFluxConstant(double vx, double vy = 0.0, double vz = 0.0) {
430 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
431 [vx, vy, vz](
432 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
433 const auto& cell = *cells[fe_values.getCellId()];
434 const auto face_id = fe_values.getFaceId();
435 const auto& normal = cell.normals().col(face_id);
436
437 // Compute velocity . n
438 double vel_dot_n = vx * normal(0);
439 if constexpr (Dim >= 2) {
440 vel_dot_n += vy * normal(1);
441 }
442 if constexpr (Dim >= 3) {
443 vel_dot_n += vz * normal(2);
444 }
445
446 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
447 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
448 });
449}

466template <std::size_t Dim, typename VelocityFunc>
467auto createUpwindFluxVariable(VelocityFunc velocity_func) {
468 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
469 [velocity_func](
470 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
471 const auto& cell = *cells[fe_values.getCellId()];
472 const auto face_id = fe_values.getFaceId();
473 const auto& normal = cell.normals().col(face_id);
474
475 // Face centroid: average of the two edge vertices for 2D polygons.
476 const Eigen::Index nvtxs = cell.vtxs().cols();
477 const Eigen::Vector<double, Dim> face_center =
478 0.5 *
479 (cell.vtxs().col(face_id) + cell.vtxs().col((face_id + 1) % nvtxs));
480
481 // Evaluate velocity at face center and compute vel . n
482 const auto vel = velocity_func(face_center);
483 const double vel_dot_n = vel.dot(normal);
484
485 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
486 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
487 });
488}

501template <typename MeshType>
502double computeError(
503 const dofs::DofHandler<MeshType>& dof_handler,
504 const la::Vector& u_h,
505 const la::Vector& u_exact,
506 const la::SparseMatrix& matrix) {
507 // Compute the residual: error_u = u_exact - u_h
508 la::Vector error_u(dof_handler, u_h.getBlockSize());
509 error_u.add(-1.0, u_h);
510 error_u.add(1.0, u_exact);
511 error_u.compress();
512
513 // Compute the error norm using the provided matrix
514 la::Vector temp(dof_handler, u_h.getBlockSize());
515 // std::cout << "Error Calc with matrix: " << std::endl;
516 matrix.vmult(error_u, temp); // temp = matrix * error_u
517
518 if (la::dot(error_u, temp) < 0) {
519 std::cout
520 << "Warning: Negative value in error computation. Setting error to 0."
521 << std::endl;
522 return 0.0;
523 } else {
524 return std::sqrt(la::dot(error_u, temp));
525 }
526}

558template <typename MeshType>
559la::Vector projectConstantOntoModalSpace(
560 const dofs::DofHandler<MeshType>& dof_handler,
561 double constant_value,
562 const la::SparseMatrix& mass_matrix,
563 const la::Vector& rhs_const) {
564 la::Vector scaled_rhs(rhs_const);
565 scaled_rhs.scale(constant_value);
566
567 la::Vector c_proj(dof_handler);
568 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
569 solver.setOperators(mass_matrix);
570 solver.solve(scaled_rhs, c_proj);
571
572 return c_proj;
573}

588template <typename MeshType>
589double forceEqualMean(
590 const dofs::DofHandler<MeshType>& dof_handler,
591 la::Vector& u,
592 const la::Vector& u_reference,
593 const la::SparseMatrix& mass_matrix,
594 const la::Vector& rhs_const,
595 double domain_measure) {
596 double avg_u = computeAverage(u, rhs_const, domain_measure);
597 double avg_reference = computeAverage(u_reference, rhs_const, domain_measure);
598
599 double shift_physical = avg_reference - avg_u;
600
601 la::Vector c_proj = projectConstantOntoModalSpace(
602 dof_handler, shift_physical, mass_matrix, rhs_const);
603
604 u.add(1.0, c_proj);
605 u.compress();
606
607 return shift_physical;
608}

618template <
619 typename MeshType,
620 typename CellManagerType,
621 typename DofHandlerType,
622 typename FeSpaceType,
623 typename Expression>
639la::Vector l2ProjectExpressionScalar(
640 const MeshType& mesh,
641 CellManagerType& cell_manager,
642 const DofHandlerType& dof_handler,
643 const FeSpaceType& fe_space,
644 const Expression& expr,
645 const la::SparseMatrix& mass_scalar) {
646 namespace fe = vulpes::fe;
647
648 fe::TestFunction v(fe_space);
649 auto wf_projection = expr * v;
650
651 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
652 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
653
654 la::Vector rhs(dof_handler);
655 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
656 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
657 int_smplx_projection.integrate(cell_id);
658 rhs.insertValues(
659 int_smplx_projection.getIntegral(&wf_projection),
660 dof_handler.getCellToDof()[cell_id]);
661 }
662 rhs.compress();
663
664 la::Vector projected(dof_handler);
665 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
666 solver.setOperators(mass_scalar);
667 solver.solve(rhs, projected);
668 return projected;
669}

708template <
709 std::size_t VecSize,
710 std::size_t NFields,
711 typename MeshType,
712 typename CellManagerType,
713 typename DofHandlerType,
714 typename FeSpaceType,
715 typename Expression>
737la::Vector l2ProjectExpressionBlockedSubset(
738 const MeshType& mesh,
739 CellManagerType& cell_manager,
740 const DofHandlerType& dof_handler,
741 const FeSpaceType& fe_space,
742 const Expression& expr,
743 std::size_t component_offset,
744 const la::SparseMatrix& mass_blocked) {
745 namespace fe = vulpes::fe;
746
747 assert(component_offset + VecSize <= NFields);
748
749 fe::TestFunction<FeSpaceType, 0, VecSize> v(fe_space);
750 auto wf_projection = fe::inner(expr, v);
751
752 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
753 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
754
755 la::Vector rhs(dof_handler, NFields);
756 const auto& cell_to_dof = dof_handler.getCellToDof();
757 const Eigen::Index ndof_comp = fe_space.getBasis().size();
758 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
759
760 std::array<PetscScalar, NFields> block_values{};
761 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
762 int_smplx_projection.integrate(cell_id);
763 const auto local = int_smplx_projection.getIntegral(&wf_projection);
764
765 for (Eigen::Index i = 0; i < ndof_comp; ++i) {
766 block_values.fill(0.0);
767 for (std::size_t c = 0; c < VecSize; ++c) {
768 block_values[component_offset + c] = local(
769 i * static_cast<Eigen::Index>(VecSize) +
770 static_cast<Eigen::Index>(c),
771 0);
772 }
773 const PetscInt block_index = cell_to_dof[cell_id][i];
774 rhs.insertValuesBlocked(block_values.data(), &block_index, 1, ADD_VALUES);
775 }
776 }
777 rhs.compress();
778
779 la::Vector projected(dof_handler, NFields);
780 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
781 solver.setOperators(mass_blocked);
782 solver.solve(rhs, projected);
783 return projected;
784}
