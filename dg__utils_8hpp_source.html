83template <typename MeshType, typename FeSpaceType, typename ExprType>
84double integrateFeExpression(
85 const MeshType& mesh,
86 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
87 const ExprType& expression) {
88 const uint32_t quadrature_order =
89 1 + static_cast<uint32_t>(expression.computeOrder());
90 assemble::ScalarExpressionIntegrator<MeshType, FeSpaceType, ExprType>
91 integrator(mesh, cell_manager, expression, quadrature_order);
92 double result = integrator.integrate();
93 if constexpr (MeshType::parallel()) {
94 MPI_Allreduce(
95 MPI_IN_PLACE, &result, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
96 }
97 return result;
98}

117template <
118 typename MeshType,
119 typename FeSpaceType,
120 typename DofHandlerType,
121 typename ExprType>
122double computeL2Error(
123 const MeshType& mesh,
124 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
125 const DofHandlerType& dof_handler,
126 const la::Vector& u_h,
127 const ExprType& u_exact_expr) {
128 constexpr std::size_t Dim = MeshType::self_dim;
129 // Create the numerical solution as an FE expression
130 fe::FeExactExpressionFeProjection<Dim, FeSpaceType, DofHandlerType> u_h_expr(
131 u_h, cell_manager.getFeSpace(), dof_handler);
132
133 // Form the difference expression: (u_h - u_exact)
134 auto diff_expr = u_h_expr - u_exact_expr;
135
136 // Form the squared difference: (u_h - u_exact)^2
137 auto diff_squared = diff_expr * diff_expr;
138
139 // Integrate the squared difference over the domain
140 double l2_error_squared =
141 integrateFeExpression(mesh, cell_manager, diff_squared);
142
143 // Return the L2 error (square root of the integral)
144 return std::sqrt(std::max(0.0, l2_error_squared));
145}

162template <
163 typename MeshType,
164 typename FeSpaceType,
165 typename DofHandlerType,
166 typename ExprType>
167double computeH1SemiError(
168 const MeshType& mesh,
169 assemble::CellManager<MeshType, FeSpaceType>& cell_manager,
170 const DofHandlerType& dof_handler,
171 const la::Vector& u_h,
172 const ExprType& grad_u_exact_expr) {
173 constexpr std::size_t Dim = MeshType::self_dim;
174
175 fe::FeGradProjectionExpr<Dim, FeSpaceType, DofHandlerType> grad_uh_expr(
176 u_h, cell_manager.getFeSpace(), dof_handler);
177
178 auto diff_expr = grad_uh_expr - grad_u_exact_expr;
179 auto diff_squared = fe::inner(diff_expr, diff_expr);
180
181 double h1_semi_error_squared =
182 integrateFeExpression(mesh, cell_manager, diff_squared);
183
184 return std::sqrt(std::max(0.0, h1_semi_error_squared));
185}

216template <std::size_t Dim>
217auto createPenaltyExpression(std::size_t order, double gamma) {
218 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
219 [order, gamma](
220 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
221 const auto& cell = *cells[fe_values.getCellId()];
222 const double penalty = gamma * order * order;
223 const double p_this = penalty / cell.diameter();
224 if (fe_values.hasFaceNeigh()) {
225 assert(&fe_values.neighValues());
226 const auto neigh_cell_id = fe_values.neighValues().getCellId();
227 const auto& neigh_cell = *cells[neigh_cell_id];
228 const double p_neigh = penalty / neigh_cell.diameter();
229 return std::max(p_this, p_neigh);
230 } else {
231 return p_this;
232 }
233 });
234}

246template <std::size_t Dim>
247auto createPenaltyExpressionKb(std::size_t order, double gamma) {
248 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
249 [order, gamma](
250 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
251 const auto& cell = *cells[fe_values.getCellId()];
252 const auto face_id = fe_values.getFaceId();
253 const double penalty = gamma * order * order;
254 const double Cinv = cell.measure() / cell.faceKb()[face_id];
255 const double facet_diam = cell.faceDiam()[face_id];
256 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
257 if (fe_values.hasFaceNeigh()) {
258 assert(&fe_values.neighValues());
259 const auto neigh_cell_id = fe_values.neighValues().getCellId();
260 const auto& neigh_cell = *cells[neigh_cell_id];
261 const auto neigh_face_id = fe_values.neighValues().getFaceId();
262 const double Cinv_neigh =
263 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
264 const double p_neigh =
265 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
266 return std::max(p_this, p_neigh);
267 } else {
268 return p_this;
269 }
270 });
271}

286template <std::size_t Dim>
287auto createPenaltyExpressionHarmonic(std::size_t order, double gamma) {
288 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
289 [order, gamma](
290 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
291 const auto& cell = *cells[fe_values.getCellId()];
292 const auto face_id = fe_values.getFaceId();
293 const double facet_diam = cell.faceDiam()(face_id);
294 const double penalty = gamma * order * order;
295 const double Cinv = cell.measure() / cell.faceKb()[face_id];
296 const double p_this = penalty * Cinv * (facet_diam / cell.measure());
297
298 if (fe_values.hasFaceNeigh()) {
299 const auto neigh_cell_id = fe_values.neighValues().getCellId();
300 const auto& neigh_cell = *cells[neigh_cell_id];
301 const auto neigh_face_id = fe_values.neighValues().getFaceId();
302 const double Cinv_neigh =
303 neigh_cell.measure() / neigh_cell.faceKb()[neigh_face_id];
304 const double p_neigh =
305 penalty * Cinv_neigh * (facet_diam / neigh_cell.measure());
306 // Media armonica tra p_this e p_neigh
307 return 2.0 * p_this * p_neigh / (p_this + p_neigh);
308 } else {
309 return p_this;
310 }
311 });
312}

320template <std::size_t Dim, typename VelocityFunc>
321class FeExpressionUpwindFromFunctionWrapper
322 : public fe::FeExpressionBase<
323 Dim,
324 FeExpressionUpwindFromFunctionWrapper<Dim, VelocityFunc>> {
325 public:
326 static constexpr bool is_leaf = true;
327 static constexpr uint8_t is_bilinear = fe::IsNothing;
328
329 explicit constexpr FeExpressionUpwindFromFunctionWrapper(
330 const VelocityFunc& velocity_func)
331 : m_velocity_func(velocity_func) {}
332 FeExpressionUpwindFromFunctionWrapper(
333 const FeExpressionUpwindFromFunctionWrapper&) = delete;
334 FeExpressionUpwindFromFunctionWrapper& operator=(
335 const FeExpressionUpwindFromFunctionWrapper&) = delete;
336 FeExpressionUpwindFromFunctionWrapper(
337 FeExpressionUpwindFromFunctionWrapper&&) = delete;
338 FeExpressionUpwindFromFunctionWrapper& operator=(
339 FeExpressionUpwindFromFunctionWrapper&&) = delete;

341 template <std::size_t I = 0, std::size_t J = 0>
342 decltype(auto) evaluate(
343 const fe::FeValues<Dim>&, std::size_t, std::size_t, std::size_t q) const {
344 return m_evals[q];
345 }
346
347 constexpr fe::UpdateFlagsType computeUpdateFlags() const {
348 return fe::UpdateQuadraturePts | fe::UpdateNormals;
349 }
350 constexpr std::size_t computeOrder() const { return 0; }
351
352 constexpr bool precompute(
353 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>&) const {
354 const Eigen::Index nqpts = fe_values.weights().size();
355 assert(fe_values.qpts().cols() == nqpts);
356 m_evals.resize(nqpts);
357 const auto& normal = fe_values.getFaceNormal();
358 for (Eigen::Index q = 0; q < nqpts; ++q) {
359 const auto vel = m_velocity_func(fe_values.qpts().col(q));
360 const double vel_dot_n = vel.dot(normal);
361 m_evals[q] = 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
362 }
363 return true;
364 }
365
366 protected:
367 const VelocityFunc m_velocity_func;
368 mutable std::vector<double> m_evals;
369};

401template <std::size_t Dim>
402auto createUpwindFluxConstant(double vx, double vy = 0.0, double vz = 0.0) {
403 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
404 [vx, vy, vz](
405 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
406 const auto& cell = *cells[fe_values.getCellId()];
407 const auto face_id = fe_values.getFaceId();
408 const auto& normal = cell.normals().col(face_id);
409
410 // Compute velocity . n
411 double vel_dot_n = vx * normal(0);
412 if constexpr (Dim >= 2) {
413 vel_dot_n += vy * normal(1);
414 }
415 if constexpr (Dim >= 3) {
416 vel_dot_n += vz * normal(2);
417 }
418
419 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
420 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
421 });
422}

439template <std::size_t Dim, typename VelocityFunc>
440auto createUpwindFluxVariable(VelocityFunc velocity_func) {
441 return fe::FeExpressionCellConstantNeedNeigh<Dim>(
442 [velocity_func](
443 const fe::FeValues<Dim>& fe_values, const fe::CellsType<Dim>& cells) {
444 const auto& cell = *cells[fe_values.getCellId()];
445 const auto face_id = fe_values.getFaceId();
446 const auto& normal = cell.normals().col(face_id);
447
448 // Face centroid: average of the two edge vertices for 2D polygons.
449 const Eigen::Index nvtxs = cell.vtxs().cols();
450 const Eigen::Vector<double, Dim> face_center =
451 0.5 *
452 (cell.vtxs().col(face_id) + cell.vtxs().col((face_id + 1) % nvtxs));
453
454 // Evaluate velocity at face center and compute vel . n
455 const auto vel = velocity_func(face_center);
456 const double vel_dot_n = vel.dot(normal);
457
458 // Upwind flux: 0.5 * (|velocity . n| + velocity . n)
459 return 0.5 * (std::abs(vel_dot_n) + vel_dot_n);
460 });
461}

474template <typename MeshType>
475double computeError(
476 const dofs::DofHandler<MeshType>& dof_handler,
477 const la::Vector& u_h,
478 const la::Vector& u_exact,
479 const la::SparseMatrix& matrix) {
480 // Compute the residual: error_u = u_exact - u_h
481 la::Vector error_u(dof_handler, u_h.getBlockSize());
482 error_u.add(-1.0, u_h);
483 error_u.add(1.0, u_exact);
484 error_u.compress();
485
486 // Compute the error norm using the provided matrix
487 la::Vector temp(dof_handler, u_h.getBlockSize());
488 // std::cout << "Error Calc with matrix: " << std::endl;
489 matrix.vmult(error_u, temp); // temp = matrix * error_u
490
491 if (la::dot(error_u, temp) < 0) {
492 std::cout
493 << "Warning: Negative value in error computation. Setting error to 0."
494 << std::endl;
495 return 0.0;
496 } else {
497 return std::sqrt(la::dot(error_u, temp));
498 }
499}

531template <typename MeshType>
532la::Vector projectConstantOntoModalSpace(
533 const dofs::DofHandler<MeshType>& dof_handler,
534 double constant_value,
535 const la::SparseMatrix& mass_matrix,
536 const la::Vector& rhs_const) {
537 la::Vector scaled_rhs(rhs_const);
538 scaled_rhs.scale(constant_value);
539
540 la::Vector c_proj(dof_handler);
541 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
542 solver.setOperators(mass_matrix);
543 solver.solve(scaled_rhs, c_proj);
544
545 return c_proj;
546}

561template <typename MeshType>
562double forceEqualMean(
563 const dofs::DofHandler<MeshType>& dof_handler,
564 la::Vector& u,
565 const la::Vector& u_reference,
566 const la::SparseMatrix& mass_matrix,
567 const la::Vector& rhs_const,
568 double domain_measure) {
569 double avg_u = computeAverage(u, rhs_const, domain_measure);
570 double avg_reference = computeAverage(u_reference, rhs_const, domain_measure);
571
572 double shift_physical = avg_reference - avg_u;
573
574 la::Vector c_proj = projectConstantOntoModalSpace(
575 dof_handler, shift_physical, mass_matrix, rhs_const);
576
577 u.add(1.0, c_proj);
578 u.compress();
579
580 return shift_physical;
581}

591template <
592 typename MeshType,
593 typename CellManagerType,
594 typename DofHandlerType,
595 typename FeSpaceType,
596 typename Expression>
612la::Vector l2ProjectExpressionScalar(
613 const MeshType& mesh,
614 CellManagerType& cell_manager,
615 const DofHandlerType& dof_handler,
616 const FeSpaceType& fe_space,
617 const Expression& expr,
618 const la::SparseMatrix& mass_scalar) {
619 namespace fe = vulpes::fe;
620
621 fe::TestFunction v(fe_space);
622 auto wf_projection = expr * v;
623
624 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
625 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
626
627 la::Vector rhs(dof_handler);
628 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
629 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
630 int_smplx_projection.integrate(cell_id);
631 rhs.insertValues(
632 int_smplx_projection.getIntegral(&wf_projection),
633 dof_handler.getCellToDof()[cell_id]);
634 }
635 rhs.compress();
636
637 la::Vector projected(dof_handler);
638 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
639 solver.setOperators(mass_scalar);
640 solver.solve(rhs, projected);
641 return projected;
642}

681template <
682 std::size_t VecSize,
683 std::size_t NFields,
684 typename MeshType,
685 typename CellManagerType,
686 typename DofHandlerType,
687 typename FeSpaceType,
688 typename Expression>
710la::Vector l2ProjectExpressionBlockedSubset(
711 const MeshType& mesh,
712 CellManagerType& cell_manager,
713 const DofHandlerType& dof_handler,
714 const FeSpaceType& fe_space,
715 const Expression& expr,
716 std::size_t component_offset,
717 const la::SparseMatrix& mass_blocked) {
718 namespace fe = vulpes::fe;
719
720 assert(component_offset + VecSize <= NFields);
721
722 fe::TestFunction<FeSpaceType, 0, VecSize> v(fe_space);
723 auto wf_projection = fe::inner(expr, v);
724
725 assemble::WeakFormIntegratorSimplexify int_smplx_projection(
726 mesh, cell_manager, std::tuple(&wf_projection), std::tuple());
727
728 la::Vector rhs(dof_handler, NFields);
729 const auto& cell_to_dof = dof_handler.getCellToDof();
730 const Eigen::Index ndof_comp = fe_space.getBasis().size();
731 const std::size_t n_cells = static_cast<std::size_t>(mesh.numCells());
732
733 std::array<PetscScalar, NFields> block_values{};
734 for (std::size_t cell_id = 0; cell_id < n_cells; ++cell_id) {
735 int_smplx_projection.integrate(cell_id);
736 const auto local = int_smplx_projection.getIntegral(&wf_projection);
737
738 for (Eigen::Index i = 0; i < ndof_comp; ++i) {
739 block_values.fill(0.0);
740 for (std::size_t c = 0; c < VecSize; ++c) {
741 block_values[component_offset + c] = local(
742 i * static_cast<Eigen::Index>(VecSize) +
743 static_cast<Eigen::Index>(c),
744 0);
745 }
746 const PetscInt block_index = cell_to_dof[cell_id][i];
747 rhs.insertValuesBlocked(block_values.data(), &block_index, 1, ADD_VALUES);
748 }
749 }
750 rhs.compress();
751
752 la::Vector projected(dof_handler, NFields);
753 la::LinearSolver solver(la::LinearSolver::PrecondOnly, la::LinearSolver::LU);
754 solver.setOperators(mass_blocked);
755 solver.solve(rhs, projected);
756 return projected;
757}
