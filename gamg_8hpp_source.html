97template <typename CellManagerType, typename DofHandlerType>
98Vector constantModalCoefficients(
99 CellManagerType& cell_manager,
100 const DofHandlerType& dof_handler,
101 double value = 1.0) {
102 const auto& mesh = cell_manager.getMesh();
103 const auto& fe_space = cell_manager.getFeSpace();
104 const auto& constant_mode = fe_space.getBasisFast()[0];
105 using Point = typename std::decay_t<decltype(mesh)>::SelfPoint;
106 const double mode_value = constant_mode.evaluate(Point::Zero());
107 assert(
108 std::abs(constant_mode.evaluate(Point::Ones()) - mode_value) <=
109 1e-12 * std::abs(mode_value) &&
110 "the first basis function must be the constant mode");
111
112 const auto n_cells = [&]() {
113 if constexpr (std::decay_t<decltype(mesh)>::parallel()) {
114 return mesh.nOwnedCells();
115 } else {
116 return mesh.numCells();
117 }
118 }();
119 Vector coeffs(dof_handler);
120 for (std::decay_t<decltype(n_cells)> cell_id = 0; cell_id < n_cells;
121 ++cell_id) {
122 cell_manager.getCell(cell_id); // builds the mapping if needed
123 // Same basis scaling as the integrators (IntegratorBase::
124 // applyBasisScaling): physical basis = reference basis / sqrt(det J)
125 double basis_value = mode_value;
126 if constexpr (std::decay_t<decltype(fe_space)>::needBasisScaling()) {
127 basis_value /=
128 std::sqrt(cell_manager.getMapping(cell_id)->jacobianDet({}));
129 }
130 coeffs.setValue(
131 dof_handler.getCellToDof()[cell_id][0], value / basis_value);
132 }
133 coeffs.compress();
134 return coeffs;
135}
