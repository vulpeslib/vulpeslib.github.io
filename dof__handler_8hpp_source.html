528 typename FeSpace>
529void DofHandler<MeshType>::buildCellToCellDof(
530 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
531 // copy the sparsity pattern from the one of `m_cell_to_dof`
532 m_cell_to_cell_dof = m_cell_to_dof;
533 // cell_id is the local id in MPI numeration
534 for (index_t cell_id = 0; cell_id < m_cell_to_cell_dof.size(); ++cell_id) {
535 for (auto& dof_id : m_cell_to_cell_dof[cell_id]) {
536 // expand information of dof_id by expoiting global (geometric)
537 // enumeration
538 index_t global_dof_id = dof_id;
539 if constexpr (MeshType::parallel()) {
540 global_dof_id = m_local_to_global_dof[dof_id];
541 }
542 index_t enty_dim = 0;
543 while (!(global_dof_id < m_glob_cum_ndof_per_entity[enty_dim + 1])) {
544 enty_dim++;
545 }
546 const index_t entity_dof_id =
547 global_dof_id - m_glob_cum_ndof_per_entity[enty_dim];
548 const index_t n_dofs_per_entity =
549 fe_space.getNumDofsForSubdimEntity(enty_dim);
550 const index_t owning_enty_id = entity_dof_id / n_dofs_per_entity;
551 const index_t enty_multiplicity = entity_dof_id % n_dofs_per_entity;
552 // pass the expanded information to the fe space and update dof_id
553 // from global to local numbering (cell based, not MPI partition based)
554 dof_id = fe_space.dofToCellDof(
555 m_mesh, cell_id, owning_enty_id, enty_dim, enty_multiplicity);
556 }
557 }

562 std::size_t Dim,
563 typename FloatType,
564 typename DofPolicy,
565 typename FeSpace>
566void DofHandler<MeshType>::computeNumDofs(
567 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
568 for (std::size_t d = 0; d < FeSpace::dim + 1; ++d) {
569 const auto n_dofs_per_entity = fe_space.getNumDofsForSubdimEntity(d);
570 const auto n_glob_entities = m_mesh.nEntity(d);
571 auto n_loc_entities = m_mesh.nLocEntity(d);
572 if constexpr (MeshType::parallel()) {
573 n_loc_entities += m_mesh.nGhostEntity(d);
574 }
575 m_glob_cum_ndof_per_entity[d + 1] =
576 m_glob_cum_ndof_per_entity[d] + n_dofs_per_entity * n_glob_entities;
577 m_loc_cum_ndof_per_entity[d + 1] =
578 m_loc_cum_ndof_per_entity[d] + n_dofs_per_entity * n_loc_entities;
579 }

582template <typename MeshType>
583template <
584 std::size_t SubentityDim,
585 std::size_t Dim,
586 typename FloatType,
587 typename DofPolicy,
588 typename FeSpace>
589void DofHandler<MeshType>::distributeSubentityDofs(
590 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space,
591 auto& dof_to_cell) {
592 const index_t n_dofs_per_entity =
593 fe_space.getNumDofsForSubdimEntity(SubentityDim);
594 if (n_dofs_per_entity > 0) {
595 index_t n_loc_entity = m_mesh.nLocEntity(SubentityDim);
596 if constexpr (MeshType::parallel()) {
597 n_loc_entity += m_mesh.nGhostEntity(SubentityDim);
598 }
599 // for each entity in the mesh, may be vertex, edge, facet, cell
600 for (index_t enty_id = 0; enty_id < n_loc_entity; ++enty_id) {
601 // for each dof that belongs to this entity
602 for (index_t j = 0; j < n_dofs_per_entity; ++j) {
603 // compute global dof id
604 index_t global_enty_id = enty_id;
605 if constexpr (MeshType::parallel()) {
606 global_enty_id = m_mesh.localToGlobal(SubentityDim)[enty_id];
607 }
608 const auto glob_dof_id = m_glob_cum_ndof_per_entity[SubentityDim] +
609 global_enty_id * n_dofs_per_entity + j;
610 auto loc_dof_id = glob_dof_id;
611 if constexpr (MeshType::parallel()) {
612 loc_dof_id = m_local_to_global_dof.size();
613 m_local_to_global_dof.push_back(glob_dof_id);
614 m_global_to_local_dof[glob_dof_id] = loc_dof_id;
615 }
616 // put the `dof_id` in the `dof_to_cell map`
617 // different behavior if we are in a cell or not
618 if constexpr (SubentityDim == Dim) {
619 // we are in a cell, we know to which cell this dof belongs to
620 dof_to_cell[loc_dof_id].push_back(global_enty_id);
621 // in parallel we must keep track of owned and shared dofs
622 if constexpr (MeshType::parallel()) {
623 // we own the cell dof if we own the cell
624 if (enty_id < m_mesh.nLocEntity(SubentityDim)) {
625 m_owned_dofs.push_back(glob_dof_id);
626 m_local_owned_dofs.push_back(loc_dof_id);
627 } else {
628 m_shared_dofs.push_back(glob_dof_id);
629 m_local_shared_dofs.push_back(loc_dof_id);
630 }
631 }
632 } else {
633 const auto& enty_to_cell_map =
634 m_mesh.template getEntityToCellGlobalMap<SubentityDim>();
635 // loop over all the cells to which this entity belongs to
636 for (const auto cell_id : enty_to_cell_map[enty_id]) {
637 // facet mappings always have two elements, one of which may be -1
638 // in this case we skip it
639 if constexpr (SubentityDim == Dim - 1) {
640 if (cell_id == static_cast<decltype(cell_id)>(-1)) {
641 continue;
642 }
643 }
644 dof_to_cell[loc_dof_id].push_back(cell_id);
645 // in parallel we must keep track of owned and shared dofs
646 if constexpr (MeshType::parallel()) {
647 // if the dof belongs to an entity that is on a ghost cell
648 // than it is a shared dof
649 const auto local_cell_id = m_mesh.globalToLocal(Dim).at(cell_id);
650 if ((local_cell_id >= m_mesh.nLocEntity(Dim)) &&
651 (m_shared_dofs.empty() ||
652 m_shared_dofs.back() != glob_dof_id)) {
653 m_shared_dofs.push_back(glob_dof_id);
654 m_local_shared_dofs.push_back(loc_dof_id);
655 }
656 }
657 }
658 // if the dof is not shared, than it is owned
659 if constexpr (MeshType::parallel()) {
660 if (!(m_shared_dofs.size() &&
661 (m_shared_dofs.back() == glob_dof_id))) {
662 m_owned_dofs.push_back(glob_dof_id);
663 m_local_owned_dofs.push_back(loc_dof_id);
664 }
665 }
666 }
667 }
668 }
669 }

693template <typename MeshType>
694void DofHandler<MeshType>::sortSharedDofs() {
695 std::vector<index_t> idxs(m_shared_dofs.size());
696 std::iota(idxs.begin(), idxs.end(), 0);
697 std::sort(idxs.begin(), idxs.end(), [this](auto a, auto b) {
698 return m_shared_dofs[a] < m_shared_dofs[b];
699 });
700 // apply sorting permutation
701 std::vector<index_t> tmp(m_shared_dofs.size());
702 for (std::size_t i = 0; i < idxs.size(); ++i) {
703 tmp[i] = m_shared_dofs[idxs[i]];
704 }
705 std::swap(tmp, m_shared_dofs);
706 for (std::size_t i = 0; i < idxs.size(); ++i) {
707 tmp[i] = m_local_shared_dofs[idxs[i]];
708 }
709 std::swap(tmp, m_local_shared_dofs);
710}

712template <typename MeshType>
713void DofHandler<MeshType>::invertDofToCellMappingAndCompress(
714 auto& dof_to_cell) {
715 constexpr std::size_t static_buffer_size =
716 vmath::pow((index_t)7, MeshType::self_dim);
717 datastructure::CrsMapUncompressed<index_t, static_buffer_size> cell_to_dof;
718 auto n_loc_cell = m_mesh.nLocEntity(MeshType::self_dim);
719 if constexpr (MeshType::parallel()) {
720 n_loc_cell += m_mesh.nGhostEntity(MeshType::self_dim);
721 }
722 cell_to_dof.resize(n_loc_cell);
723 for (std::size_t dof_id = 0; dof_id < dof_to_cell.size(); ++dof_id) {
724 // we exploit the fact that the dofs are sorted later
725 // when we compute the sparsity pattern
726 // TODO: is this already sorted if in serial?
727 std::sort(dof_to_cell[dof_id].begin(), dof_to_cell[dof_id].end());
728 for (const auto global_cell_id : dof_to_cell[dof_id]) {
729 auto loc_cell_id = global_cell_id;
730 if constexpr (MeshType::parallel()) {
731 loc_cell_id =
732 m_mesh.globalToLocal(MeshType::self_dim).at(global_cell_id);
733 }
734 cell_to_dof[loc_cell_id].push_back(dof_id);
735 }
736 }
737
738 m_dof_to_cell = dof_to_cell.template compress<index_t>();
739 m_cell_to_dof = cell_to_dof.template compress<index_t>();
740}

742template <typename MeshType>
743void DofHandler<MeshType>::computeOwnedCellDofsAndBndDofs() {
744 const auto max_dof_idx =
745 m_cell_to_dof.indices()[m_mesh.nLocEntity(MeshType::self_dim)];
746 const auto& data = m_cell_to_dof.data();
747 boost::unordered_flat_set<index_t> owned_cells_dof(
748 data.begin(), data.begin() + max_dof_idx);
749 m_owned_cells_dofs.assign(owned_cells_dof.begin(), owned_cells_dof.end());
750
751 // find which dofs lie on the partition
752 // i.e. they are both shared and belong to owned cells
753 for (auto loc_dof_id : m_owned_cells_dofs) {
754 const auto glob_dof_id = m_local_to_global_dof[loc_dof_id];
755 const auto it = std::lower_bound(
756 m_shared_dofs.begin(), m_shared_dofs.end(), glob_dof_id);
757 if ((it != m_shared_dofs.end()) && (*it == glob_dof_id)) {
758 m_partition_bnd_dofs.push_back(glob_dof_id);
759 }
760 }
761 std::sort(m_partition_bnd_dofs.begin(), m_partition_bnd_dofs.end());
762}

764template <typename MeshType>
765template <
766 std::size_t Dim,
767 typename FloatType,
768 typename DofPolicy,
769 typename FeSpace>
770void DofHandler<MeshType>::distributeDofs(
771 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
772 // smallest row size that make sense
773 // namely we fill all the space we have in the `SmallVector`
774 constexpr std::size_t static_row_size =
775 sizeof(std::vector<index_t>) / sizeof(index_t);
776 datastructure::CrsMapUncompressed<index_t, static_row_size> dof_to_cell;
777 static_assert(FeSpace::dim == MeshType::self_dim);
778
779 computeNumDofs(fe_space);
780 dof_to_cell.resize(m_loc_cum_ndof_per_entity[FeSpace::dim + 1]);
781 PROFILE_INTERVAL(distributeAllSubentityDofs(fe_space, dof_to_cell));
782
783 if (MeshType::parallel()) {
784 // we sort them because it is easier for rank 0 to use them
785 // when computing the global renumbering
786 PROFILE_INTERVAL(sortSharedDofs());
787 }
788
789 PROFILE_INTERVAL(invertDofToCellMappingAndCompress(dof_to_cell));
790
791 PROFILE_INTERVAL(buildCellToCellDof(fe_space));
792
793 if constexpr (MeshType::parallel()) {
794 PROFILE_INTERVAL(computeOwnedCellDofsAndBndDofs());
795 }
796
797 PROFILE_INTERVAL(computeSparsityPattern<FeSpace>());
798
799 if constexpr (MeshType::parallel()) {
800 m_all_dof_offset.resize(mpi::size() + 1, 0);
801 const auto total_dofs = m_glob_cum_ndof_per_entity.back();
802 for (int p = 0; p < mpi::size(); ++p) {
803 const auto reminder = (p < (total_dofs % mpi::size())) ? 1 : 0;
804 m_all_dof_offset[p + 1] =
805 m_all_dof_offset[p] + total_dofs / mpi::size() + reminder;
806 }
807 }
808}

810template <typename MeshType>
811template <bool DofInteractionDistance>
812std::pair<typename MeshType::index_t, typename MeshType::index_t>
813DofHandler<MeshType>::computeSparsityForDof(
814 typename MeshType::index_t dof_id,
815 typename MeshType::index_t* buffer1,
816 typename MeshType::index_t* buffer2) {
817 static std::array<std::size_t, 2> buffers_size = {0, 0};
818 static std::size_t active_buffer = 0;
819 buffers_size = {0, 0};
820 active_buffer = 0;
821 std::array<index_t*, 2> buffers = {buffer1, buffer2};
822
823 // merge all the dofs that belong to this cell (`dofs_view`) and the
824 // others (`buffer[active_buffer]`) into the inactive buffer (`buffer[1 -
825 // active_buffer]`). We assume the dofs are sorted and we remove
826 // duplicates
827 const auto updateBuffer = [&](auto dofs_view) {
828 const index_t* f1 = dofs_view.data();
829 const index_t* l1 = f1 + dofs_view.size();
830 const index_t* f2 = buffers[active_buffer];
831 const index_t* l2 = f2 + buffers_size[active_buffer];
832 index_t* o = buffers[1 - active_buffer];
833 // WARNING: this is not thread-safe
834 buffers_size[1 - active_buffer] =
835 std::distance(o, std::set_union(f1, l1, f2, l2, o));
836 // at each iteration we swap the source buffer with all the dofs and
837 // the destination buffer where we make the union, to avoid an extra
838 // copy
839 buffers_size[active_buffer] = 0;
840 active_buffer = 1 - active_buffer;
841 };
842
843 // for each cell to which this dof belongs to
844 for (index_t global_cell_id : m_dof_to_cell[dof_id]) {
845 auto local_cell_id = global_cell_id;
846 if constexpr (MeshType::parallel()) {
847 local_cell_id =
848 m_mesh.globalToLocal(MeshType::self_dim).at(global_cell_id);
849 }
850 updateBuffer(m_cell_to_dof[local_cell_id]);
851 // if there is interaction with also the neighbours
852 // e.g. if we are in DG each dof belongs to one cell but there
853 // are face couplings
854 if constexpr (DofInteractionDistance) {
855 for (index_t loc_neigh_cell_id : m_mesh.getDualMap()[local_cell_id]) {
856 updateBuffer(m_cell_to_dof[loc_neigh_cell_id]);
857 }
858 }
859 }
860
861 return {active_buffer, buffers_size[active_buffer]};
862}

864template <typename MeshType>
865template <typename FeSpace>
866void DofHandler<MeshType>::computeSparsityPattern() {
867 static constexpr std::size_t n_max_dofs_per_row = 1048576;
868 static std::array<std::array<index_t, n_max_dofs_per_row>, 2> buffers;
869
870 // Probe a good static size for the uncompressed map.
871 // We compute the sparsity pattern for some random dof ids
872 // and check which is the number of nnz at that row.
873 // We then ask for a static size that is greater than the average
874 constexpr index_t probe_size = 1000;
875 index_t nnz = 16;
876 if (m_dof_to_cell.size() > probe_size) {
877 nnz = 0;
878 std::mt19937 gen(0);
879 std::uniform_int_distribution<uint64_t> dis(0, m_dof_to_cell.size() - 1);
880 for (index_t i = 0; i < probe_size; ++i) {
881 const auto random_dof_id = dis(gen);
882 const auto [active_buffer, buffer_size] =
883 computeSparsityForDof<FeSpace::dofsInteractionDistance()>(
884 random_dof_id, buffers[0].data(), buffers[1].data());
885 nnz += buffer_size;
886 }
887 nnz /= probe_size;
888 }
889 std::cout << "Sparsity pattern probing suggested size: " << nnz << std::endl;
890 auto uncompressed_sparsity_pattern =
891 datastructure::getUncompressedCrsMapWithStaticStorageGreaterThan<index_t>(
892 nnz);
893 std::visit(
894 [&](auto& map) { map.resize(m_dof_to_cell.size()); },
895 uncompressed_sparsity_pattern);
896
897 // compute the sparsity pattern for all the dofs
898 for (index_t dof_id = 0; dof_id < m_dof_to_cell.size(); ++dof_id) {
899 const auto [active_buffer, buffer_size] =
900 computeSparsityForDof<FeSpace::dofsInteractionDistance()>(
901 dof_id, buffers[0].data(), buffers[1].data());
902 const auto nnz_dofid = buffer_size;
903 // copy the sparsity pattern from the static buffer to the uncompressed map
904 std::visit(
905 [dof_id, nnz_dofid, active_buffer](auto& map) {
906 map[dof_id].assign(
907 buffers[active_buffer].data(),
908 buffers[active_buffer].data() + nnz_dofid);
909 },
910 uncompressed_sparsity_pattern);
911 }
912
913 // compress the sparsity pattern
914 m_sparsity_pattern = std::visit(
915 [](const auto& map) { return map.template compress<index_t>(); },
916 uncompressed_sparsity_pattern);
917}

923 const auto size = mpi::size();
924 // collect the size of owned dofs and the list of shared dofs

926 m_all_shared_dof.resize(size);
927 m_all_partition_bnd_dofs.resize(size);
928

930 cum_dof_offset[1] = m_owned_dofs.size();

932 m_all_partition_bnd_dofs[0] = m_partition_bnd_dofs;

934 const auto owned_dof_p = mpi::recv<std::size_t>(p);
935 cum_dof_offset[p + 1] = cum_dof_offset[p] + owned_dof_p;
936 mpi::recv(m_all_shared_dof[p], p);
937 mpi::recv(m_all_partition_bnd_dofs[p], p);
938 }
939 } else {
940 mpi::send(m_owned_dofs.size(), 0);
941 mpi::send(m_shared_dofs, 0);
942 mpi::send(m_partition_bnd_dofs, 0);
943 }
944 // root renumbers the shared dofs and sends them
945 std::vector<index_t> shared_renumbering;
946 m_dof_offset = 0;
947 if (mpi::root()) {
948 // all the shared dofs without duplicates
949 std::vector<index_t> unq_shared_dof;
950 // list of boundary shared dofs "belonging" to proc p
951 // that is a bnd dof is owned by the proc with smallest rank that shares it
952 std::vector<index_t> bnd_dof_belonging_to_p;
953 // global-global old to new numbering
954 m_all_old_to_new_numbering.resize(m_glob_cum_ndof_per_entity.back(), -1);
955 std::vector<index_t> tmp_buffer;
956
957 for (int p = 0; p < size - 1; ++p) {
958 // to be numbered, they belong to p
959 std::set_difference(
960 m_all_partition_bnd_dofs[p].begin(),
961 m_all_partition_bnd_dofs[p].end(), unq_shared_dof.begin(),
962 unq_shared_dof.end(), std::back_inserter(bnd_dof_belonging_to_p));
963 // compute the dof offset taking into account the shared between 0 and p
964 for (int i = p; i < size; ++i) {
965 cum_dof_offset[i + 1] += bnd_dof_belonging_to_p.size();
966 }
967 mpi::send(cum_dof_offset[p + 1], p + 1);
968 const auto offset_without_shared =
969 cum_dof_offset[p + 1] - bnd_dof_belonging_to_p.size();
970 for (std::size_t i = 0; i < bnd_dof_belonging_to_p.size(); ++i) {
971 m_all_old_to_new_numbering[bnd_dof_belonging_to_p[i]] =
972 offset_without_shared + i;
973 }
974 // update unq_shared_dof
975 std::set_union(
976 unq_shared_dof.begin(), unq_shared_dof.end(),
977 m_all_partition_bnd_dofs[p].begin(),
978 m_all_partition_bnd_dofs[p].end(), std::back_inserter(tmp_buffer));
979 std::swap(unq_shared_dof, tmp_buffer);
980 bnd_dof_belonging_to_p.resize(0);
981 tmp_buffer.resize(0);
982 }
983
984 // renumber owning according to offset in sequence for self
985 m_old_to_new_numbering.resize(m_local_to_global_dof.size());
986 for (std::size_t i = 0; i < m_owned_dofs.size(); ++i) {
987 m_old_to_new_numbering[m_global_to_local_dof.at(m_owned_dofs[i])] = i;
988 m_all_old_to_new_numbering[m_owned_dofs[i]] = i;
989 }
990
991 // recive enumeration from all
992 std::vector<index_t> tmp_own_global_old;
993 std::vector<index_t> tmp_own_global_new;
994 for (int p = 1; p < size; ++p) {
995 mpi::recv(tmp_own_global_old, p);
996 mpi::recv(tmp_own_global_new, p);
997 for (std::size_t i = 0; i < tmp_own_global_old.size(); ++i) {
998 m_all_old_to_new_numbering[tmp_own_global_old[i]] =
999 tmp_own_global_new[i];
1000 }
1001 }
1002
1003 // find the enumeration for each proc
1004 for (int p = 1; p < size; ++p) {
1005 shared_renumbering.resize(m_all_shared_dof[p].size());
1006 for (std::size_t i = 0; i < m_all_shared_dof[p].size(); ++i) {
1007 shared_renumbering[i] =
1008 m_all_old_to_new_numbering[m_all_shared_dof[p][i]];
1009 }
1010 mpi::send(shared_renumbering, p);
1011 }
1012 // find the enumeration for self
1013 shared_renumbering.resize(m_all_shared_dof[0].size());
1014 for (std::size_t i = 0; i < m_all_shared_dof[0].size(); ++i) {
1015 shared_renumbering[i] =
1016 m_all_old_to_new_numbering[m_all_shared_dof[0][i]];
1017 }
1018 } else {
1019 m_dof_offset = mpi::recv<std::size_t>(0);
1020 // renumber according to offset in sequence
1021 std::vector<index_t> own_global_old;
1022 std::vector<index_t> own_global_new;
1023 m_old_to_new_numbering.resize(m_local_to_global_dof.size());
1024 for (std::size_t i = 0; i < m_owned_dofs.size(); ++i) {
1025 const auto glob_own_id = m_owned_dofs[i];
1026 const auto loc_own_id = m_global_to_local_dof.at(glob_own_id);
1027 const index_t new_glob_id = i + m_dof_offset;
1028 m_old_to_new_numbering[loc_own_id] = new_glob_id;
1029 own_global_old.push_back(glob_own_id);
1030 own_global_new.push_back(new_glob_id);
1031 }
1032 mpi::send(own_global_old, 0);
1033 mpi::send(own_global_new, 0);
1034 mpi::recv(shared_renumbering, 0);
1035 }
1036 // broadcast the offset
1037 m_all_dof_offset.resize(mpi::size() + 1);
1038 m_all_dof_offset.back() = m_glob_cum_ndof_per_entity.back();
1039 for (int p = 0; p < mpi::size(); ++p) {
1040 m_all_dof_offset[p] = m_dof_offset;
1041 mpi::bcast(m_all_dof_offset[p], p);
1042 }
1043 // apply the shared dof renumbering
1044 assert(shared_renumbering.size() == m_shared_dofs.size());
1045 for (std::size_t i = 0; i < m_shared_dofs.size(); ++i) {
1046 m_old_to_new_numbering[m_local_shared_dofs[i]] = shared_renumbering[i];
1047 }
1048 // invert mapping
1049 m_new_to_old_numbering.clear();
1050 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1051 m_new_to_old_numbering[m_old_to_new_numbering[i]] = i;
1052 }

1055template <typename MeshType>
1056void DofHandler<MeshType>::applyRenumbering(
1057 const RenumberingFunctor<typename MeshType::index_t>& renumbering_func,
1058 bool reversed) {
1059 m_has_renumbering = true;
1060 assert(m_old_to_new_numbering.size() == m_new_to_old_numbering.size());
1061 using SparsityPatternType = decltype(m_sparsity_pattern);
1062 SparsityPatternType* sparsity_pattern_ptr = &m_sparsity_pattern;
1063 std::shared_ptr<SparsityPatternType> owned_sparsity_pattern;
1064 if (MeshType::parallel()) {
1065 owned_sparsity_pattern =
1066 std::make_shared<SparsityPatternType>(computeOwnedSparsityPattern());
1067 sparsity_pattern_ptr = owned_sparsity_pattern.get();
1068 }
1069 const auto [perm, perm_inv] =
1070 renumbering_func(*sparsity_pattern_ptr, reversed);
1071 assert(perm.size() == perm_inv.size());
1072 // apply perm and perm_inv to mappings
1073 if constexpr (MeshType::parallel()) {
1074 const auto size = mpi::size();
1075 // update the old to new for owned dofs
1076 std::vector<index_t> new_owned_dof_id(m_owned_dofs.size());
1077 {
1078 std::vector<index_t> tmp = m_old_to_new_numbering;
1079 for (std::size_t i = 0; i < perm.size(); ++i) {
1080 tmp[m_local_owned_dofs[i]] =
1081 m_old_to_new_numbering[m_local_owned_dofs[perm[i]]];
1082 new_owned_dof_id[i] =
1083 m_old_to_new_numbering[m_local_owned_dofs[perm[i]]];
1084 }
1085 m_old_to_new_numbering = std::move(tmp);
1086 }
1087
1088 std::vector<index_t> new_shared_numbering;
1089 if (mpi::root()) {
1090 // get the new numbering of the dofs owned by each proc
1091 std::vector<index_t> owned_dofs_p = m_owned_dofs;
1092 for (int p = 0; p < size; ++p) {
1093 if (p > 0) {
1094 mpi::recv(owned_dofs_p, p);
1095 mpi::recv(new_owned_dof_id, p);
1096 }
1097 assert(new_owned_dof_id.size() == owned_dofs_p.size());
1098 for (std::size_t i = 0; i < new_owned_dof_id.size(); ++i) {
1099 m_all_old_to_new_numbering[owned_dofs_p[i]] = new_owned_dof_id[i];
1100 }
1101 }
1102
1103 // send new numbering of shared
1104 for (int p = 1; p < size; ++p) {
1105 new_shared_numbering.resize(m_all_shared_dof[p].size());
1106 for (std::size_t i = 0; i < m_all_shared_dof[p].size(); ++i) {
1107 new_shared_numbering[i] =
1108 m_all_old_to_new_numbering[m_all_shared_dof[p][i]];
1109 }
1110 mpi::send(new_shared_numbering, p);
1111 }
1112 // do the same for self
1113 new_shared_numbering.resize(m_shared_dofs.size());
1114 for (std::size_t i = 0; i < m_shared_dofs.size(); ++i) {
1115 new_shared_numbering[i] = m_all_old_to_new_numbering[m_shared_dofs[i]];
1116 }
1117 } else {
1118 mpi::send(m_owned_dofs, 0);
1119 // new glob ids for m_owned_dofs
1120 mpi::send(new_owned_dof_id, 0);
1121 // new enumeration for shared
1122 mpi::recv(new_shared_numbering, 0);
1123 }
1124 // apply new enumeration for shared
1125 assert(m_local_shared_dofs.size() == new_shared_numbering.size());
1126 for (std::size_t i = 0; i < m_local_shared_dofs.size(); ++i) {
1127 m_old_to_new_numbering[m_local_shared_dofs[i]] = new_shared_numbering[i];
1128 }
1129 // invert the old_to_new
1130 m_new_to_old_numbering.clear();
1131 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1132 m_new_to_old_numbering[m_old_to_new_numbering[i]] = i;
1133 }
1134 } else {
1135 if (m_old_to_new_numbering.size() != perm.size()) {
1136 m_old_to_new_numbering = std::move(perm);
1137 m_new_to_old_numbering = std::move(perm_inv);
1138 } else {
1139 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1140 m_old_to_new_numbering[i] = perm[m_old_to_new_numbering[i]];
1141 m_new_to_old_numbering[i] = perm_inv[m_new_to_old_numbering[i]];
1142 }
1143 }
1144 }

1147template <typename MeshType>
1148auto DofHandler<MeshType>::computeOwnedSparsityPattern() {
1149 if (MeshType::parallel()) {
1150 // build inverse map (from the local id of an owned dof to the owned id)
1151 const auto n_local_dofs = m_owned_dofs.size() + m_shared_dofs.size();
1152 std::vector<index_t> loc_owned_dofs_inv(n_local_dofs, -1);
1153 for (std::size_t i = 0; i < m_local_owned_dofs.size(); ++i) {
1154 loc_owned_dofs_inv[m_local_owned_dofs[i]] = i;
1155 }
1156 //
1157 std::decay_t<decltype(m_sparsity_pattern)> owned_map({}, {});
1158 const auto& indices = m_sparsity_pattern.indices();
1159 const auto& data = m_sparsity_pattern.data();
1160 owned_map.indices().push_back(0);
1161 for (const auto own_loc_dof_id : m_local_owned_dofs) {
1162 const auto row_start = indices[own_loc_dof_id];
1163 const index_t row_size = indices[own_loc_dof_id + 1] - row_start;
1164 index_t real_size = 0;
1165 for (index_t j = 0; j < row_size; ++j) {
1166 const index_t new_glob_id = data[row_start + j];
1167 const index_t loc_id = m_new_to_old_numbering[new_glob_id];
1168 const index_t old_glob_id = m_local_to_global_dof[loc_id];
1169 const auto it = std::lower_bound(
1170 m_shared_dofs.begin(), m_shared_dofs.end(), old_glob_id);
1171 // if it is not shared, than it is owned
1172 // shared are many less, thus it is quicker than to search in owned
1173 if ((it == m_shared_dofs.end()) || (*it != old_glob_id)) {
1174 owned_map.data().push_back(loc_owned_dofs_inv[loc_id]);
1175 real_size++;
1176 }
1177 }
1178 owned_map.indices().push_back(real_size + owned_map.indices().back());
1179 }
1180 return owned_map;
1181 } else {
1182 return m_sparsity_pattern;
1183 }

1186template <typename MeshType>
1187void DofHandler<MeshType>::finalize() {
1188 if constexpr (MeshType::parallel()) {
1189 // the parallel renumbering maintains the local numeration of rows
1190 if (m_has_renumbering) {
1191 for (auto& col : m_sparsity_pattern.data()) {
1192 col = m_old_to_new_numbering[col];
1193 }
1194 } else {
1195 for (auto& col : m_sparsity_pattern.data()) {
1196 col = m_local_to_global_dof[col];
1197 }
1198 }
1199 } else {
1200 if (m_has_renumbering) {
1201 m_sparsity_pattern = m_sparsity_pattern.permute(
1202 m_old_to_new_numbering, m_new_to_old_numbering);
1203 }
1204 }
1205
1206 // apply the cell-to-celldof mapping (to account for basis function numbering
1207 // in non-discontinous elements)
1208 assert(m_cell_to_cell_dof.size() == m_cell_to_dof.size());
1209 std::vector<index_t> tmp_buffer;
1210 for (index_t cell_id = 0; cell_id < m_cell_to_dof.size(); ++cell_id) {
1211 assert(m_cell_to_dof[cell_id].size() == m_cell_to_cell_dof[cell_id].size());
1212 const auto row_size = m_cell_to_dof[cell_id].size();
1213 if (tmp_buffer.size() < static_cast<std::size_t>(row_size)) {
1214 tmp_buffer.resize(row_size);
1215 }
1216 for (index_t i = 0; i < row_size; ++i) {
1217 tmp_buffer[m_cell_to_cell_dof[cell_id][i]] = m_cell_to_dof[cell_id][i];
1218 }
1219 std::copy(
1220 tmp_buffer.begin(), tmp_buffer.begin() + row_size,
1221 m_cell_to_dof[cell_id].begin());
1222 }
1223
1224 // apply the local to global to cell-to-dof
1225 if (m_has_renumbering) {
1226 for (auto& loc_dof : m_cell_to_dof.data()) {
1227 loc_dof = m_old_to_new_numbering[loc_dof];
1228 }
1229 } else {
1230 if constexpr (MeshType::parallel()) {
1231 for (auto& loc_dof : m_cell_to_dof.data()) {
1232 loc_dof = m_local_to_global_dof[loc_dof];
1233 }
1234 }
1235 }
1236
1237 buildGhostDofs();

1240template <typename MeshType>
1241void DofHandler<MeshType>::buildGhostDofs() {
1242 m_ghost_dofs.clear();
1243 m_ghost_dof_to_local.clear();
1244 if constexpr (MeshType::parallel()) {
1245 const auto n_owned = m_mesh.nLocEntity(MeshType::self_dim);
1246 const auto n_ghost = m_mesh.nGhostEntity(MeshType::self_dim);
1247 // A dof on a ghost cell can, at a shared partition boundary, also be
1248 // one of this rank's *owned* dofs (renumberMpiSubdomainWise() assigns
1249 // such shared dofs to a single canonical owner, which may be this
1250 // rank). Exclude those: they are already directly readable via the

1256 if (dof_id >= lo && dof_id < hi) {
1257 continue;

1259 const auto [it, inserted] = m_ghost_dof_to_local.emplace(
1260 dof_id, static_cast<index_t>(m_ghost_dofs.size()));

1262 m_ghost_dofs.push_back(dof_id);

1269template <typename MeshType>
1270auto DofHandler<MeshType>::gatherSparsityPatternRank0() const {
1271 using SpPatternType = datastructure::CrsMap<index_t, index_t>;
1272 using LocToNewMap = std::vector<index_t>;
1273 // recv from ranks the sp-pattern and the loc-to-new
1274 std::vector<SpPatternType> sp_patterns(mpi::size());
1275 std::vector<LocToNewMap> loc_to_new_maps(mpi::size());
1276 std::vector<std::vector<index_t>> all_owned_dofs(mpi::size());
1277 const auto& map =
1278 m_has_renumbering ? m_old_to_new_numbering : m_local_to_global_dof;
1279 if (mpi::root()) {
1280 sp_patterns[0] = m_sparsity_pattern;
1281 loc_to_new_maps[0] = map;
1282 all_owned_dofs[0] = m_owned_cells_dofs;
1283 for (int p = 1; p < mpi::size(); ++p) {
1284 mpi::recv(sp_patterns[p], p);
1285 mpi::recv(loc_to_new_maps[p], p);
1286 mpi::recv(all_owned_dofs[p], p);
1287 }
1288 } else {
1289 mpi::send(m_sparsity_pattern, 0);
1290 mpi::send(map, 0);
1291 mpi::send(m_owned_cells_dofs, 0);
1292 }
1293
1294 // for each proc; for each row; if empty -> add; else -> check eq;
1295 datastructure::CrsMapUncompressed<index_t, 128> global_sp_pattern;
1296 if (mpi::root()) {
1297 for (int p = 0; p < mpi::size(); ++p) {
1298 for (index_t loc_dof_id = 0; loc_dof_id < sp_patterns[p].size();
1299 ++loc_dof_id) {
1300 const auto glob_dof_id = loc_to_new_maps[p][loc_dof_id];
1301 auto& glob_row = global_sp_pattern.getRow(glob_dof_id);
1302 const auto loc_row = sp_patterns[p][loc_dof_id];
1303 std::sort(loc_row.begin(), loc_row.end());
1304
1305 // the sparsity pattern computed is complete (correct) only
1306 // if the dof belongs to a cell the proc owns.
1307 if (std::find(
1308 all_owned_dofs[p].begin(), all_owned_dofs[p].end(),
1309 loc_dof_id) != all_owned_dofs[p].end()) {
1310 if (glob_row.size()) {
1311 assert(glob_row.size() == static_cast<std::size_t>(loc_row.size()));
1312 assert(
1313 std::equal(glob_row.begin(), glob_row.end(), loc_row.begin()));
1314 } else {
1315 glob_row.assign(loc_row.begin(), loc_row.end());
1316 }
1317 }
1318 }
1319 }
1320 }
1321 return global_sp_pattern.template compress<index_t>();
1322}
