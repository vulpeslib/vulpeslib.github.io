530void DofHandler<MeshType>::buildCellToCellDof(
531 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
532 // copy the sparsity pattern from the one of `m_cell_to_dof`
533 m_cell_to_cell_dof = m_cell_to_dof;
534 // cell_id is the local id in MPI numeration
535 for (index_t cell_id = 0; cell_id < m_cell_to_cell_dof.size(); ++cell_id) {
536 for (auto& dof_id : m_cell_to_cell_dof[cell_id]) {
537 // expand information of dof_id by exploiting global (geometric)
538 // enumeration
539 index_t global_dof_id = dof_id;
540 if constexpr (MeshType::parallel()) {
541 global_dof_id = m_local_to_global_dof[dof_id];
542 }
543 index_t enty_dim = 0;
544 while (!(global_dof_id < m_glob_cum_ndof_per_entity[enty_dim + 1])) {
545 enty_dim++;
546 }
547 const index_t entity_dof_id =
548 global_dof_id - m_glob_cum_ndof_per_entity[enty_dim];
549 const index_t n_dofs_per_entity =
550 fe_space.getNumDofsForSubdimEntity(enty_dim);
551 const index_t owning_enty_id = entity_dof_id / n_dofs_per_entity;
552 const index_t enty_multiplicity = entity_dof_id % n_dofs_per_entity;
553 // pass the expanded information to the fe space and update dof_id
554 // from global to local numbering (cell based, not MPI partition based)
555 dof_id = fe_space.dofToCellDof(
556 m_mesh, cell_id, owning_enty_id, enty_dim, enty_multiplicity);
557 }
558 }

562template <
563 std::size_t Dim,
564 typename FloatType,
565 typename DofPolicy,
566 typename FeSpace>
567void DofHandler<MeshType>::computeNumDofs(
568 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
569 for (std::size_t d = 0; d < FeSpace::dim + 1; ++d) {
570 const auto n_dofs_per_entity = fe_space.getNumDofsForSubdimEntity(d);
571 const auto n_glob_entities = m_mesh.nEntity(d);
572 auto n_loc_entities = m_mesh.nLocEntity(d);
573 if constexpr (MeshType::parallel()) {
574 n_loc_entities += m_mesh.nGhostEntity(d);
575 }
576 m_glob_cum_ndof_per_entity[d + 1] =
577 m_glob_cum_ndof_per_entity[d] + n_dofs_per_entity * n_glob_entities;
578 m_loc_cum_ndof_per_entity[d + 1] =
579 m_loc_cum_ndof_per_entity[d] + n_dofs_per_entity * n_loc_entities;
580 }

583template <typename MeshType>
584template <
585 std::size_t SubentityDim,
586 std::size_t Dim,
587 typename FloatType,
588 typename DofPolicy,
589 typename FeSpace>
590void DofHandler<MeshType>::distributeSubentityDofs(
591 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space,
592 auto& dof_to_cell) {
593 const index_t n_dofs_per_entity =
594 fe_space.getNumDofsForSubdimEntity(SubentityDim);
595 if (n_dofs_per_entity > 0) {
596 index_t n_loc_entity = m_mesh.nLocEntity(SubentityDim);
597 if constexpr (MeshType::parallel()) {
598 n_loc_entity += m_mesh.nGhostEntity(SubentityDim);
599 }
600 // for each entity in the mesh, may be vertex, edge, facet, cell
601 for (index_t enty_id = 0; enty_id < n_loc_entity; ++enty_id) {
602 // for each dof that belongs to this entity
603 for (index_t j = 0; j < n_dofs_per_entity; ++j) {
604 // compute global dof id
605 index_t global_enty_id = enty_id;
606 if constexpr (MeshType::parallel()) {
607 global_enty_id = m_mesh.localToGlobal(SubentityDim)[enty_id];
608 }
609 const auto glob_dof_id = m_glob_cum_ndof_per_entity[SubentityDim] +
610 global_enty_id * n_dofs_per_entity + j;
611 auto loc_dof_id = glob_dof_id;
612 if constexpr (MeshType::parallel()) {
613 loc_dof_id = m_local_to_global_dof.size();
614 m_local_to_global_dof.push_back(glob_dof_id);
615 m_global_to_local_dof[glob_dof_id] = loc_dof_id;
616 }
617 // put the `dof_id` in the `dof_to_cell map`
618 // different behavior if we are in a cell or not
619 if constexpr (SubentityDim == Dim) {
620 // we are in a cell, we know to which cell this dof belongs to
621 dof_to_cell[loc_dof_id].push_back(global_enty_id);
622 // in parallel we must keep track of owned and shared dofs
623 if constexpr (MeshType::parallel()) {
624 // we own the cell dof if we own the cell
625 if (enty_id < m_mesh.nLocEntity(SubentityDim)) {
626 m_owned_dofs.push_back(glob_dof_id);
627 m_local_owned_dofs.push_back(loc_dof_id);
628 } else {
629 m_shared_dofs.push_back(glob_dof_id);
630 m_local_shared_dofs.push_back(loc_dof_id);
631 }
632 }
633 } else {
634 const auto& enty_to_cell_map =
635 m_mesh.template getEntityToCellGlobalMap<SubentityDim>();
636 // loop over all the cells to which this entity belongs to
637 for (const auto cell_id : enty_to_cell_map[enty_id]) {
638 // facet mappings always have two elements, one of which may be -1
639 // in this case we skip it
640 if constexpr (SubentityDim == Dim - 1) {
641 if (cell_id == static_cast<decltype(cell_id)>(-1)) {
642 continue;
643 }
644 }
645 dof_to_cell[loc_dof_id].push_back(cell_id);
646 // in parallel we must keep track of owned and shared dofs
647 if constexpr (MeshType::parallel()) {
648 // if the dof belongs to an entity that is on a ghost cell
649 // than it is a shared dof
650 const auto local_cell_id = m_mesh.globalToLocal(Dim).at(cell_id);
651 if ((local_cell_id >= m_mesh.nLocEntity(Dim)) &&
652 (m_shared_dofs.empty() ||
653 m_shared_dofs.back() != glob_dof_id)) {
654 m_shared_dofs.push_back(glob_dof_id);
655 m_local_shared_dofs.push_back(loc_dof_id);
656 }
657 }
658 }
659 // if the dof is not shared, than it is owned
660 if constexpr (MeshType::parallel()) {
661 if (!(m_shared_dofs.size() &&
662 (m_shared_dofs.back() == glob_dof_id))) {
663 m_owned_dofs.push_back(glob_dof_id);
664 m_local_owned_dofs.push_back(loc_dof_id);
665 }
666 }
667 }
668 }
669 }
670 }

694template <typename MeshType>
695void DofHandler<MeshType>::sortSharedDofs() {
696 std::vector<index_t> idxs(m_shared_dofs.size());
697 std::iota(idxs.begin(), idxs.end(), 0);
698 std::sort(idxs.begin(), idxs.end(), [this](auto a, auto b) {
699 return m_shared_dofs[a] < m_shared_dofs[b];
700 });
701 // apply sorting permutation
702 std::vector<index_t> tmp(m_shared_dofs.size());
703 for (std::size_t i = 0; i < idxs.size(); ++i) {
704 tmp[i] = m_shared_dofs[idxs[i]];
705 }
706 std::swap(tmp, m_shared_dofs);
707 for (std::size_t i = 0; i < idxs.size(); ++i) {
708 tmp[i] = m_local_shared_dofs[idxs[i]];
709 }
710 std::swap(tmp, m_local_shared_dofs);
711}

713template <typename MeshType>
714void DofHandler<MeshType>::invertDofToCellMappingAndCompress(
715 auto& dof_to_cell) {
716 constexpr std::size_t static_buffer_size =
717 vmath::pow((index_t)7, MeshType::self_dim);
718 datastructure::CrsMapUncompressed<index_t, static_buffer_size> cell_to_dof;
719 auto n_loc_cell = m_mesh.nLocEntity(MeshType::self_dim);
720 if constexpr (MeshType::parallel()) {
721 n_loc_cell += m_mesh.nGhostEntity(MeshType::self_dim);
722 }
723 cell_to_dof.resize(n_loc_cell);
724 for (std::size_t dof_id = 0; dof_id < dof_to_cell.size(); ++dof_id) {
725 // we exploit the fact that the dofs are sorted later
726 // when we compute the sparsity pattern
727 // TODO: is this already sorted if in serial?
728 std::sort(dof_to_cell[dof_id].begin(), dof_to_cell[dof_id].end());
729 for (const auto global_cell_id : dof_to_cell[dof_id]) {
730 auto loc_cell_id = global_cell_id;
731 if constexpr (MeshType::parallel()) {
732 loc_cell_id =
733 m_mesh.globalToLocal(MeshType::self_dim).at(global_cell_id);
734 }
735 cell_to_dof[loc_cell_id].push_back(dof_id);
736 }
737 }
738
739 m_dof_to_cell = dof_to_cell.template compress<index_t>();
740 m_cell_to_dof = cell_to_dof.template compress<index_t>();
741}

743template <typename MeshType>
744void DofHandler<MeshType>::computeOwnedCellDofsAndBndDofs() {
745 const auto max_dof_idx =
746 m_cell_to_dof.indices()[m_mesh.nLocEntity(MeshType::self_dim)];
747 const auto& data = m_cell_to_dof.data();
748 boost::unordered_flat_set<index_t> owned_cells_dof(
749 data.begin(), data.begin() + max_dof_idx);
750 m_owned_cells_dofs.assign(owned_cells_dof.begin(), owned_cells_dof.end());
751
752 // find which dofs lie on the partition
753 // i.e. they are both shared and belong to owned cells
754 for (auto loc_dof_id : m_owned_cells_dofs) {
755 const auto glob_dof_id = m_local_to_global_dof[loc_dof_id];
756 const auto it = std::lower_bound(
757 m_shared_dofs.begin(), m_shared_dofs.end(), glob_dof_id);
758 if ((it != m_shared_dofs.end()) && (*it == glob_dof_id)) {
759 m_partition_bnd_dofs.push_back(glob_dof_id);
760 }
761 }
762 std::sort(m_partition_bnd_dofs.begin(), m_partition_bnd_dofs.end());
763}

765template <typename MeshType>
766template <
767 std::size_t Dim,
768 typename FloatType,
769 typename DofPolicy,
770 typename FeSpace>
771void DofHandler<MeshType>::distributeDofs(
772 const fe::FeSpaceBase<Dim, FloatType, DofPolicy, FeSpace>& fe_space) {
773 // smallest row size that make sense
774 // namely we fill all the space we have in the `SmallVector`
775 constexpr std::size_t static_row_size =
776 sizeof(std::vector<index_t>) / sizeof(index_t);
777 datastructure::CrsMapUncompressed<index_t, static_row_size> dof_to_cell;
778 static_assert(FeSpace::dim == MeshType::self_dim);
779
780 computeNumDofs(fe_space);
781 dof_to_cell.resize(m_loc_cum_ndof_per_entity[FeSpace::dim + 1]);
782 PROFILE_INTERVAL(distributeAllSubentityDofs(fe_space, dof_to_cell));
783
784 if (MeshType::parallel()) {
785 // we sort them because it is easier for rank 0 to use them
786 // when computing the global renumbering
787 PROFILE_INTERVAL(sortSharedDofs());
788 }
789
790 PROFILE_INTERVAL(invertDofToCellMappingAndCompress(dof_to_cell));
791
792 PROFILE_INTERVAL(buildCellToCellDof(fe_space));
793
794 if constexpr (MeshType::parallel()) {
795 PROFILE_INTERVAL(computeOwnedCellDofsAndBndDofs());
796 }
797
798 PROFILE_INTERVAL(computeSparsityPattern<FeSpace>());
799
800 if constexpr (MeshType::parallel()) {
801 m_all_dof_offset.resize(mpi::size() + 1, 0);
802 const auto total_dofs = m_glob_cum_ndof_per_entity.back();
803 for (int p = 0; p < mpi::size(); ++p) {
804 const auto reminder = (p < (total_dofs % mpi::size())) ? 1 : 0;
805 m_all_dof_offset[p + 1] =
806 m_all_dof_offset[p] + total_dofs / mpi::size() + reminder;
807 }
808 }
809}

811template <typename MeshType>
812template <bool DofInteractionDistance>
813std::pair<typename MeshType::index_t, typename MeshType::index_t>
814DofHandler<MeshType>::computeSparsityForDof(
815 typename MeshType::index_t dof_id,
816 typename MeshType::index_t* buffer1,
817 typename MeshType::index_t* buffer2) {
818 static std::array<std::size_t, 2> buffers_size = {0, 0};
819 static std::size_t active_buffer = 0;
820 buffers_size = {0, 0};
821 active_buffer = 0;
822 std::array<index_t*, 2> buffers = {buffer1, buffer2};
823
824 // merge all the dofs that belong to this cell (`dofs_view`) and the
825 // others (`buffer[active_buffer]`) into the inactive buffer (`buffer[1 -
826 // active_buffer]`). We assume the dofs are sorted and we remove
827 // duplicates
828 const auto updateBuffer = [&](auto dofs_view) {
829 const index_t* f1 = dofs_view.data();
830 const index_t* l1 = f1 + dofs_view.size();
831 const index_t* f2 = buffers[active_buffer];
832 const index_t* l2 = f2 + buffers_size[active_buffer];
833 index_t* o = buffers[1 - active_buffer];
834 // WARNING: this is not thread-safe
835 buffers_size[1 - active_buffer] =
836 std::distance(o, std::set_union(f1, l1, f2, l2, o));
837 // at each iteration we swap the source buffer with all the dofs and
838 // the destination buffer where we make the union, to avoid an extra
839 // copy
840 buffers_size[active_buffer] = 0;
841 active_buffer = 1 - active_buffer;
842 };
843
844 // for each cell to which this dof belongs to
845 for (index_t global_cell_id : m_dof_to_cell[dof_id]) {
846 auto local_cell_id = global_cell_id;
847 if constexpr (MeshType::parallel()) {
848 local_cell_id =
849 m_mesh.globalToLocal(MeshType::self_dim).at(global_cell_id);
850 }
851 updateBuffer(m_cell_to_dof[local_cell_id]);
852 // if there is interaction with also the neighbours
853 // e.g. if we are in DG each dof belongs to one cell but there
854 // are face couplings
855 if constexpr (DofInteractionDistance) {
856 for (index_t loc_neigh_cell_id : m_mesh.getDualMap()[local_cell_id]) {
857 updateBuffer(m_cell_to_dof[loc_neigh_cell_id]);
858 }
859 }
860 }
861
862 return {active_buffer, buffers_size[active_buffer]};
863}

865template <typename MeshType>
866template <typename FeSpace>
867void DofHandler<MeshType>::computeSparsityPattern() {
868 static constexpr std::size_t n_max_dofs_per_row = 1048576;
869 static std::array<std::array<index_t, n_max_dofs_per_row>, 2> buffers;
870
871 // Probe a good static size for the uncompressed map.
872 // We compute the sparsity pattern for some random dof ids
873 // and check which is the number of nnz at that row.
874 // We then ask for a static size that is greater than the average
875 constexpr index_t probe_size = 1000;
876 index_t nnz = 16;
877 if (m_dof_to_cell.size() > probe_size) {
878 nnz = 0;
879 std::mt19937 gen(0);
880 std::uniform_int_distribution<uint64_t> dis(0, m_dof_to_cell.size() - 1);
881 for (index_t i = 0; i < probe_size; ++i) {
882 const auto random_dof_id = dis(gen);
883 const auto [active_buffer, buffer_size] =
884 computeSparsityForDof<FeSpace::dofsInteractionDistance()>(
885 random_dof_id, buffers[0].data(), buffers[1].data());
886 nnz += buffer_size;
887 }
888 nnz /= probe_size;
889 }
890 pcout.level(utils::LogLevel::DEBUG)
891 << "Sparsity pattern probing suggested size: " + std::to_string(nnz) +
892 "\n";
893 auto uncompressed_sparsity_pattern =
894 datastructure::getUncompressedCrsMapWithStaticStorageGreaterThan<index_t>(
895 nnz);
896 std::visit(
897 [&](auto& map) { map.resize(m_dof_to_cell.size()); },
898 uncompressed_sparsity_pattern);
899
900 // compute the sparsity pattern for all the dofs
901 for (index_t dof_id = 0; dof_id < m_dof_to_cell.size(); ++dof_id) {
902 const auto [active_buffer, buffer_size] =
903 computeSparsityForDof<FeSpace::dofsInteractionDistance()>(
904 dof_id, buffers[0].data(), buffers[1].data());
905 const auto nnz_dofid = buffer_size;
906 // copy the sparsity pattern from the static buffer to the uncompressed map
907 std::visit(
908 [dof_id, nnz_dofid, active_buffer](auto& map) {
909 map[dof_id].assign(
910 buffers[active_buffer].data(),
911 buffers[active_buffer].data() + nnz_dofid);
912 },
913 uncompressed_sparsity_pattern);
914 }
915
916 // compress the sparsity pattern
917 m_sparsity_pattern = std::visit(
918 [](const auto& map) { return map.template compress<index_t>(); },
919 uncompressed_sparsity_pattern);
920}

923void DofHandler<MeshType>::renumberMpiSubdomainWise() {
924 m_has_renumbering = true;
925 const auto size = mpi::size();
926 // collect the size of owned dofs and the list of shared dofs
927 std::vector<std::size_t> cum_dof_offset(size + 1, 0);
928 m_all_shared_dof.resize(size);
929 m_all_partition_bnd_dofs.resize(size);
930
931 if (mpi::root()) {
932 cum_dof_offset[1] = m_owned_dofs.size();
933 m_all_shared_dof[0] = m_shared_dofs;
934 m_all_partition_bnd_dofs[0] = m_partition_bnd_dofs;
935 for (int p = 1; p < size; ++p) {
936 const auto owned_dof_p = mpi::recv<std::size_t>(p);
937 cum_dof_offset[p + 1] = cum_dof_offset[p] + owned_dof_p;
938 mpi::recv(m_all_shared_dof[p], p);
939 mpi::recv(m_all_partition_bnd_dofs[p], p);
940 }
941 } else {
942 mpi::send(m_owned_dofs.size(), 0);
943 mpi::send(m_shared_dofs, 0);
944 mpi::send(m_partition_bnd_dofs, 0);
945 }
946 // root renumbers the shared dofs and sends them
947 std::vector<index_t> shared_renumbering;
948 m_dof_offset = 0;
949 if (mpi::root()) {
950 // all the shared dofs without duplicates
951 std::vector<index_t> unq_shared_dof;
952 // list of boundary shared dofs "belonging" to proc p
953 // that is a bnd dof is owned by the proc with smallest rank that shares it
954 std::vector<index_t> bnd_dof_belonging_to_p;
955 // global-global old to new numbering
956 m_all_old_to_new_numbering.resize(m_glob_cum_ndof_per_entity.back(), -1);
957 std::vector<index_t> tmp_buffer;
958
959 for (int p = 0; p < size - 1; ++p) {
960 // to be numbered, they belong to p
961 std::set_difference(
962 m_all_partition_bnd_dofs[p].begin(),
963 m_all_partition_bnd_dofs[p].end(), unq_shared_dof.begin(),
964 unq_shared_dof.end(), std::back_inserter(bnd_dof_belonging_to_p));
965 // compute the dof offset taking into account the shared between 0 and p
966 for (int i = p; i < size; ++i) {
967 cum_dof_offset[i + 1] += bnd_dof_belonging_to_p.size();
968 }
969 mpi::send(cum_dof_offset[p + 1], p + 1);
970 const auto offset_without_shared =
971 cum_dof_offset[p + 1] - bnd_dof_belonging_to_p.size();
972 for (std::size_t i = 0; i < bnd_dof_belonging_to_p.size(); ++i) {
973 m_all_old_to_new_numbering[bnd_dof_belonging_to_p[i]] =
974 offset_without_shared + i;
975 }
976 // update unq_shared_dof
977 std::set_union(
978 unq_shared_dof.begin(), unq_shared_dof.end(),
979 m_all_partition_bnd_dofs[p].begin(),
980 m_all_partition_bnd_dofs[p].end(), std::back_inserter(tmp_buffer));
981 std::swap(unq_shared_dof, tmp_buffer);
982 bnd_dof_belonging_to_p.resize(0);
983 tmp_buffer.resize(0);
984 }
985
986 // renumber owning according to offset in sequence for self
987 m_old_to_new_numbering.resize(m_local_to_global_dof.size());
988 for (std::size_t i = 0; i < m_owned_dofs.size(); ++i) {
989 m_old_to_new_numbering[m_global_to_local_dof.at(m_owned_dofs[i])] = i;
990 m_all_old_to_new_numbering[m_owned_dofs[i]] = i;
991 }
992
993 // receive enumeration from all
994 std::vector<index_t> tmp_own_global_old;
995 std::vector<index_t> tmp_own_global_new;
996 for (int p = 1; p < size; ++p) {
997 mpi::recv(tmp_own_global_old, p);
998 mpi::recv(tmp_own_global_new, p);
999 for (std::size_t i = 0; i < tmp_own_global_old.size(); ++i) {
1000 m_all_old_to_new_numbering[tmp_own_global_old[i]] =
1001 tmp_own_global_new[i];
1002 }
1003 }
1004
1005 // find the enumeration for each proc
1006 for (int p = 1; p < size; ++p) {
1007 shared_renumbering.resize(m_all_shared_dof[p].size());
1008 for (std::size_t i = 0; i < m_all_shared_dof[p].size(); ++i) {
1009 shared_renumbering[i] =
1010 m_all_old_to_new_numbering[m_all_shared_dof[p][i]];
1011 }
1012 mpi::send(shared_renumbering, p);
1013 }
1014 // find the enumeration for self
1015 shared_renumbering.resize(m_all_shared_dof[0].size());
1016 for (std::size_t i = 0; i < m_all_shared_dof[0].size(); ++i) {
1017 shared_renumbering[i] =
1018 m_all_old_to_new_numbering[m_all_shared_dof[0][i]];
1019 }
1020 } else {
1021 m_dof_offset = mpi::recv<std::size_t>(0);
1022 // renumber according to offset in sequence
1023 std::vector<index_t> own_global_old;
1024 std::vector<index_t> own_global_new;
1025 m_old_to_new_numbering.resize(m_local_to_global_dof.size());
1026 for (std::size_t i = 0; i < m_owned_dofs.size(); ++i) {
1027 const auto glob_own_id = m_owned_dofs[i];
1028 const auto loc_own_id = m_global_to_local_dof.at(glob_own_id);
1029 const index_t new_glob_id = i + m_dof_offset;
1030 m_old_to_new_numbering[loc_own_id] = new_glob_id;
1031 own_global_old.push_back(glob_own_id);
1032 own_global_new.push_back(new_glob_id);
1033 }
1034 mpi::send(own_global_old, 0);
1035 mpi::send(own_global_new, 0);
1036 mpi::recv(shared_renumbering, 0);
1037 }
1038 // broadcast the offset
1039 m_all_dof_offset.resize(mpi::size() + 1);
1040 m_all_dof_offset.back() = m_glob_cum_ndof_per_entity.back();
1041 for (int p = 0; p < mpi::size(); ++p) {
1042 m_all_dof_offset[p] = m_dof_offset;
1043 mpi::bcast(m_all_dof_offset[p], p);
1044 }
1045 // apply the shared dof renumbering
1046 assert(shared_renumbering.size() == m_shared_dofs.size());
1047 for (std::size_t i = 0; i < m_shared_dofs.size(); ++i) {
1048 m_old_to_new_numbering[m_local_shared_dofs[i]] = shared_renumbering[i];
1049 }
1050 // invert mapping
1051 m_new_to_old_numbering.clear();
1052 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1053 m_new_to_old_numbering[m_old_to_new_numbering[i]] = i;
1054 }
1055}

1057template <typename MeshType>
1058void DofHandler<MeshType>::applyRenumbering(
1059 const RenumberingFunctor<typename MeshType::index_t>& renumbering_func,
1060 bool reversed) {
1061 m_has_renumbering = true;
1062 assert(m_old_to_new_numbering.size() == m_new_to_old_numbering.size());
1063 using SparsityPatternType = decltype(m_sparsity_pattern);
1064 SparsityPatternType* sparsity_pattern_ptr = &m_sparsity_pattern;
1065 std::shared_ptr<SparsityPatternType> owned_sparsity_pattern;
1066 if (MeshType::parallel()) {
1067 owned_sparsity_pattern =
1068 std::make_shared<SparsityPatternType>(computeOwnedSparsityPattern());
1069 sparsity_pattern_ptr = owned_sparsity_pattern.get();
1070 }
1071 const auto [perm, perm_inv] =
1072 renumbering_func(*sparsity_pattern_ptr, reversed);
1073 assert(perm.size() == perm_inv.size());
1074 // apply perm and perm_inv to mappings
1075 if constexpr (MeshType::parallel()) {
1076 const auto size = mpi::size();
1077 // update the old to new for owned dofs
1078 std::vector<index_t> new_owned_dof_id(m_owned_dofs.size());
1079 {
1080 std::vector<index_t> tmp = m_old_to_new_numbering;
1081 for (std::size_t i = 0; i < perm.size(); ++i) {
1082 tmp[m_local_owned_dofs[i]] =
1083 m_old_to_new_numbering[m_local_owned_dofs[perm[i]]];
1084 new_owned_dof_id[i] =
1085 m_old_to_new_numbering[m_local_owned_dofs[perm[i]]];
1086 }
1087 m_old_to_new_numbering = std::move(tmp);
1088 }
1089
1090 std::vector<index_t> new_shared_numbering;
1091 if (mpi::root()) {
1092 // get the new numbering of the dofs owned by each proc
1093 std::vector<index_t> owned_dofs_p = m_owned_dofs;
1094 for (int p = 0; p < size; ++p) {
1095 if (p > 0) {
1096 mpi::recv(owned_dofs_p, p);
1097 mpi::recv(new_owned_dof_id, p);
1098 }
1099 assert(new_owned_dof_id.size() == owned_dofs_p.size());
1100 for (std::size_t i = 0; i < new_owned_dof_id.size(); ++i) {
1101 m_all_old_to_new_numbering[owned_dofs_p[i]] = new_owned_dof_id[i];
1102 }
1103 }
1104
1105 // send new numbering of shared
1106 for (int p = 1; p < size; ++p) {
1107 new_shared_numbering.resize(m_all_shared_dof[p].size());
1108 for (std::size_t i = 0; i < m_all_shared_dof[p].size(); ++i) {
1109 new_shared_numbering[i] =
1110 m_all_old_to_new_numbering[m_all_shared_dof[p][i]];
1111 }
1112 mpi::send(new_shared_numbering, p);
1113 }
1114 // do the same for self
1115 new_shared_numbering.resize(m_shared_dofs.size());
1116 for (std::size_t i = 0; i < m_shared_dofs.size(); ++i) {
1117 new_shared_numbering[i] = m_all_old_to_new_numbering[m_shared_dofs[i]];
1118 }
1119 } else {
1120 mpi::send(m_owned_dofs, 0);
1121 // new glob ids for m_owned_dofs
1122 mpi::send(new_owned_dof_id, 0);
1123 // new enumeration for shared
1124 mpi::recv(new_shared_numbering, 0);
1125 }
1126 // apply new enumeration for shared
1127 assert(m_local_shared_dofs.size() == new_shared_numbering.size());
1128 for (std::size_t i = 0; i < m_local_shared_dofs.size(); ++i) {
1129 m_old_to_new_numbering[m_local_shared_dofs[i]] = new_shared_numbering[i];
1130 }
1131 // invert the old_to_new
1132 m_new_to_old_numbering.clear();
1133 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1134 m_new_to_old_numbering[m_old_to_new_numbering[i]] = i;
1135 }
1136 } else {
1137 if (m_old_to_new_numbering.size() != perm.size()) {
1138 m_old_to_new_numbering = std::move(perm);
1139 m_new_to_old_numbering = std::move(perm_inv);
1140 } else {
1141 for (std::size_t i = 0; i < m_old_to_new_numbering.size(); ++i) {
1142 m_old_to_new_numbering[i] = perm[m_old_to_new_numbering[i]];
1143 m_new_to_old_numbering[m_old_to_new_numbering[i]] = i;
1144 }
1145 }
1146 }

1149template <typename MeshType>
1150auto DofHandler<MeshType>::computeOwnedSparsityPattern() {
1151 if (MeshType::parallel()) {
1152 // build inverse map (from the local id of an owned dof to the owned id)
1153 const auto n_local_dofs = m_owned_dofs.size() + m_shared_dofs.size();
1154 std::vector<index_t> loc_owned_dofs_inv(n_local_dofs, -1);
1155 for (std::size_t i = 0; i < m_local_owned_dofs.size(); ++i) {
1156 loc_owned_dofs_inv[m_local_owned_dofs[i]] = i;
1157 }
1158 //
1159 std::decay_t<decltype(m_sparsity_pattern)> owned_map({}, {});
1160 const auto& indices = m_sparsity_pattern.indices();
1161 const auto& data = m_sparsity_pattern.data();
1162 owned_map.indices().push_back(0);
1163 for (const auto own_loc_dof_id : m_local_owned_dofs) {
1164 const auto row_start = indices[own_loc_dof_id];
1165 const index_t row_size = indices[own_loc_dof_id + 1] - row_start;
1166 index_t real_size = 0;
1167 for (index_t j = 0; j < row_size; ++j) {
1168 const index_t new_glob_id = data[row_start + j];
1169 const index_t loc_id = m_new_to_old_numbering[new_glob_id];
1170 const index_t old_glob_id = m_local_to_global_dof[loc_id];
1171 const auto it = std::lower_bound(
1172 m_shared_dofs.begin(), m_shared_dofs.end(), old_glob_id);
1173 // if it is not shared, than it is owned
1174 // shared are many less, thus it is quicker than to search in owned
1175 if ((it == m_shared_dofs.end()) || (*it != old_glob_id)) {
1176 owned_map.data().push_back(loc_owned_dofs_inv[loc_id]);
1177 real_size++;
1178 }
1179 }
1180 owned_map.indices().push_back(real_size + owned_map.indices().back());
1181 }
1182 return owned_map;
1183 } else {
1184 return m_sparsity_pattern;
1185 }

1193 for (auto& col : m_sparsity_pattern.data()) {

1197 for (auto& col : m_sparsity_pattern.data()) {

1209 // in non-discontinous elements)
1210 assert(m_cell_to_cell_dof.size() == m_cell_to_dof.size());
1211 std::vector<index_t> tmp_buffer;
1212 for (index_t cell_id = 0; cell_id < m_cell_to_dof.size(); ++cell_id) {
1213 assert(m_cell_to_dof[cell_id].size() == m_cell_to_cell_dof[cell_id].size());
1214 const auto row_size = m_cell_to_dof[cell_id].size();
1215 if (tmp_buffer.size() < static_cast<std::size_t>(row_size)) {
1216 tmp_buffer.resize(row_size);
1217 }
1218 for (index_t i = 0; i < row_size; ++i) {
1219 tmp_buffer[m_cell_to_cell_dof[cell_id][i]] = m_cell_to_dof[cell_id][i];
1220 }
1221 std::copy(
1222 tmp_buffer.begin(), tmp_buffer.begin() + row_size,
1223 m_cell_to_dof[cell_id].begin());
1224 }
1225
1226 // apply the local to global to cell-to-dof
1227 if (m_has_renumbering) {
1228 for (auto& loc_dof : m_cell_to_dof.data()) {
1229 loc_dof = m_old_to_new_numbering[loc_dof];
1230 }
1231 } else {
1232 if constexpr (MeshType::parallel()) {
1233 for (auto& loc_dof : m_cell_to_dof.data()) {
1234 loc_dof = m_local_to_global_dof[loc_dof];
1235 }
1236 }
1237 }
1238
1239 buildGhostDofs();

1242template <typename MeshType>
1243void DofHandler<MeshType>::buildGhostDofs() {
1244 m_ghost_dofs.clear();
1245 m_ghost_dof_to_local.clear();

1271template <typename MeshType>
1272auto DofHandler<MeshType>::gatherSparsityPatternRank0() const {
1273 using SpPatternType = datastructure::CrsMap<index_t, index_t>;
1274 using LocToNewMap = std::vector<index_t>;
1275 // recv from ranks the sp-pattern and the loc-to-new
1276 std::vector<SpPatternType> sp_patterns(mpi::size());
1277 std::vector<LocToNewMap> loc_to_new_maps(mpi::size());
1278 std::vector<std::vector<index_t>> all_owned_dofs(mpi::size());
1279 const auto& map =
1280 m_has_renumbering ? m_old_to_new_numbering : m_local_to_global_dof;
1281 if (mpi::root()) {
1282 sp_patterns[0] = m_sparsity_pattern;
1283 loc_to_new_maps[0] = map;
1284 all_owned_dofs[0] = m_owned_cells_dofs;
1285 for (int p = 1; p < mpi::size(); ++p) {
1286 mpi::recv(sp_patterns[p], p);
1287 mpi::recv(loc_to_new_maps[p], p);
1288 mpi::recv(all_owned_dofs[p], p);
1289 }
1290 } else {
1291 mpi::send(m_sparsity_pattern, 0);
1292 mpi::send(map, 0);
1293 mpi::send(m_owned_cells_dofs, 0);
1294 }
1295
1296 // for each proc; for each row; if empty -> add; else -> check eq;
1297 datastructure::CrsMapUncompressed<index_t, 128> global_sp_pattern;
1298 if (mpi::root()) {
1299 for (int p = 0; p < mpi::size(); ++p) {
1300 for (index_t loc_dof_id = 0; loc_dof_id < sp_patterns[p].size();
1301 ++loc_dof_id) {
1302 const auto glob_dof_id = loc_to_new_maps[p][loc_dof_id];
1303 auto& glob_row = global_sp_pattern.getRow(glob_dof_id);
1304 const auto loc_row = sp_patterns[p][loc_dof_id];
1305 std::sort(loc_row.begin(), loc_row.end());
1306
1307 // the sparsity pattern computed is complete (correct) only
1308 // if the dof belongs to a cell the proc owns.
1309 if (std::find(
1310 all_owned_dofs[p].begin(), all_owned_dofs[p].end(),
1311 loc_dof_id) != all_owned_dofs[p].end()) {
1312 if (glob_row.size()) {
1313 assert(glob_row.size() == static_cast<std::size_t>(loc_row.size()));
1314 assert(
1315 std::equal(glob_row.begin(), glob_row.end(), loc_row.begin()));
1316 } else {
1317 glob_row.assign(loc_row.begin(), loc_row.end());
1318 }
1319 }
1320 }
1321 }
1322 }
1323 return global_sp_pattern.template compress<index_t>();
1324}
