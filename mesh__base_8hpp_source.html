131template <typename VecType>
132void checkEmptySets(const VecType& labels) {
133 std::vector<bool> class_found;
134 for (size_t i = 0; i < labels.size(); i++) {
135 if (class_found.size() <= labels[i]) {
136 class_found.resize(labels[i] + 1, false);
137 }
138 class_found[labels[i]] = true;
139 }
140 for (size_t i = 0; i < class_found.size(); i++) {
141 if (!class_found[i]) {
142 std::cerr << "WARNING: The class " << i << " of " << class_found.size()
143 << " is empty!" << std::endl;
144 }
145 }
146}

163template <typename index_t>
164datastructure::CrsMap<index_t, index_t> buildSubDual(
165 const datastructure::CrsMap<index_t, index_t>& dual_map,
166 const std::vector<index_t>& subset_idxs) {
167 using IndMap = datastructure::CrsMap<index_t, index_t>;
168 using IndVec = std::vector<index_t>;
169
170 // sort subset_idxs
171 IndVec tmp;
172 const IndVec* subset_idxs_sorted = &subset_idxs;
173 if (!std::is_sorted(subset_idxs.begin(), subset_idxs.end())) {
174 tmp = subset_idxs;
175 std::sort(tmp.begin(), tmp.end());
176 subset_idxs_sorted = &tmp;
177 }
178
179 // compute the old to new cell id map
180 index_t n_nodes = dual_map.size();
181 IndVec old_to_new(n_nodes, -1);
182 const index_t n_subset_idxs = subset_idxs.size();
183 for (index_t new_id = 0; new_id < n_subset_idxs; ++new_id) {
184 old_to_new[(*subset_idxs_sorted)[new_id]] = new_id;
185 }
186
187 // extract the dual map of the subset directly in compressed form!
188 IndMap sub_dual_map;
189 sub_dual_map.data().resize(0);
190 sub_dual_map.indices().resize(0);
191 sub_dual_map.indices().push_back(0);
192 for (index_t ncid = 0; ncid < n_subset_idxs; ++ncid) {
193 const auto ocid = (*subset_idxs_sorted)[ncid];
194 for (const auto neigh_ocid : dual_map[ocid]) {
195 const auto it = std::lower_bound(
196 (*subset_idxs_sorted).begin(), (*subset_idxs_sorted).end(),
197 neigh_ocid);
198 if (it != (*subset_idxs_sorted).end() && (*it) == neigh_ocid) {
199 sub_dual_map.data().push_back(old_to_new[neigh_ocid]);
200 }
201 }
202 sub_dual_map.indices().push_back(sub_dual_map.data().size());
203 }
204 return sub_dual_map;
205};

222template <typename index_t>
223index_t removeEmptySets(std::vector<index_t>& labels) {
224 constexpr auto npos = static_cast<index_t>(-1);
225 if (labels.empty()) {
226 return 0;
227 }
228 // NOTE: we need much less memory than labels.size()
229 // we assume that few labels are missing
230 const index_t max_label_id = *std::max_element(labels.begin(), labels.end());
231 std::vector<index_t> old_to_new(max_label_id + 1, npos);
232 index_t new_id = 0;
233 for (std::size_t i = 0; i < labels.size(); i++) {
234 if (old_to_new[labels[i]] == npos) {
235 old_to_new[labels[i]] = new_id++;
236 }
237 labels[i] = old_to_new[labels[i]];
238 }
239 return new_id;
240}

254template <typename IndexType>
255void importVtkMeshTags(
256 vtkFieldData* field_data,
257 std::string_view association,
258 std::unordered_map<std::string, std::vector<IndexType>>& tags) {
259 const vtkIdType n_arrays = field_data->GetNumberOfArrays();
260 for (vtkIdType i = 0; i < n_arrays; ++i) {
261 vtkAbstractArray* abstract_array = field_data->GetAbstractArray(i);
262 const char* raw_name =
263 abstract_array == nullptr ? nullptr : abstract_array->GetName();
264 const std::string array_name =
265 raw_name != nullptr && *raw_name != '\0'
266 ? raw_name
267 : "<unnamed array at index " + std::to_string(i) + ">";
268
269 vtkDataArray* data_array = vtkDataArray::SafeDownCast(abstract_array);
270 if (data_array == nullptr) {
271 throw std::runtime_error(
272 "Cannot import " + std::string(association) + " mesh-tag array \"" +
273 array_name + "\": the array is nonnumeric");
274 }
275
276 const int n_components = data_array->GetNumberOfComponents();
277 if (n_components != 1) {
278 // TODO: Support multicomponent mesh tags once the mesh-tag
279 // representation can retain all components.
280 std::cerr << "WARNING: Skipping " << association << " mesh-tag array \""
281 << array_name << "\" because it has " << n_components
282 << " components; only scalar arrays are supported.\n";
283 continue;
284 }
285
286 const int data_type = data_array->GetDataType();
287 if (data_type == VTK_FLOAT || data_type == VTK_DOUBLE) {
288 std::cerr << "WARNING: Converting floating-point " << association
289 << " mesh-tag array \"" << array_name
290 << "\" to an integer type; fractional values may be "
291 "truncated.\n";
292 }
293
294 auto& tag = tags[array_name];
295 const vtkIdType n_tuples = data_array->GetNumberOfTuples();
296 tag.reserve(static_cast<std::size_t>(n_tuples));
297 for (vtkIdType j = 0; j < n_tuples; ++j) {
298 tag.push_back(static_cast<IndexType>(data_array->GetComponent(j, 0)));
299 }
300 }
301}

760datastructure::CrsMap<IndexType, IndexType>
761MeshBase<FloatType, IndexType, OuterDim, SelfDim, Derived>::
762 invertCellAgglomeration(
763 const std::vector<IndexType>& old_to_new_cell) const {
764 datastructure::CrsMapUncompressed<IndexType, 64> new_to_old_cell_uncompr;
765 for (std::size_t cell_id = 0; cell_id < old_to_new_cell.size(); ++cell_id) {
766 const auto cell_id_new = old_to_new_cell[cell_id];
767 if (cell_id_new == UNDEF) {
768 continue;
769 }
770 new_to_old_cell_uncompr.getRow(cell_id_new).push_back(cell_id);
771 }
772 return new_to_old_cell_uncompr.template compress<IndexType>();
773}

775template <
776 typename FloatType,
777 typename IndexType,
778 IndexType OuterDim,
779 IndexType SelfDim,
780 typename Derived>
781std::vector<IndexType>
782MeshBase<FloatType, IndexType, OuterDim, SelfDim, Derived>::
783 buildNewToOldFacetAgglomerationAndCellToFacet(
784 const auto& new_to_old_cell, Derived& agg_mesh) const {
785 std::vector<IndexType> new_to_old_facet;
786 std::vector<IndexType> old_to_new_facet(m_facet_to_cell.size(), -1);
787 datastructure::CrsMapUncompressed<IndexType, 64> new_cell_to_facets_uncpr;
788 new_cell_to_facets_uncpr.resize(new_to_old_cell.size());
789 // NOTE: a flat vector of booleans here performed much worse when tested
790 boost::unordered_flat_map<IndexType, IndexType> facet_table;
791 for (IndexType new_cid = 0; new_cid < new_to_old_cell.size(); ++new_cid) {
792 for (const auto old_cid : new_to_old_cell[new_cid]) {
793 for (const auto old_fid : m_cell_to_facet[old_cid]) {
794 // we only care if the count is only exactly 1
795 // it can be only 0 [face does not belong to cell]
796 // 1 [face is on boundary]
797 // 2 [face is internal] -> not(not(0)) == 0
798 facet_table[old_fid] = !facet_table[old_fid];
799 }
800 }
801 for (auto it = facet_table.begin(); it != facet_table.end(); ++it) {
802 const auto count = it->second;
803 const auto old_fid = it->first;
804 if (count) {
805 IndexType new_fid = old_to_new_facet[old_fid];
806 if (new_fid == UNDEF) {
807 new_fid = new_to_old_facet.size();
808 old_to_new_facet[old_fid] = new_fid;
809 new_to_old_facet.push_back(old_fid);
810 }
811 new_cell_to_facets_uncpr[new_cid].push_back(new_fid);
812 }
813 }
814 facet_table.clear();
815 }
816 agg_mesh.m_cell_to_facet =
817 new_cell_to_facets_uncpr.template compress<IndexType>();
818 return new_to_old_facet;
819}

821template <
822 typename FloatType,
823 typename IndexType,
824 IndexType OuterDim,
825 IndexType SelfDim,
826 typename Derived>
827auto MeshBase<FloatType, IndexType, OuterDim, SelfDim, Derived>::
828 buildFacetToCellAndFacetToVtxAgglomeration(
829 const std::vector<IndexType>& new_to_old_facet,
830 const std::vector<IndexType>& old_to_new_cell,
831 Derived& agg_mesh) const {
832 std::vector<IndexType> new_to_old_vtx;
833 std::vector<IndexType> old_to_new_vtx(m_vertexes.size(), UNDEF);
834
835 agg_mesh.m_facet_to_cell.resize(new_to_old_facet.size(), {UNDEF, UNDEF});
836 agg_mesh.m_facet_to_vtx.indices().push_back(0);
837
838 for (std::size_t new_fid = 0; new_fid < new_to_old_facet.size(); ++new_fid) {
839 const auto old_fid = new_to_old_facet[new_fid];
840 // face to cell
841 for (std::size_t d = 0; d < 2; ++d) {
842 const auto old_cid = m_facet_to_cell[old_fid][d];
843 if (old_cid != UNDEF) {
844 const auto new_cid = old_to_new_cell[old_cid];
845 assert(new_cid != UNDEF);
846 agg_mesh.m_facet_to_cell[new_fid][d] = new_cid;
847 }
848 }
849
850 // facet to vertex and vertex to vertex (old/new)
851 for (const auto old_vid : m_facet_to_vtx[old_fid]) {
852 IndexType new_vid = old_to_new_vtx[old_vid];
853 if (new_vid == UNDEF) {
854 new_vid = new_to_old_vtx.size();
855 new_to_old_vtx.push_back(old_vid);
856 old_to_new_vtx[old_vid] = new_vid;
857 }
858 agg_mesh.m_facet_to_vtx.data().push_back(new_vid);
859 }
860 if constexpr (vtx_per_face == IndexType(-1)) {
861 agg_mesh.m_facet_to_vtx.indices().push_back(
862 agg_mesh.m_facet_to_vtx.data().size());
863 }
864 }
865
866 // add vtxs
867 for (std::size_t ovid = 0; ovid < old_to_new_vtx.size(); ++ovid) {
868 const auto nvid = old_to_new_vtx[ovid];
869 if (nvid != UNDEF) {
870 if (agg_mesh.m_vertexes.size() < static_cast<std::size_t>(nvid + 1)) {
871 agg_mesh.m_vertexes.resize(nvid + 1);
872 }
873 agg_mesh.m_vertexes[nvid] = m_vertexes[ovid];
874 }
875 }
876
877 return old_to_new_vtx;
878}

880template <
881 typename FloatType,
882 typename IndexType,
883 IndexType OuterDim,
884 IndexType SelfDim,
885 typename Derived>
886void MeshBase<FloatType, IndexType, OuterDim, SelfDim, Derived>::
887 buildCellToVtxAgglomeration(
888 const auto& new_to_old_cell,
889 const auto& old_to_new_vtx,
890 Derived& agg_mesh) const {
891 agg_mesh.m_cell_to_vtx.indices().push_back(0);
892 std::vector<bool> is_vtx_present(old_to_new_vtx.size(), false);
893
894 for (IndexType new_cid = 0; new_cid < new_to_old_cell.size(); ++new_cid) {
895 for (const auto old_cid : new_to_old_cell[new_cid]) {
896 for (const auto old_vid : m_cell_to_vtx[old_cid]) {
897 const auto new_vid = old_to_new_vtx[old_vid];
898 if (new_vid != UNDEF && (!is_vtx_present[old_vid])) {
899 agg_mesh.m_cell_to_vtx.data().push_back(new_vid);
900 is_vtx_present[old_vid] = true;
901 }
902 }
903 }
904 std::fill(is_vtx_present.begin(), is_vtx_present.end(), false);
905 agg_mesh.m_cell_to_vtx.indices().push_back(

910template <
911 typename FloatType,
912 typename IndexType,
913 IndexType OuterDim,
914 IndexType SelfDim,
915 typename Derived>
916void MeshBase<FloatType, IndexType, OuterDim, SelfDim, Derived>::
917 agglomerateCheckConnected(const datastructure::CrsMap<IndexType, IndexType>&
918 new_to_old_cell) const {
919 // safety check: check each new cell is made by connected components
920 std::vector<IndexType> glob_to_loc(numCells(), UNDEF);
921 IndexType prev_new_cid = UNDEF;
922 for (IndexType new_cid = 0; new_cid < new_to_old_cell.size(); ++new_cid) {
923 if (new_to_old_cell[new_cid].size() <= 1) {
924 continue;
925 }
926 std::vector<IndexType> old_cids(
927 new_to_old_cell[new_cid].begin(), new_to_old_cell[new_cid].end());
928 // overwrite old map entries
929 if (VULPES_UNLIKELY(prev_new_cid != UNDEF)) {
930 for (auto id : new_to_old_cell[prev_new_cid]) {
931 glob_to_loc[id] = UNDEF;
932 }
933 }
934 prev_new_cid = new_cid;
935 // update map
936 for (std::size_t i = 0; i < old_cids.size(); ++i) {
937 glob_to_loc[old_cids[i]] = i;
938 }
939 // dfs to check it's connected
940 std::vector<bool> visited(old_cids.size(), false);
941 std::vector<IndexType> to_visit = {old_cids[0]};
942 while (!to_visit.empty()) {
943 const auto curr = to_visit.back();
944 visited[glob_to_loc[curr]] = true;
945 to_visit.pop_back();
946 for (const auto neigh : m_dual_graph[curr]) {
947 if (glob_to_loc[neigh] == UNDEF) {
948 continue;
949 }
950 if (!visited[glob_to_loc[neigh]]) {
951 to_visit.push_back(neigh);
952 }
953 }
954 }
955 // check all are visited
956 for (std::size_t i = 0; i < visited.size(); ++i) {
957 if (!visited[i]) {
958 std::cout << "ERROR: in new cell " + std::to_string(new_cid) +
959 ", starting from old cell " +
960 std::to_string(old_cids[0]) +
961 " is not possible to reach the old cell " +
962 std::to_string(old_cids[i])
963 << std::endl;
964
965 for (auto cid : {old_cids[0], old_cids[i]}) {
966 SelfPoint avg = SelfPoint::Zero();
967 for (const auto id : m_cell_to_vtx[cid]) {
968 avg += m_vertexes[id];
969 }
970 avg /= m_cell_to_vtx[cid].size();
971 std::cout << "Old cell " << cid << " avg: " << avg.transpose()
972 << std::endl;
973 }
974 throw std::runtime_error("agglomerated cell is not connected");
975 }
976 }
977 }
