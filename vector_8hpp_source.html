100 template <typename DofHandler>
101 void init(const DofHandler& dof_handler, PetscInt block_size) {
102 createVector();
103 if (block_size > 1) {
104 PetscCallAbortWorld(VecSetBlockSize(m_petsc_vector, block_size));
105 }
106 const PetscInt n_glob = dof_handler.ndofsGlobal();
107 PetscInt dof_begin = 0;
108 PetscInt dof_end = n_glob;
109 if constexpr (DofHandler::parallel) {
110 dof_begin = dof_handler.dofOffset()[mpi::rank()];
111 dof_end = dof_handler.dofOffset()[mpi::rank() + 1];
112 }
113 const PetscInt n_loc = dof_end - dof_begin;
114 setVectorSizes(n_loc * block_size, n_glob * block_size);
115 setupVector();
116 }

120 template <typename DofHandler>
121 void initGhosted(const DofHandler& dof_handler) {
122 static_assert(std::is_same_v<typename DofHandler::index_t, PetscInt>);
123 const auto& ghost_dofs = dof_handler.getGhostDofs();
124 const PetscInt n_glob = dof_handler.ndofsGlobal();
125 const PetscInt n_owned = dof_handler.nOwnedDofs();
126 const PetscInt n_ghost = static_cast<PetscInt>(ghost_dofs.size());
127 PetscCallAbortWorld(VecCreateGhost(
128 PETSC_COMM_WORLD, n_owned, n_glob, n_ghost, ghost_dofs.data(),
129 &m_petsc_vector));
130 m_is_ghosted = true;
131 }
