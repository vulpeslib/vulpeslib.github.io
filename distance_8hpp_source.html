195template <typename FloatType>
196class PolygonDistance
197 : public DistanceExpression<FloatType, 2, PolygonDistance<FloatType>> {
198 public:
200 static constexpr size_t max_edges = 2000;
201
206 static constexpr bool is_leaf = true;
213 static constexpr size_t size() { return max_edges + 1; }
214
224 explicit PolygonDistance(const std::vector<Point<FloatType, 2>>& vertices)
225 : m_vertices(vertices) {
226 if (m_vertices.size() < 3) {
227 throw std::invalid_argument(
228 "A polygon must have at least three vertices");
229 }
230 if (m_vertices.size() > max_edges) {
231 throw std::invalid_argument("Polygon has too many edges");
232 }
233 // Determine the vertex ordering so every edge distance points outward.
234 FloatType signed_area = 0.;
235 for (size_t i = 0; i < m_vertices.size(); ++i) {
236 const auto& start = m_vertices[i];
237 const auto& end = m_vertices[(i + 1) % m_vertices.size()];
238 signed_area += start[0] * end[1] - end[0] * start[1];
239 }
240 if (signed_area == FloatType(0.)) {
241 throw std::invalid_argument("Polygon has zero area");
242 }
243 if (hasIntersectingEdges()) {
244 throw std::invalid_argument("Polygon has edges that cross or touch");
245 }
246 m_outward_sign = signed_area > 0. ? -1. : 1.;
247 }
248
256 void operator()(
257 const Point<FloatType, 2>& point, FloatType* distance_buffer) const {
258 // Find the minimum Euclidean distance from the point to the polygon edges.
259 const FloatType infinity = std::numeric_limits<FloatType>::max();
260 FloatType min_distance = infinity;
261 bool point_on_boundary = false;
262 bool point_inside = false;
263
264 // Unused edge slots must not be considered by VulPoly.
265 std::fill(distance_buffer, distance_buffer + max_edges, infinity);
266
267 // Process every edge, including the closing edge from the last vertex to
268 // the first one.
269 for (size_t i = 0; i < m_vertices.size(); ++i) {
270 const auto& start = m_vertices[i];
271 const auto& end = m_vertices[(i + 1) % m_vertices.size()];
272 const FloatType dx = end[0] - start[0];
273 const FloatType dy = end[1] - start[1];
274 const FloatType edge_length_squared = dx * dx + dy * dy;
275
276 // Project the point onto the edge and clamp the projection to the
277 // segment endpoints.
278 FloatType projection = 0.;
279 if (edge_length_squared > 0.) {
280 projection = ((point[0] - start[0]) * dx + (point[1] - start[1]) * dy) /
281 edge_length_squared;
282 projection = std::clamp(projection, FloatType(0.), FloatType(1.));
283 }
284
285 const FloatType closest_x = start[0] + projection * dx;
286 const FloatType closest_y = start[1] + projection * dy;
287 const FloatType diff_x = point[0] - closest_x;
288 const FloatType diff_y = point[1] - closest_y;
289 const FloatType edge_distance =
290 std::sqrt(diff_x * diff_x + diff_y * diff_y);
291 min_distance = std::min(min_distance, edge_distance);
292
293 // Store the signed distance to the line through this edge.
294 const FloatType cross =
295 dx * (point[1] - start[1]) - dy * (point[0] - start[0]);
296 distance_buffer[i] =
297 edge_length_squared > 0.
298 ? m_outward_sign * cross / std::sqrt(edge_length_squared)
299 : infinity;
300
301 // Use a scale-aware tolerance to classify points very close to an edge
302 // as boundary points despite floating-point roundoff.
303 const FloatType scale = std::max(
304 {FloatType(1.), std::abs(point[0]), std::abs(point[1]),
305 std::abs(start[0]), std::abs(start[1]), std::abs(end[0]),
306 std::abs(end[1])});
307 if (edge_distance <=
308 FloatType(8.) * std::numeric_limits<FloatType>::epsilon() * scale) {
309 point_on_boundary = true;
310 }
311
312 // Ray casting: each crossing of the horizontal ray toggles the
313 // inside/outside state. An odd number of crossings means inside.
314 if ((start[1] > point[1]) != (end[1] > point[1]) &&
315 point[0] < (end[0] - start[0]) * (point[1] - start[1]) /
316 (end[1] - start[1]) +
317 start[0]) {
318 point_inside = !point_inside;
319 }
320 }
321
322 // Inside and boundary points are negative; outside points are positive.
323 const FloatType signed_distance =
324 point_on_boundary || point_inside ? -min_distance : min_distance;
325 distance_buffer[max_edges] = signed_distance;
326 }
327
333 std::array<Point<FloatType, 2>, 2> bbox() const {
334 Point<FloatType, 2> lower = m_vertices.front();
335 Point<FloatType, 2> upper = m_vertices.front();
336 for (const auto& vertex : m_vertices) {
337 lower[0] = std::min(lower[0], vertex[0]);
338 lower[1] = std::min(lower[1], vertex[1]);
339 upper[0] = std::max(upper[0], vertex[0]);
340 upper[1] = std::max(upper[1], vertex[1]);
341 }
342 return {lower, upper};
343 }
344
345 private:
347 bool hasIntersectingEdges() const {
348 using Point2 = Point<FloatType, 2>;
349 const auto orientation = [](const Point2& a, const Point2& b,
350 const Point2& c) {
351 return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
352 };
353 // For c collinear with a and b: true if c is on the segment between them
354 const auto onSegment = [](const Point2& a, const Point2& b,
355 const Point2& c) {
356 return std::min(a[0], b[0]) <= c[0] && c[0] <= std::max(a[0], b[0]) &&
357 std::min(a[1], b[1]) <= c[1] && c[1] <= std::max(a[1], b[1]);
358 };
359 const size_t n = m_vertices.size();
360 for (size_t i = 0; i < n; ++i) {
361 const auto& a = m_vertices[i];
362 const auto& b = m_vertices[(i + 1) % n];
363 for (size_t j = i + 2; j < n; ++j) {
364 // the first and the last edges share a vertex
365 if (i == 0 && j == n - 1) {
366 continue;
367 }
368 const auto& c = m_vertices[j];
369 const auto& d = m_vertices[(j + 1) % n];
370 const FloatType side_c = orientation(a, b, c);
371 const FloatType side_d = orientation(a, b, d);
372 const FloatType side_a = orientation(c, d, a);
373 const FloatType side_b = orientation(c, d, b);
374 if (side_c * side_d < 0. && side_a * side_b < 0.) {
375 return true;
376 }
377 if ((side_c == 0. && onSegment(a, b, c)) ||
378 (side_d == 0. && onSegment(a, b, d)) ||
379 (side_a == 0. && onSegment(c, d, a)) ||
380 (side_b == 0. && onSegment(c, d, b))) {
381 return true;
382 }
383 }
384 }
385 return false;
386 }
387
388 const std::vector<Point<FloatType, 2>>
389 m_vertices;
390 FloatType m_outward_sign;
391};

401template <typename FloatType>
402class CircleDistance

404 public:
409 static constexpr bool is_leaf = true;

416 static constexpr size_t size() { return 2; }
417

425 : m_center(center), m_radius(radius) {}
426
434 void operator()(
435 const Point<FloatType, 2>& point, FloatType* distance_buffer) const {
436 const FloatType dx = point[0] - m_center[0];

602template <typename FloatType, int32_t Dim, typename D1, typename D2>
603class UnionDistance : public DistanceExpression<
604 FloatType,
605 Dim,
606 UnionDistance<FloatType, Dim, D1, D2>> {
609 typename std::conditional_t<D1::is_leaf, const D1&, const D1> m_u;
612 typename std::conditional_t<D2::is_leaf, const D2&, const D2> m_v;
613
614 public:
615 static constexpr bool is_leaf = false;
616
623 UnionDistance(D1 const& u, D2 const& v) : m_u(u), m_v(v) {}
624
633 static constexpr size_t size() {
634 return std::decay_t<decltype(m_v)>::size() +
635 std::decay_t<decltype(m_u)>::size() - 1;
636 }
637
648 void operator()(
649 const Point<FloatType, Dim>& point, FloatType* distance_buffer) const {
650 constexpr auto u_distance_size = std::decay_t<decltype(m_u)>::size();
651 constexpr auto this_size = std::decay_t<decltype(*this)>::size();
652 m_u(point, distance_buffer);
653 FloatType tmp = distance_buffer[u_distance_size - 1];
654 m_v(point, distance_buffer + u_distance_size - 1);
655 distance_buffer[this_size - 1] =
656 std::min(tmp, distance_buffer[this_size - 1]);
657 }
658
666 std::array<Point<FloatType, Dim>, 2> bbox() const {
667 const auto bbox_u = m_u.bbox();
668 const auto bbox_v = m_v.bbox();
669 Point<FloatType, Dim> corner_low;
670 Point<FloatType, Dim> corner_high;
671 for (size_t d = 0; d < Dim; ++d) {
672 corner_low[d] = std::min(bbox_u[0][d], bbox_v[0][d]);
673 corner_high[d] = std::max(bbox_u[1][d], bbox_v[1][d]);
674 }
675 return {corner_low, corner_high};
676 }
677};
